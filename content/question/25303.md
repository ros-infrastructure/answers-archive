  body_markdown: "Hello, \n\nyou should:\n - create a `gazebo::transport::Node` to\
    \ subscribe the Gazebo topic\n - create a ROS 2 node and create the gith publisher.\
    \ \n - In the callback you need to fill the ROS 2 message with the Gazebo IMU\
    \ message.\n\nHere you can a find a snippet of how to implement this idea.\n\n\
    \    gazebo_ros::Node::SharedPtr ros_node;\n    \n    rclcpp::Publisher<sensor_msgs::msg::Imu>::SharedPtr\
    \ imu_pub;\n    \n    gazebo::transport::NodePtr gz_node;\n    \n    // Function\
    \ is called everytime a message is received.\n    void cb(ConstImuPtr &_msg)\n\
    \    {\n      gazebo_msgs::msg::IMU imu_msg;\n     \n     // fill imu_msg message\
    \ based on _msg\n     \n      imu_pub->publish(imu_msg);\n    \n    }\n    \n\
    \    /////////////////////////////////////////////////\n    int main(int argc,\
    \ char ** argv)\n    {\n      // Load gazebo\n      gazebo::client::setup(argc,\
    \ argv);\n    \n      if (!rclcpp::is_initialized()) {\n        rclcpp::init(argc,\
    \ argv);\n        ros_node = gazebo_ros::Node::Get();\n      } else {\n      \
    \  ros_node = gazebo_ros::Node::Get();\n      }\n    \n      imu_pub = ros_node->create_publisher<sensor_msgs::msg::Imu>(\"\
    /turtlebot/rack/imu_sensor/imu\", rclcpp::SensorDataQoS(rclcpp::KeepLast(1)));\n\
    \    \n      // Gazebo transport\n      gz_node = gazebo::transport::NodePtr(new\
    \ gazebo::transport::Node());\n      gz_node->Init();\n    \n      // Listen to\
    \ Gazebo world_stats topic\n      gazebo::transport::SubscriberPtr sub = gz_node->Subscribe(\"\
    ~/turtlebot/rack/imu_sensor/imu\", cb);\n    \n      while (true)\n        gazebo::common::Time::MSleep(10);\n\
    \    \n      // Make sure to shut everything down.\n      gazebo::client::shutdown();\n\
    \    }"
