          <joint name="my_box_joint" type="continuous" >
            <!-- axis is in the parent link frame coordintates -->
            <axis xyz="0 1 0" />
            <parent link="world" />
            <child link="my_box" />
            <!-- initial pose of my_box joint/link in the parent frame coordiantes -->
            <origin xyz="0 0 2" rpy="0 0 0" />
          </joint>
