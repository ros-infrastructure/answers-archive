  body_markdown: "Lorenzo,\n\nIt should be pretty easy.  Basically, you are going\
    \ to put a different transform between the world frame (in this case 'odom_combined')\
    \ and the frame associated with the base of the robot ('base_footprint').  If\
    \ operating with a running system you'll take the current state, apply whatever\
    \ multi-dof and joint states are in the planning_scene_diff, and push the result\
    \ to all components, who will then respond accordingly.  You'll want to interact\
    \ with the components directly instead of using MoveArm, as you probably don't\
    \ want to actually act based on state that's not the current one, though you are\
    \ given the option.  \n\nHere's some code for generating IK solutions given another\
    \ base configuration for your robot: \n\n    #include <ros/ros.h>\n    \n    #include\
    \ <kinematics_msgs/GetKinematicSolverInfo.h>\n    #include <kinematics_msgs/GetConstraintAwarePositionIK.h>\n\
    \    #include <arm_navigation_msgs/SetPlanningSceneDiff.h>\n    \n    static const\
    \ std::string ARM_COLLISION_IK_NAME = \"/pr2_right_arm_kinematics/get_constraint_aware_ik\"\
    ;\n    static const std::string ARM_QUERY_NAME = \"/pr2_right_arm_kinematics/get_ik_solver_info\"\
    ;\n    static const std::string SET_PLANNING_SCENE_DIFF_NAME = \"/environment_server/set_planning_scene_diff\"\
    ;\n    \n    int main(int argc, char **argv){\n      ros::init (argc, argv, \"\
    get_fk\");\n      ros::NodeHandle rh;\n    \n      ros::service::waitForService(ARM_QUERY_NAME);\n\
    \      ros::service::waitForService(ARM_COLLISION_IK_NAME);\n      ros::service::waitForService(SET_PLANNING_SCENE_DIFF_NAME);\n\
    \    \n      ros::ServiceClient set_planning_scene_diff_client = rh.serviceClient<arm_navigation_msgs::SetPlanningSceneDiff>(SET_PLANNING_SCENE_DIFF_NAME);\n\
    \      ros::ServiceClient query_client = rh.serviceClient<kinematics_msgs::GetKinematicSolverInfo>(ARM_QUERY_NAME);\n\
    \      ros::ServiceClient ik_with_collision_client = rh.serviceClient<kinematics_msgs::GetConstraintAwarePositionIK>(ARM_COLLISION_IK_NAME);\n\
    \    \n      arm_navigation_msgs::SetPlanningSceneDiff::Request planning_scene_req;\n\
    \      arm_navigation_msgs::SetPlanningSceneDiff::Response planning_scene_res;\n\
    \    \n      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.stamp\
    \ = ros::Time::now();\n      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.joint_names.push_back(\"\
    world_joint\");\n      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.frame_ids.push_back(\"\
    odom_combined\");\n      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.child_frame_ids.push_back(\"\
    base_footprint\");\n      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.poses.resize(1);\n\
    \      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.poses[0].position.x\
    \ = 4.0;\n      planning_scene_req.planning_scene_diff.robot_state.multi_dof_joint_state.poses[0].orientation.w\
    \ = 1.0;\n    \n      if(!set_planning_scene_diff_client.call(planning_scene_req,\
    \ planning_scene_res)) {\n        ROS_WARN(\"Can't get planning scene\");\n  \
    \      return -1;\n      }\n    \n      // define the service messages\n     \
    \ kinematics_msgs::GetKinematicSolverInfo::Request request;\n      kinematics_msgs::GetKinematicSolverInfo::Response\
    \ response;\n    \n      if(query_client.call(request,response))\n      {\n  \
    \      for(unsigned int i=0; i< response.kinematic_solver_info.joint_names.size();\
    \ i++)\n        {\n          ROS_DEBUG(\"Joint: %d %s\",i,response.kinematic_solver_info.joint_names[i].c_str());\n\
    \        }\n      }\n      else\n      {\n        ROS_ERROR(\"Could not call query\
    \ service\");\n        ros::shutdown();\n        exit(-1);\n      }\n    \n  \
    \    // define the service messages\n      kinematics_msgs::GetConstraintAwarePositionIK::Request\
    \  gpik_req;\n      kinematics_msgs::GetConstraintAwarePositionIK::Response gpik_res;\n\
    \    \n      gpik_req.timeout = ros::Duration(5.0);\n      gpik_req.ik_request.ik_link_name\
    \ = \"r_wrist_roll_link\";\n    \n      gpik_req.ik_request.pose_stamped.header.frame_id\
    \ = \"odom_combined\";\n      gpik_req.ik_request.pose_stamped.pose.position.x\
    \ = 4.75;\n      gpik_req.ik_request.pose_stamped.pose.position.y = -0.188;\n\
    \      gpik_req.ik_request.pose_stamped.pose.position.z = .94;\n    \n      gpik_req.ik_request.pose_stamped.pose.orientation.x\
    \ = 0.0;\n      gpik_req.ik_request.pose_stamped.pose.orientation.y = 0.0;\n \
    \     gpik_req.ik_request.pose_stamped.pose.orientation.z = 0.0;\n      gpik_req.ik_request.pose_stamped.pose.orientation.w\
    \ = 1.0;\n    \n      gpik_req.ik_request.ik_seed_state.joint_state.position.resize(response.kinematic_solver_info.joint_names.size());\n\
    \      gpik_req.ik_request.ik_seed_state.joint_state.name = response.kinematic_solver_info.joint_names;\n\
    \    \n      for(unsigned int i=0; i< response.kinematic_solver_info.joint_names.size();\
    \ i++)\n      {\n        gpik_req.ik_request.ik_seed_state.joint_state.position[i]\
    \ = (response.kinematic_solver_info.limits[i].min_position + response.kinematic_solver_info.limits[i].max_position)/2.0;\n\
    \      }\n      if(ik_with_collision_client.call(gpik_req, gpik_res))\n      {\n\
    \        if(gpik_res.error_code.val == gpik_res.error_code.SUCCESS)\n        {\n\
    \          for(unsigned int i=0; i < gpik_res.solution.joint_state.name.size();\
    \ i ++)\n          {\n            ROS_INFO(\"Joint: %s %f\",gpik_res.solution.joint_state.name[i].c_str(),gpik_res.solution.joint_state.position[i]);\n\
    \          } \n        }\n        else\n        {\n          ROS_ERROR(\"Inverse\
    \ kinematics failed\");\n        }\n      }\n      else\n      {\n        ROS_ERROR(\"\
    Inverse kinematics service call failed\");\n      }\n      ros::shutdown();\n\
    \    }"
