    \ = \"box_frame\";\n      co.id = \"box\";\n      geometry_msgs::Pose pose;\n\

    \ \n    \n      moveit_msgs::CollisionObject move_object;\n      move_object.id\

    \      move_object.operation = move_object.MOVE;\n    \n      ros::Rate loop_rate(100);\n\
