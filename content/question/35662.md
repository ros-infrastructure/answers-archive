    \ PATH=$ROS_ROOT/bin:$PATH\n\nexport PYTHONPATH=$ROS_ROOT/core/roslib/src:$PYTHONPATH\n\
    \nexport ROS_PACKAGE_PATH=/opt/ros/electric/stacks:~/ros_workspace:$ROS_PACKAGE_PATH\
