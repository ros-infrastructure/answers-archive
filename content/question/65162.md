void diagAggCallback(const diagnostic_msgs::DiagnosticArray::ConstPtr &msg)
{
  // flag is changed from 1 to 0 if any of the 14 diagnostic status messages contains errors or warnings
  bool allClear=1;

  for (int i=0; i<14; i++)
  {
    diagnostic_msgs::DiagnosticStatus diagStatus = msg->status[i];
    int level = diagStatus.level;

    if (level==1)
    {
      std::cout << "WARNING!!   name: " + diagStatus.name + " message: " + diagStatus.message << std::endl;
      ab0rt = 1;
      allClear=0;
    }

    if (level==2)
    {
      std::cout << "ERROR!!   name: " + diagStatus.name + " message: " + diagStatus.message << std::endl;
      ab0rt = 1;
      allClear=0;
    }
  }

  if (allClear == 1)
    ab0rt = 0;
}
