    \ TURTLEBOT_3D_SENSOR)\"/>  <!-- r200, kinect, asus_xtion_pro -->\n      <include\
    \ file=\"$(find turtlebot_bringup)/launch/3dsensor.launch\">\n        <arg name=\"\
    rgb_processing\" value=\"false\" />\n        <arg name=\"depth_registration\"\
    \ value=\"false\" />\n        <arg name=\"depth_processing\" value=\"false\" />\n\
