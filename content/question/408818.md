Steps to reproduce:
- make an action server
- make a service server
- call the action server with the callback of the service server
- call the service server

Output:
The action server callback gets called and return if called alone from the terminal 
The call to the action server WITHIN the service server callback behaves as follows:
 - The call to the actions senver gets sent --> the action server callbacks actually get executed
 - GoalOptions of the action client never executes. (unless the service callback finishes, which is not the needed behavoir)
 - the future obtained by sending the goal to the action server never goes into the ready state.

Note: the action client is in a adhoc created callback group.

Ros version: galactic

    namespace test { using actionmsg = srv_action_interfaces::action::TestAction; using clienthandle = rclcpp_action::ClientGoalHandle<actionmsg>;
    
        struct service_server_node : public rclcpp::Node
        {
            service_server_node(rclcpp::NodeOptions const &Options = rclcpp::NodeOptions()) : rclcpp::Node("serviceServernode", Options)
            {
                cb_group_action_client = this->create_callback_group(rclcpp::CallbackGroupType::Reentrant);
                cb_group_service_server = this->create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive);
                this->action_client_ = rclcpp_action::create_client<srv_action_interfaces::action::TestAction>(this, "/capra/test/action_server_name", cb_group_action_client);
                this->SendGoalOptions.feedback_callback = [this](clienthandle::SharedPtr handle, std::shared_ptr<const actionmsg::Feedback> feedback) -> void
                {
                    std::ostringstream oss;
                    std::copy(feedback->partial_sequence.begin(), feedback->partial_sequence.end(), std::ostream_iterator<int>(oss, ", "));
                    RCLCPP_INFO(get_logger(), "feedback : %s", oss.str().c_str());
                };
    
                this->SendGoalOptions.result_callback = [this](clienthandle::WrappedResult const &WrappedResult) -> void
                {
                    std::lock_guard lk(mtx);
                    std::ostringstream oss;
                    std::copy(WrappedResult.result->sequence.begin(), WrappedResult.result->sequence.end(), std::ostream_iterator<int>(oss, ", "));
                    RCLCPP_INFO(get_logger(), "RESULT RECEIVED! : %s", oss.str().c_str());
                    result_received_flag = true;
                    cv.notify_all();
                };
    
                this->SendGoalOptions.goal_response_callback = [this](std::shared_ptr<clienthandle> handle) -> void
                {
                    RCLCPP_INFO(get_logger(), "STATUS: %d", handle->get_status());
                };
    
                this->service_server = this->create_service<srv_action_interfaces::srv::TestService>("/capra/test/service_server_name", std::bind(&capra::test::service_server_node::service_cb, this, std::placeholders::_1, std::placeholders::_2), rmw_qos_profile_services_default, this->cb_group_action_client);
            };
    
            // Connection to action server
            rclcpp_action::Client<actionmsg>::SharedPtr action_client_;
            rclcpp_action::Client<actionmsg>::SendGoalOptions SendGoalOptions{};
            // Service server bring up
            void service_cb(const srv_action_interfaces::srv::TestService_Request::SharedPtr req, srv_action_interfaces::srv::TestService_Response::SharedPtr res)

                RCLCPP_INFO(rclcpp::get_logger("SERVICE SERVER CALLBACK"), "sending request to action server");

                goal.order = req->order;
