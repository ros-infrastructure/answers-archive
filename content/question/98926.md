    \   double secs1;\n    public:\n    ShapeAction(std::string name,double gettime)\
    \ : \n    as_(nh_, name),\n    action_name_(name)\n    {\n    secs1=gettime;\n\
    \n    //register the goal and feeback callbacks\n    as_.registerGoalCallback(boost::bind(&ShapeAction::goalCB,\
    \ this));\n    as_.registerPreemptCallback(boost::bind(&ShapeAction::preemptCB,\
    \ this));\n\n    //subscribe to the data topic of interest\n    sub_ = nh_.subscribe(\"\

    \ }\n    void goalCB()\n    {\n    turtle_actionlib::ShapeGoal goal = *as_.acceptNewGoal();\n\
    \      }\n\n    void preemptCB()\n      {\n    ROS_INFO(\"%s: Preempted\", action_name_.c_str());\n\
    \    // set the action state to preempted\n    as_.setPreempted();\n      }\n\
    \    void controlCB(const turtlesim::Pose::ConstPtr& msg)\n      {\n    // make\
    \ sure that the action hasn't been canceled\n    if (!as_.isActive())\n      return;\n\
    \n    // scalar values for drive the turtle faster and straighter\n    double\
    \ l_scale = 0.9;\n    double a_scale = 0.9;\n    \n    double secs2 =ros::Time::now().toSec();\n\
    \n    double secs3=secs2-secs1;\n    ROS_INFO(\"Time=%f\", secs3); \n  \n    if\
    \ (secs3 < 17)   \n    {   \n    //desired trajectory \n\tstart_x_ = 5+secs3/4;\n\
    \        start_y_ = 5+secs3/4;\n        start_theta_ = M_PI/4;\n\n      dis_error_\
    \ =  fabs(sqrt((start_x_- msg->x)*(start_x_-msg->x) + (start_y_-msg->y)*(start_y_-msg->y)));\n\
