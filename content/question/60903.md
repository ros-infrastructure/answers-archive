  body_markdown: "The TimeSynchronizer will only work when the timestamps are exactly\
    \ equal. You need to use the ApproximateTimeSyncronizer:\nExample (copied from\
    \ rgbdslam):\n\n    typedef message_filters::sync_policies::ApproximateTime<sensor_msgs::Image,\
    \ \n                                                            sensor_msgs::Image,\n\
    \                                                            sensor_msgs::CameraInfo>\
    \ NoCloudSyncPolicy;\n\n    class MyClass {\n    message_filters::Subscriber<sensor_msgs::Image>\
    \ *visua_sub_;      \n    message_filters::Subscriber<sensor_msgs::Image> *depth_sub_;\
    \      \n    message_filters::Subscriber<sensor_msgs::CameraInfo> *cinfo_sub_;\
    \      \n    message_filters::Synchronizer<NoCloudSyncPolicy>* no_cloud_sync_;\n\
    \n    // method definitions needed here\n    }\n\n    void MyClass::init()\n \
    \   {\n    std::string visua_tpc(\"/topic_image_mono\");\n    std::string depth_tpc(\"\
    /topic_image_depth\");\n    std::string cinfo_tpc(\"/camera_info_topic\");\n \
    \   int q = 5; //queue size\n    ros::NodeHandle nh;\n\n    visua_sub_ = new image_sub_type(nh,\
    \ visua_tpc, q);\n    depth_sub_ = new image_sub_type(nh, depth_tpc, q);\n   \
    \ cinfo_sub_ = new cinfo_sub_type(nh, cinfo_tpc, q);\n\n    no_cloud_sync_ = new\
    \ message_filters::Synchronizer<NoCloudSyncPolicy>(NoCloudSyncPolicy(q),  *visua_sub_,\
    \ *depth_sub_, *cinfo_sub_);\n   \n    no_cloud_sync_->registerCallback(boost::bind(&MyClass::callbackMethod,\
    \ this, _1, _2, _3));\n    }\n\n    //The callback method\n    void MyClass::callbackMethod\
    \ (const sensor_msgs::ImageConstPtr& visual_img_msg,\n                       \
    \           const sensor_msgs::ImageConstPtr& depth_img_msg,\n               \
    \                   const sensor_msgs::CameraInfoConstPtr& cam_info_msg) \n  \
    \  {\n         //Your processing code\n    }\n"
