  body_markdown: "Got this working for python on UR5 in Gazebo\n   \n\n    import\
    \ yaml\n    import os\n        \n    plan = group.plan()\n            \n    file_path\
    \ = os.path.join(os.path.expanduser('~'), 'saved_trajectories', 'plan.yaml')\n\
    \    with open(file_path, 'w') as file_save:\n        yaml.dump(plan, file_save,\
    \ default_flow_style=True)\n            \n    with open(file_path, 'r') as file_open:\n\
    \        loaded_plan = yaml.load(file_open)\n            \n    group.execute(loaded_plan)"
