    \n    roslaunch myexamplefile.launch\n\nHere is an example file which works with\

    \ 0 0 /odom /base_link 100\"/>\n     \t<node pkg=\"tf\" type=\"static_transform_publisher\"\
    \ name=\"base_lnk_to_laser_lnk\" args=\"0 0 0 0 0 0 /base_link /camera_link 100\"\
    />\n     \n     \t<param name=\"pub_map_odom_transform\" value=\"true\"/>\n  \
    \   \t<param name=\"map_frame\" value=\"map\" />\n     \t<param name=\"base_frame\"\
