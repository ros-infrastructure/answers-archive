    {
        float suma=0;
        float total=0;
        float cdg=0;
        for (int i=0; i<nc; i++)
        {
            suma += i*(255-ptrFila[i]);
            total += 255-ptrFila[i];
            circle(colorFrame, Point(i, fila), 1, Scalar(255, 0, 0));
        }
        if(total>0)
            cdg=suma/total;
        else
            cdg=-1;
    
        return cdg;
    }
