    \ the original code:\n     \n    scan_msg.angle_min = M_PI - angle_min;\n    scan_msg.angle_max\
    \ = M_PI - angle_max;\n     \n    to:\n     \n    scan_msg.angle_min = -1*(M_PI\
    \ - angle_min);\n    scan_msg.angle_max = M_PI - angle_min; \n\nhttps://gist.github.com/Po-Jen\n"
