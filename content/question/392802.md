  body_markdown: "Figured it out\n\n    #!/usr/bin/env python3\n    \n    import rospy\n\
    \    from sensor_msgs.msg import Image\n    from cv_bridge import CvBridge\n \
    \   import cv2\n      \n    def publish_message():\n     \n      # Node is publishing\
    \ to the video_frames topic using \n      # the message type Image\n      pub\
    \ = rospy.Publisher('video_frames', Image, queue_size=10)\n         \n      #\
    \ Tells rospy the name of the node.\n      # Anonymous = True makes sure the node\
    \ has a unique name. Random\n      # numbers are added to the end of the name.\n\
    \      rospy.init_node('video_pub_py', anonymous=True)\n         \n      # Go\
    \ through the loop 10 times per second\n      rate = rospy.Rate(10) # 10hz\n \
    \        \n      # Create a VideoCapture object\n      # The argument '0' gets\
    \ the default webcam.\n      #cap = cv2.VideoCapture(0)\n      cap = cv2.VideoCapture('../video/ros.mp4')\n\
    \         \n      # Used to convert between ROS and OpenCV images\n      br =\
    \ CvBridge()\n     \n      # While ROS is still running.\n      while not rospy.is_shutdown():\n\
    \         \n          # Capture frame-by-frame\n          # This method returns\
    \ True/False as well\n          # as the video frame.\n          ret, frame =\
    \ cap.read()\n             \n          if ret == True:\n            # Print debugging\
    \ information to the terminal\n            rospy.loginfo('publishing video frame')\n\
    \                 \n            # Publish the image.\n            # The 'cv2_to_imgmsg'\
    \ method converts an OpenCV\n            # image to a ROS image message\n    \
    \        pub.publish(br.cv2_to_imgmsg(frame))\n                 \n          #\
    \ Sleep just enough to maintain the desired rate\n          rate.sleep()\n   \
    \          \n    if __name__ == '__main__':\n      try:\n        publish_message()\n\
    \      except rospy.ROSInterruptException:\n        pass"
