  body_markdown: "In order to use qexserialport in ros I suggest you this cmake.This\
    \ program writes \"h\" on the port an receives some data from micro.\nCmake\n\n\
    \    cmake_minimum_required(VERSION 2.4.6)\n    include($ENV{ROS_ROOT}/core/rosbuild/rosbuild.cmake)\n\
    \    \n    \n    find_package(Qt4 COMPONENTS QtCore QtGui)\n    INCLUDE(${QT_USE_FILE})\n\
    \    ADD_DEFINITIONS(${QT_DEFINITIONS})\n    link_directories(/usr/include/qt4/QtExtSerialPort)\n\
    \    \n    rosbuild_init()\n    \n    set(EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/bin)\n\
    \    set(LIBRARY_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/lib)\n    \n    rosbuild_add_executable(first_test\
    \ src/first_test.cpp)\n    target_link_libraries(first_test ${QT_LIBRARIES} -lqextserialport\
    \ -lpthread)\n\nand the program\n\n    /*\n     * first_test.cpp\n     *\n   \
    \  *  Created on: Dec 19, 2013\n     *      Author: hamid\n     */\n    \n   \
    \ #include \"ros/ros.h\"\n    #include <QtExtSerialPort/qextserialport.h>\n  \
    \  #include \"QDebug\"\n    #include \"QCoreApplication\"\n    \n    \n    int\
    \ main(int argc, char** argv)\n    {\n        ros::init(argc, argv, \"first_test\"\
    );\n        QCoreApplication app(argc, argv);\n    \n        ros::NodeHandle n;\n\
    \        QextSerialPort *port;\n        QByteArray bytes;\n        QByteArray\
    \ bytes2;\n    \n    \n        QString portName = QLatin1String(\"ttyUSB0\");\n\
    \        port = new QextSerialPort(QString(portName), QextSerialPort::EventDriven);\n\
    \    \n        port->setBaudRate(BAUD9600);\n        port->setFlowControl(FLOW_OFF);\n\
    \        port->setParity(PAR_NONE);\n        port->setDataBits(DATA_8);\n    \
    \    port->setStopBits(STOP_1);\n    \n    \n        if (port->open(QIODevice::ReadWrite)\
    \ == true)\n    \t {\n            \tqDebug() << \"listening for data on\" << port->portName();\n\
    \             }\n        else \n    \t {\n                   qDebug() << \"device\
    \ failed to open:\" << port->errorString();\n             }\n    \n        bytes[0]='h';\n\
    \        int total = port->write(bytes,bytes.size());\n        qDebug() << total;\n\
    \        sleep(1);\n        int a = port->bytesAvailable();\n        qDebug()\
    \ << a;\n        bytes2.resize(a);\n        port->read(bytes2.data(),bytes2.size());\n\
    \        qDebug() <<(QString::fromAscii(bytes2).toUcs4());\n    \n        return\
    \ 0;\n    }"
