  body_markdown: "Here is snippet for publish for \"*Fly forward*\", create a node\
    \ for publishing messages.\n\n    ros::Rate loop_rate(10); // publishing frequency\n\
    \    while (ros::ok())\n    {\n    \n        //define publisher with topicname\
    \ \"/cmd_vel\"\n        ros::Publisher cmd_vel_pub;\n        cmd_vel_pub = nh_.advertise<geometry_msgs::Twist>(\"\
    /cmd_vel\", 1); \n        //declare message of geometry_msgs::Twist type\n   \
    \     geometry_msgs::Twist cmd_vel;    \n        //populate message\n        cmd_vel.linear.x\
    \ = 1.0;\n        cmd_vel.linear.y = 0.0;\n        cmd_vel.linear.z = 0.0;\n \
    \       cmd_vel.angular.x = 0.0;\n        cmd_vel.angular.y = 0.0;\n        cmd_vel.angular.z\
    \ = 0.0;    \n        //publish message\n        cmd_vel_pub.publish(cmd_vel);\n\
    \    \n        ros::spinOnce(); // not required if you dont have callbacks\n \
    \   \n        loop_rate.sleep();\n    }"
