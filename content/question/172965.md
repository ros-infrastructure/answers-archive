    \ code that  send randomly-generated velocity commands to a turtlesim turtle\n\
    \n    #include <ros/ros.h>\n    #include <turtlesim/Velocity.h>  // For turtlesim::Velocity\n\
    \    #include <stdlib.h> // For rand() and RAND_MAX\n    \n    int main(int argc,\
    \ char **argv) {\n      // Initialize the ROS system and become a node.\n    \
    \  ros::init(argc, argv, \"publish_velocity\");\n      ros::NodeHandle nh;\n \
    \   \n      // Create a publisher object.\n      ros::Publisher pub = nh.advertise<turtlesim::Velocity>(\n\
    \        \"turtle1/command_velocity\", 1000);\n    \n      // Seed the random\
    \ number generator.\n      srand(time(0));\n    \n      // Loop at 2Hz until the\
    \ node is shut down.\n      ros::Rate rate(2);\n      while(ros::ok()) {\n   \
    \     // Create and fill in the message.\n        turtlesim::Velocity msg;\n \
    \       msg.linear = float(rand())/float(RAND_MAX);\n        msg.angular = 2*float(rand())/float(RAND_MAX)\
    \ - 1;\n    \n        // Publish the message.\n        pub.publish(msg);\n   \
    \ \n        // Send a message to rosout with the details.\n        ROS_INFO_STREAM(\"\
    Sending random velocity command:\"\n          << \" linear=\" << msg.linear\n\
    \          << \" angular=\" << msg.angular);\n    \n        // Wait until it's\
