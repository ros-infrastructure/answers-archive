  body_markdown: "I found a solution using the `OpaqueFunction`. \n\nArguments are\
    \ now evaluated and set in the create_robot_description which returns a LaunchConfiguration\
    \ which than can be used as argument in the LaunchDescription.\n\n    import os\n\
    \    from ament_index_python.packages import get_package_share_directory\n   \
    \ from launch import LaunchDescription\n    from launch.actions import DeclareLaunchArgument\n\
    \    from launch.substitutions import LaunchConfiguration, TextSubstitution\n\
    \    from launch_ros.actions import Node\n    from launch.actions import OpaqueFunction\n\
    \    from launch.actions import SetLaunchConfiguration\n    \n    import xacro\n\
    \    \n    def generate_launch_description():\n    \n        use_sim_time    \
    \ = LaunchConfiguration('use_sim_time', default='true')\n        namespace_arg\
    \    = DeclareLaunchArgument('namespace', default_value=TextSubstitution(text='r0'))\n\
    \        robot_arg        = DeclareLaunchArgument('robot',     default_value=TextSubstitution(text='pioneer3dx'))\n\
    \        X_launch_arg     = DeclareLaunchArgument('X',         default_value=TextSubstitution(text='0.0'))\n\
    \        Y_launch_arg     = DeclareLaunchArgument('Y',         default_value=TextSubstitution(text='0.0'))\n\
    \        Theta_launch_arg = DeclareLaunchArgument('Theta',     default_value=TextSubstitution(text='0.0'))\n\
    \    \n        models_dir = get_package_share_directory('tuw_gazebo_models') +\
    \ '/models'\n    \n    \n    \n        def create_robot_description(context):\
    \ \n            xacro_file = os.path.join(get_package_share_directory('tuw_gazebo_models'),\
    \ 'models', context.launch_configurations['robot'], 'main.xacro')    \n      \
    \      assert os.path.exists(xacro_file), \"The main.xacro doesnt exist in \"\
    +str(xacro_file)\n            robot_description_config = xacro.process_file(xacro_file,\
    \ \n                mappings={  \"namespace\": context.launch_configurations['namespace'],\
    \ \n                            \"models_dir\": models_dir})\n            robot_desc\
    \ = robot_description_config.toxml()\n            return [SetLaunchConfiguration('robot_desc',\
    \ robot_desc)]\n    \n        create_robot_description_arg = OpaqueFunction(function=create_robot_description)\n\
    \    \n        return LaunchDescription([\n            namespace_arg,\n      \
    \      robot_arg,\n            X_launch_arg,\n            Y_launch_arg,\n    \
    \        Theta_launch_arg,\n            create_robot_description_arg,\n      \
    \      DeclareLaunchArgument(\n                'use_sim_time',\n             \
    \   default_value='false',\n                description='Use simulation (Gazebo)\
    \ clock if true'),\n    \n            Node(package='tuw_gazebo_models', \n   \
    \             name=\"publisher_robot\", \n                executable='spawn_robot.py',\
    \ \n                arguments=[LaunchConfiguration('robot_desc')],\n         \
    \       parameters=[{\n                    \"X\": LaunchConfiguration('X'),\n\
    \                    \"Y\": LaunchConfiguration('Y'),\n                    \"\
    Theta\": LaunchConfiguration('Theta'),\n                    \"namespace\": LaunchConfiguration('namespace')}],\n\
    \                output='screen'),\n            Node(\n                package=\"\
    robot_state_publisher\",\n                executable=\"robot_state_publisher\"\
    ,\n                name=\"robot_state_publisher\",\n                namespace=[LaunchConfiguration('namespace')],\n\
    \                parameters=[{\n                    \"robot_description\": LaunchConfiguration('robot_desc')}],\n\
    \                output=\"screen\"),\n        ])"
