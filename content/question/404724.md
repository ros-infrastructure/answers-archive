Starting GazeboRosSkidSteerDrive plugin (ns = /agent1/) 
This can't continue. It will stop at this step
However, it can continue without ns

    <group>
      <arg name="robot_namespace" default="/agent1"/>
      <arg name="urdf_extras" default="$(find scout_description)/urdf/empty.urdf"/>
      <param name="robot_description" command="$(find xacro)/xacro '$(find scout_description)/urdf/mini copy.xacro'
          robot_namespace:=$(arg robot_namespace)
          urdf_extras:=$(arg urdf_extras)" 
          />

      <node name="spawn_scout_mini_model" pkg="gazebo_ros" type="spawn_model" args="-x $(arg x)
                -y $(arg y)
                -z $(arg z)
                -Y $(arg yaw)
                -unpause
                -urdf
                -param robot_description
                -model 'scout_mini$(arg robot_namespace)'" />
    
        <!-- Load joint controller configurations from YAML file to parameter server -->

      <rosparam file="$(find scout_gazebo_sim)/config/scout_v2_control.yaml" command="load" ns="/agent1"/>
      <!-- load the controllers -->
      <node name="scout_mini_controller_spawner" pkg="controller_manager" type="spawner" respawn="false" output="screen" ns="/agent1" args="--scout_state_controller scout_motor_fr_controller scout_motor_fl_controller scout_motor_rl_controller scout_motor_rr_controller"/>
      <node name="scout_mini_robot_state_publisher" pkg="robot_state_publisher" type="robot_state_publisher" ns="/agent1"/>
     <node name="scout_skid_steer_controller" pkg="scout_gazebo_sim" type="scout_skid_steer_controller">
          <param name="robot_namespace" type="string" value="$(arg robot_namespace)" />
      </node>
    </group>
