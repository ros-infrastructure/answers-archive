		v = StringVar()
		v.set(distance)
		self.clock = Label(frame, font=('times', 20, 'bold'), bg='green', textvariable = v)
		self.clock.pack(fill=BOTH, expand=1)
		rate = rospy.Rate(2)
		
		while not rospy.is_shutdown():
