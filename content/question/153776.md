- body_markdown: "launch file : \n\n<!-- Launch file for visualizing OSM data\n\n\
    \     arg: url = URL for map data\n\n     $Id$\n  -->\n\n<launch>\n\n  <arg name=\"\
    url\" default=\"package://osm_cartography/tests/maprobot.osm\" />\n\n  <!-- Transform\
    \ /map into /local_map frame. Rviz cannot handle large\n       UTM coordinates,\
    \ they require double precision, so we visualize\n       using the /local_map.\
    \  Start the transform early to minimize TF\n       errors in rviz.\n  -->\n \
    \ <node pkg=\"tf\" type=\"static_transform_publisher\" name=\"local_map_tf\"\n\
    \        args=\"598022.3 5633435.3 0 0 0 0 1 map local_map   100\" />\n\n <!--\
    \ map  odom    -->\n  <!-- start map server -->\n  <node pkg=\"osm_cartography\"\
    \ type=\"osm_server\" name=\"osm_server\" />\n\n  <!-- start map visualization\
    \ -->\n  <node pkg=\"osm_cartography\" type=\"viz_osm\" name=\"viz_osm\" >\n \
    \   <param name=\"map_url\" value=\"$(arg url)\" />\n   \n  </node>\n\n</launch>"
