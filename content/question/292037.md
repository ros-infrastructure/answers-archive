    \ using boost::bind (see https://answers.ros.org/question/233525/template-argument-deduction-fails/),\
    \ so you need to be more explicit:\n\n\n    #include <ros/ros.h>\n    #include\
    \ <std_srvs/SetBool.h>\n    \n    bool test(std_srvs::SetBool::Request& req, std_srvs::SetBool::Response&\
    \ resp)\n    {\n      return true;\n    }\n\n    bool test2(std_srvs::SetBool::Request&\
    \ req, std_srvs::SetBool::Response& resp,\n        bool toggle)\n    {\n     \
    \ return true;\n    }\n    \n    int main(int argc, char **argv)\n    {\n    \
    \  ros::init(argc, argv, \"node\");\n      ros::NodeHandle nh;\n      // works\
    \ without template parameters\n      ros::ServiceServer service0 = nh.advertiseService(\"\
    my_service0\", test);\n      // compile error: no matching function for call to\
    \ \u2018ros::NodeHandle::advertiseService...\n\n      // ros::ServiceServer service1\
    \ = nh.advertiseService(\"my_service1\",\n      //    boost::bind(&test2, _1,\
    \ _2, false));\n      ros::ServiceServer service2 = nh.advertiseService<std_srvs::SetBool::Request,\n\
    \          std_srvs::SetBool::Response>(\"my_service2\",\n          boost::bind(&test2,\
    \ _1, _2, true));\n      ros::spin();\n      return 0;\n    }"
