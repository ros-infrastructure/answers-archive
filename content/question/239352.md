    "my_package_msgs/MessageOrService.h"
it always takes a few "catkin_makes" for my system to recognize the library. Here is my CMakeList file:

    cmake_minimum_required(VERSION 2.8.3)
    project(message_package_msgs)
    find_package(catkin REQUIRED COMPONENTS
      roscpp
      rospy
      std_msgs
      message_generation
      geometry_msgs
      sensor_msgs
    )
    add_message_files(
      DIRECTORY
        msg
      FILES
        Message_EXAMPLE.msg
    )
    add_service_files(
      DIRECTORY
        srv
      FILES
        Service_EXAMPLE.srv
    )
    generate_messages(
      DEPENDENCIES
      std_msgs
      geometry_msgs
      sensor_msgs
    )
    catkin_package(
      CATKIN_DEPENDS roscpp rospy std_msgs message_runtime geometry_msgs sensor_msgs
    include_directories(
      ${catkin_INCLUDE_DIRS}
    )

      <buildtool_depend>catkin</buildtool_depend>
