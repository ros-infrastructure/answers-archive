  body_markdown: "The current *stageros* implementation supports only one ranger/laser\
    \ sensor per robot.\n----------------------------------------\n\n**You have to\
    \ use a stage model without sonar to use the laser sensor.**\n\nAttached you will\
    \ find my cave-pionee.world which includes a working pioneer stage simulation.\
    \ The background file cave.png must be in the same folder to get it working\n\n\
    \    # cave-pionee.world \n    # simple cave environment with a pioneer robot\
    \ based on the basic world file examples of \n    # Richard Vaughan Richard Vaughan,\
    \ Andrew Howard, Luis Riazuelo\n    # Authors: Markus Bader\n    \n    \n    define\
    \ hokuyolaser ranger\n    (\n      sensor(       \n        # laser-specific properties\n\
    \        # factory settings for LMS200  \n        range [ 0.0  5.0 ]\n       \
    \ fov 270.0\n        samples 270\n      )\n      model\n      (\n        # generic\
    \ model properties\n        size [ 0.07 0.07 0.05 ] # dimensions from LMS200 data\
    \ sheet  \n        color \"blue\"\n      )\n    )\n    \n    define my_block model\n\
    \    (\n      size [0.5 0.5 0.5]\n      gui_nose 0\n    )\n    \n    \n    define\
    \ floorplan model\n    (\n      # sombre, sensible, artistic\n      color \"gray30\"\
    \n    \n      # most maps will need a bounding box\n      boundary 1\n    \n \
    \     gui_nose 0\n      gui_grid 0\n    \n      gui_outline 0\n      gripper_return\
    \ 0\n      fiducial_return 0\n      laser_return 1\n    )\n    \n    # set the\
    \ resolution of the underlying raytrace model in meters\n    resolution 0.02\n\
    \    \n    interval_sim 100  # simulation timestep in milliseconds\n    \n   \
    \ \n    # configure the GUI window\n    window\n    (\n      size [ 635.000 666.000\
    \ ] # in pixels\n      scale 36.995   # pixels per meter\n      center [ -0.040\
    \  -0.274 ]\n      rotate [ 0  0 ]\n            \n      show_data 1          \
    \    # 1=on 0=off\n    )\n    \n    # load an environment bitmap\n    floorplan\n\
    \    ( \n      name \"cave\"\n      size [16.000 16.000 0.800]\n      pose [0\
    \ 0 0 0]\n      bitmap \"cave.png\"\n    )\n    # set the resolution of\n    \n\
    \    \n    # throw in a robot\n    my_block( pose [ 5 4 0 180.000 ] color \"green\"\
    )\n    \n    \n    define pioneer_base position \n    (\n      color \"red\" \
    \     # Default color.\n      drive \"diff\"       # Differential steering model.\n\
    \      gui_nose 1                    # Draw a nose on the robot so we can see\
    \ which way it points\n      obstacle_return 1             # Can hit things.\n\
    \      ranger_return 0.5              # reflects sonar beams\n      blob_return\
    \ 1                 # Seen by blobfinders  \n      fiducial_return 1         \
    \    # Seen as \"1\" fiducial finders\n    \n      localization \"gps\"      \
    \       \n      localization_origin [0 0 0 0]   # Start odometry at (0, 0, 0).\n\
    \    \n      # alternative odometric localization with simple error model\n  \
    \    # localization \"odom\"             # Change to \"gps\" to have impossibly\
    \ perfect, global odometry\n      #  odom_error [ 0.05 0.05 0.1 ]     # Odometry\
    \ error or slip in X, Y and Theta\n                                         #\
    \ (Uniform random distribution)   \n    \n      # four DOF kinematics limits\n\
    \      # [ xmin xmax ymin ymax zmin zmax amin amax ]        \n      velocity_bounds\
    \ [-0.5 0.5 0 0 0 0 -90.0 90.0 ]          \n      acceleration_bounds [-0.5 0.5\
    \ 0 0 0 0 -90 90.0 ]\n    )\n    \n    \n    \n    define pioneer2dx_base_no_sonar\
    \ pioneer_base\n    (\n      # actual size\n      size [0.44 0.38 0.22] # sizes\
    \ from MobileRobots' web site\n    \n      # the pioneer's center of rotation\
    \ is offset from its center of area\n      origin [-0.04 0 0 0]\n    \n      #\
    \ draw a nose on the robot so we can see which way it points\n      gui_nose 1\n\
    \    \n      # estimated mass in KG\n      mass 23.0 \n      # differential steering\
    \ model\n      drive \"diff\"\n     )\n    \n    # as above, but with front sonar\
    \ only\n    define pioneer2dx_no_sonar pioneer2dx_base_no_sonar\n    (  \n   \
    \   # simplified Body shape:\n      block( \n        points 8\n        point[0]\
    \ [-0.2 0.12]\n        point[1] [-0.2 -0.12]\n        point[2] [-0.12 -0.2555]\n\
    \        point[3] [0.12 -0.2555]\n        point[4] [0.2 -0.12]\n        point[5]\
    \ [0.2 0.12]\n        point[6] [0.12 0.2555]\n        point[7] [-0.12 0.2555]\n\
    \        z [0 0.22]\n      )\n    )\n    \n    \n    pioneer2dx_no_sonar \n  \
    \  (\n      # can refer to the robot by this name\n      name \"r0\"\n      pose\
    \ [ -7 -7 0 45 ] \n      hokuyolaser(pose [ 0.225 0.000 -0.15 0.000 ])\n    )"
