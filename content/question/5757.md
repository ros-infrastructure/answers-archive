The name of a existing model is "ball".My source is:

transport::PublisherPtr modelPub = this->node->Advertise<msgs::Model>("~/model/modify");

msgs::Model msg;

msg.set_name("ball");

msg.set_ is_ static(true);;

modelPub->Publish(msg);
