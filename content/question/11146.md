    \ target that will do the copy when the 'all' target fires\n    add_custom_target(deploy\
    \ ALL\n                      COMMAND cp $(CMAKE_SOURCE_DIR)/undeployed /tmp/deployed)\n\
    \    # Declare a dependency on the 'rosbuild_precompile' target, to ensure \n\
    \    # that message generation happens before we copy.\n    add_dependencies(deploy\
    \ rosbuild_precompile)"
