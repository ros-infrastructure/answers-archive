    class SituationalMotionPlanning(Node):

        def __init__(self):
            super().__init__('SituationalMotionPlanning')
        
            self._tf_buffer = Buffer()
            self._tf_listener = TransformListener(self._tf_buffer, self)
            self._output_timer = self.create_timer(0.1, self.call_tf)
    
        def call_tf(self):
            try:
                now = rclpy.time.Time()

                trans =  self._tf_buffer.lookup_transform('map', 'base_link', now, timeout=Duration(seconds=1.0))
            except LookupException:
                self.get_logger().info('transform not ready')
                return
            state_x = trans.transform.translation.x
            state_y = trans.transform.translation.y

            rot_x = trans.transform.rotation.x
            rot_y = trans.transform.rotation.y
            rot_z = trans.transform.rotation.z
            rot_w = trans.transform.rotation.w
            t3 = 2*(rot_w * rot_z + rot_x * rot_y)
            t4 = 1 -2*(rot_y * rot_y + rot_z * rot_z)
            state_yaw = math.atan2(t3, t4)

            state_x = round(state_x,2)
            state_y = round(state_y,2)
            state_yaw = round(state_yaw,2)

            self.get_logger().info('x = ' + str(state_x) + ' y =  '+ str(state_y), ' yaw = '+ str(state_yaw))


    def main(args=None):
        rclpy.init(args=args)

        controller = SituationalMotionPlanning()

        rclpy.spin(controller)

        controller.destroy_node()
        rclpy.shutdown()

    if __name__ == '__main__':
        main()
