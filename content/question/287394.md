  body_markdown: "There is a node under moveit_ros_planning_type pkg, where you can\
    \ pass argument and publish scene.\nJust add this to your launch file:\n\n   \
    \ <arg name=\"scene_file\" default=\"$(find project_name)/config/file.scene\"\
    /> \n    <node name = \"moveit_publish_scene_from_text\" pkg= \"moveit_ros_planning\"\
    \ type = \"moveit_publish_scene_from_text\" args= \"$(arg scene_file)\"/>"
