  body_markdown: "I have put a node below that publishes static GPS data in a sensor_msgs/NavSatFix\
    \ message to the topic 'gps/fix' since the navsat_transform_node of robot_localization\
    \ subscribes to gps/fix. If you want the GPS data to update you can change the\
    \ values on the attributes you need e.g. latitude/longitude over time in the pattern\
    \ you want.  \n\n    import rclpy\n    import os\n    from rclpy.node import Node\n\
    \    \n    from sensor_msgs.msg import NavSatFix\n    from sensor_msgs.msg import\
    \ NavSatStatus\n    from std_msgs.msg import Header\n    \n    \n    class GpsNode(Node):\n\
    \    \n        def __init__(self):\n            super().__init__('gps_node')\n\
    \            self.publisher_ = self.create_publisher(NavSatFix, 'gps/fix', 10)\n\
    \            timer_period = 0.5  # seconds\n            self.timer = self.create_timer(timer_period,\
    \ self.timer_callback)\n            \n        def timer_callback(self):\n    \
    \            msg = NavSatFix()\n                msg.header = Header()\n      \
    \          msg.header.stamp = self.get_clock().now().to_msg()\n              \
    \  msg.header.frame_id = \"gps\"\n    \n                msg.status.status = NavSatStatus.STATUS_FIX\n\
    \                msg.status.service = NavSatStatus.SERVICE_GPS\n    \n       \
    \         # Position in degrees.\n                msg.latitude = 57.047218\n \
    \               msg.longitude = 9.920100\n                \n                #\
    \ Altitude in metres.\n                msg.altitude = 1.15\n                \n\
    \                msg.position_covariance[0] = 0\n                msg.position_covariance[4]\
    \ = 0\n                msg.position_covariance[8] = 0\n                msg.position_covariance_type\
    \ = NavSatFix.COVARIANCE_TYPE_DIAGONAL_KNOWN\n                \n             \
    \   self.publisher_.publish(msg)\n                self.best_pos_a = None\n   \
    \ \n    \n    def main(args=None):\n        rclpy.init(args=args)\n    \n    \
    \    gps_node = GpsNode()\n    \n        rclpy.spin(gps_node)\n    \n        #\
    \ Destroy the node explicitly\n        # (optional - otherwise it will be done\
    \ automatically\n        # when the garbage collector destroys the node object)\n\
    \        gps_node.destroy_node()\n        rclpy.shutdown()\n    \n    \n    if\
    \ __name__ == '__main__':\n        main()\n"
