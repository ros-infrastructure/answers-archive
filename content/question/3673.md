    \    \n            ////////////////////////////////////////////////////////////////\n\
    \            /////// SDF PARAMETERS\n    \n            ////////////// Get nr of\
    \ spheres\n            if (!_sdf->HasElement(\"nr_spheres\"))\n            {\n\
    \              std::cout << \"Missing parameter <nr_spheres> in FactoryLiquid,\
    \ default to 0\" << std::endl;\n              nr_spheres = 0;\n            }\n\
    \            else nr_spheres = _sdf->GetElement(\"nr_spheres\")->GetValueUInt();\n\
    \    \n            ////////////// Set up the initial position parameter\n    \
    \        if (!_sdf->HasElement(\"init_pos\"))\n            {\n              std::cout\
    \ << \"Missing parameter <init_pos> in FactoryLiquid, default to 0 0 0\" << std::endl;\n\
    \              init_pos.x = 0.0;\n              init_pos.y = 0.0;\n          \
    \    init_pos.z = 0.0;\n            }\n            else init_pos = _sdf->GetElement(\"\
    init_pos\")->GetValueVector3();\n             \n            // etc. other parameters\
    \ from the world file\n    \n            /////// END SDF PARAMETERS\n        \
    \    //////////////////////////////////////////////////////////////////\n    \n\
    \    \n            //////////////////////////////////////////////////////////////////\n\
    \            //////////////////////////////////////////////////////////////////\n\
    \            //////////////////////////// START XML LIQUID\n            xml <<\
    \ \"<?xml version='1.0'?>\\n\";\n            xml << \"<sdf version='1.4'>\\n\"\
    ;\n            xml << \"<model name='liquid_spheres'>\\n\";\n            xml <<\
    \ \"\\t<static>false</static>\\n\";\n            xml << \"\\t<pose>\" << init_pos.x\
    \ << \" \" << init_pos.y << \" \" << init_pos.z << \" 0 0 0 </pose>\\n\";\n  \
    \  \n            for (unsigned int i=0; i<nr_spheres; i++)\n            {\n  \
    \                  p3 = FactoryLiquid::part_position(i, radius, spawned, level);\n\

    \ << \" \" << p3.z << \" 0 0 0</pose>\\n\";\n    \n                    xml <<\

    \ 0 0 0 0 0 0 </pose>\\n\";\n                    xml << \"\\t\\t\\t\\t<inertia>\\\

    \ xml << \"\\t\\t\\t\\t\\t<iyy>\" << inertia << \"</iyy>\\n\";\n             \
