    def follow_kerb(self) :
        move = Twist()
    
        avr = self.getAverage( self.clean_set(self.regions['1'])  + self.clean_set(self.regions['2']) )
        control = 0.5 * (1.5 - avr)
    
        move.linear.x = 0.2
        if control > 0 :
            move.angular.z += 0.2
            rospy.logerr("Turning [+]: AVG==> %s, CNTRL: %s", avr, control)
        if control < 0 :
            move.angular.z -= 0.2
            rospy.logwarn("Turning [-]: AVG==> %s, CNTRL: %s", avr, control)
        return move
