              delay(10);

              delay(10);

          delay(1000)

              servos[i].write(180-pos);
              delay(10);
            }
            else {
              servos[i].write(pos);
              delay(10);
            }   } }
        
        ros::Subscriber<std_msgs::UInt16> sub("snacbotservo", open_close);
