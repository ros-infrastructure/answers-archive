    \ master using the `ros::master::getNodes(v)` This will push the names of the\
    \ nodes registered into the vector of strings,  v.\n\n   \n\n         while (ros::ok())\
    \ \n       {\n            if(!ros::master::getNodes(v))\n            {\n     \
    \         ROS_INFO(\"Error\");\n        \n            }   \n             else\n\
    \            {  \n               //do your stuff \n            }\n        }\n\n\
    You will have to include  the libraries \n\n    #include \"ros/this_node.h\" \n\
    \     #include \"ros/master.h\""
