```
    <joint name='base_link2link_link' type='revolute2'>
      <pose relative_to='base_link'>0 0 0 0 -0 0</pose>
      <parent>base_link</parent>
      <child>link_link</child>
      <axis>
        <xyz>1 0 0</xyz>
        <limit>
          <lower>-1e+16</lower>
          <upper>1e+16</upper>
        </limit>
      </axis>
      <axis2>
        <xyz>0 1 0</xyz>
```
