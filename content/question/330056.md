  body_markdown: "Yes, you can use a template.\n\n    template<typename ServiceT>\n\
    \    typename ServiceT::Response::SharedPtr send_request(\n      rclcpp::Node::SharedPtr\
    \ node,\n      typename rclcpp::Client<ServiceT>::SharedPtr client,\n      typename\
    \ ServiceT::Request::SharedPtr request)\n    {\n      auto result = client->async_send_request(request);\n\
    \      // Wait for the result.\n      if (rclcpp::spin_until_future_complete(node,\
    \ result) ==\n        rclcpp::executor::FutureReturnCode::SUCCESS)\n      {\n\
    \        return result.get();\n      } else {\n        return NULL;\n      }\n\
    \    }\n\n\nThen you can call it like this\n\n      using ServiceT1 = ....\n \
    \     using ServiceT2 = ....\n    \n      auto node = rclcpp::Node::make_shared(\"\
    client_node\");\n      auto client_1 = node->create_client<ServiceT1>(\"my_service_1_name\"\
    );\n      auto client_2 = node->create_client<ServiceT2>(\"my_service_2_name\"\
    );\n    \n      auto request_1 = std::make_shared<ServiceT1::Request>();\n   \
    \   auto result_1 = send_request<ServiceT1>(node, client_1, request_1);\n    \n\
    \      auto request_2 = std::make_shared<ServiceT2::Request>();\n      auto result_2\
    \ = send_request<ServiceT2>(node, client_2, request_2);\n\n\nP.S. In ROS2 this\
    \ is not the recommended style for creating nodes. You should create a class that\
    \ extends `rclcpp::Node` and define the `send_request` function and the clients\
    \ as members of this class."
