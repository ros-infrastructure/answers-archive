    <pre>\ndef bresenham_march(img, p1, p2):\n   x1 = p1[0]\n   y1 = p1[1]\n   x2\
    \ = p2[0]\n   y2 = p2[1]\n   steep = math.fabs(y2 - y1) > math.fabs(x2 - x1)\n\
    \   if steep:\n      t = x1\n      x1 = y1\n      y1 = t\n\n      t = x2\n   \
    \   x2 = y2\n      y2 = t\n   also_steep = x1 > x2\n   if also_steep:\n      \n\
    \      t = x1\n      x1 = x2\n      x2 = t\n\n      t = y1\n      y1 = y2\n  \
    \    y2 = t\n\n   dx = x2 - x1\n   dy = math.fabs(y2 - y1)\n   error = 0.0\n \
    \  delta_error = 0.0; # Default if dx is zero\n   if dx != 0:\n       delta_error\
    \ = math.fabs(dy/dx)\n\n   if y1 < y2:\n      y_step = 1 \n   else:\n      y_step\
    \ = -1\n\n   y = y1\n   ret = list([])\n   for x in range(x1, x2):\n      if steep:\n\
    \         p = (y, x)\n      else:\n         p = (x, y)\n      \n      (b,r,g,a)\
    \ = (-1,)*4\n      if p[0] < img.width and p[1] < img.height:\n         (b,r,g,\
    \ a) = cv.Get2D(img, p[1], p[0])\n\n      ret.append((p,(b,r,g)))\n\n      error\
    \ += delta_error\n      if error >= 0.5:\n         y += y_step\n         error\
