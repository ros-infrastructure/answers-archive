  body_markdown: "Here is the code that worked for me, and as steve said on the comment\
    \ above, you have to pay attention for hardcoded topics.\nFor namespace:\n\n \
    \   <arg name=\"model\" default=\"$(find your_package)/urdf/robot.urdf.xacro\"\
    />\n    \n    <!-- Launch gazebo -->\n    <include file=\"$(find your_package)/launch/gazebo.launch\"\
    >\n      <arg name=\"model\" value=\"$(arg model)\"/>\n     </include>  \n   \
    \ \n    <group ns=\"robot1\">\n            <param name=\"tf_prefix\" value=\"\
    robot1_tf\" />\n        \n            <include file=\"$(find your_package)/launch/robot.launch\"\
    \ >\n              <arg name=\"init_pose\" value=\"-x 0 -y 1 -z 0\" />\n     \
    \         <arg name=\"robot_name\"  value=\"robot1\" />\n            </include>\n\
    \        \n            <include file=\"$(find ira_laser_tools)/launch/laserscan_multi_merger.launch\"\
    >\n              <arg name=\"robot_name\" value=\"robot1\"/>\n            </include>\n\
    \       </group>\n\nthen for robot.launch:\n\n    <launch>\n        <arg name=\"\
    robot_name\" default=\"your_default_value\"/>\n        <arg name=\"init_pose\"\
    \ default=\"your_default_value\"/>\n    \n    \n    <node name=\"urdf_spawner\"\
    \ pkg=\"gazebo_ros\" type=\"spawn_model\"\n         args=\"$(arg init_pose) -urdf\
    \ -param /robot_description -model $(arg robot_name)\"\n         respawn=\"false\"\
    \ output=\"screen\" />\n    \n        <node pkg=\"robot_state_publisher\" type=\"\
    robot_state_publisher\" \n              name=\"robot_state_publisher\" output=\"\
    screen\"/>\n    \n    </launch>"
