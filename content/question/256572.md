I have made an odometer using Arduino and a couple of IR LEDs. Below is the sketch:


    int counter; 
    int currReading; 
    int prevReading = 0; 
    int threashold = 600;
    
    void setup()
    {   
     Serial.begin(9600); 
    }
    
    void loop() 
    {
      currReading = analogRead (A0);
      delay(1);     
      if(currReading > threashold &&  prevReading < threashold)
    {
      counter = ++counter;   
    }
      prevReading = currReading;
      Serial.println (counter);    
    }
