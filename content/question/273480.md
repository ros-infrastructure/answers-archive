  body_markdown: "Hello,\n\nI've created a new message to do that, including an array\
    \ and another field, just to show how you can do that.\n\nThe command to create\
    \ the package: `catkin_create_pkg my_pkg roscpp std_msgs geometry_msgs`\n\nThe\
    \ message definition (**my_pkg/msg/my_msg.msg**)\n\n    geometry_msgs/Point[]\
    \ points\n    uint8 another_field\n\nIn the **my_pkg/package.xml** file, don't\
    \ forget to add the dependencies for generating the messages:\n\n      <build_depend>message_generation</build_depend>\n\
    \      <run_depend>message_runtime</run_depend>\n\n**my_pkg/CMakeLists.txt** file:\n\
    \n    cmake_minimum_required(VERSION 2.8.3)\n    project(my_pkg)\n    \n    find_package(catkin\
    \ REQUIRED COMPONENTS\n      geometry_msgs\n      roscpp\n      std_msgs\n   \
    \   message_generation\n    )\n    \n    ## Generate messages in the 'msg' folder\n\
    \    add_message_files(\n      FILES\n      my_msg.msg\n    #   Message2.msg\n\
    \    )\n    \n    ## Generate added messages and services with any dependencies\
    \ listed here\n    generate_messages(\n      DEPENDENCIES\n      geometry_msgs#\
    \   std_msgs\n    )\n    \n    catkin_package(\n    #  INCLUDE_DIRS include\n\
    \    #  LIBRARIES my_pkg\n       CATKIN_DEPENDS geometry_msgs roscpp std_msgs\
    \ message_runtime\n    #  DEPENDS system_lib\n    )\n    \n    include_directories(\n\
    \      ${catkin_INCLUDE_DIRS}\n    )\n    \n    add_executable(my_publisher src/my_publisher.cpp)\n\
    \    add_executable(my_subscriber src/my_subscriber.cpp)\n\n    add_dependencies(my_publisher\
    \ ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})\n    add_dependencies(my_subscriber\
    \ ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})\n    \n    target_link_libraries(my_publisher\
    \ ${catkin_LIBRARIES})\n    target_link_libraries(my_subscriber ${catkin_LIBRARIES})\n\
    \nFinally, the publisher and subscriber files:\n\n**my_pkg/src/my_publisher.cpp**\n\
    \n    #include \"ros/ros.h\"\n    #include \"geometry_msgs/Point.h\"\n    #include\
    \ \"my_pkg/my_msg.h\"\n    \n    #include <vector>\n    \n    struct Point {\n\
    \        float x;\n        float y;\n    };\n    \n    int main(int argc, char\
    \ **argv)\n    {\n      // ROS objects\n      ros::init(argc, argv, \"my_publisher\"\
    );\n      ros::NodeHandle n;\n      ros::Publisher pub = n.advertise<my_pkg::my_msg>(\"\
    my_topic\", 1);\n      ros::Rate loop_rate(0.5);\n      \n      // the message\
    \ to be published\n      my_pkg::my_msg msg;\n      msg.another_field = 0;\n \
    \     \n      // creating the vector\n      Point my_array[11];\n      Point point;\n\
    \      for (int i=0; i < 11; i++) {\n        point.x = i;\n        point.y = i;\n\
    \        my_array[i] = point;\n      }\n      std::vector<Point> my_vector (my_array,\
    \ my_array + sizeof(my_array) / sizeof(Point));\n      \n      // loop control\n\
    \      int count = 0;\n      while (ros::ok())\n      {\n        msg.points.clear();\n\
    \        \n        msg.another_field = count;\n        int i = 0;\n        for\
    \ (std::vector<Point>::iterator it = my_vector.begin(); it != my_vector.end();\
    \ ++it) {\n            geometry_msgs::Point point;\n            point.x = (*it).x;\n\
    \            point.y = (*it).y;\n            point.z = 0;\n            msg.points.push_back(point);\n\
    \            i++;\n        }\n      \n        ROS_INFO(\"%d\", msg.another_field);\n\
    \        \n        pub.publish(msg);\n    \n        ros::spinOnce();\n    \n \
    \       loop_rate.sleep();\n        ++count;\n      }\n    \n    \n      return\
    \ 0;\n    }\n\n**my_pkg/src/my_subscriber.cpp**\n\n    #include \"ros/ros.h\"\n\
    \    #include \"my_pkg/my_msg.h\"\n    \n    void clbk(const my_pkg::my_msg::ConstPtr&\
    \ msg) {\n        ROS_INFO(\"%d\", msg->another_field);\n        ROS_INFO(\"first\
    \ point: x=%.2f, y=%.2f\", msg->points[0].x, msg->points[0].y);\n    }\n    \n\
    \    int main(int argc, char **argv)\n    {\n      // ROS objects\n      ros::init(argc,\
    \ argv, \"my_subscriber\");\n      ros::NodeHandle n;\n      ros::Subscriber sub\
    \ = n.subscribe(\"my_topic\", 1, clbk);\n      \n      ros::spin();\n      \n\
    \    }\n\nIf you want to see step-by-step, I've created a video (https://youtu.be/4eHWSXGqXg8)."
