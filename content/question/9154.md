  body_markdown: "Thanks fergs. It was actually pretty easy once I looked at the geometric_shapes\
    \ package (motion_planning_common stack - version 3.0 or newer). I copied code\
    \ from 2 different locations to accomplish the 2 things I listed above.\n\n\n\
    \    bool SBPLCollisionSpace::getBoundingCylinderOfMesh(std::string mesh_file,\
    \ shapes::Shape &mesh, bodies::BoundingCylinder &cyl)\n    {\n      bool retval\
    \ = false;\n    \n      if (!mesh_file.empty())\n      {\n        resource_retriever::Retriever\
    \ retriever;\n        resource_retriever::MemoryResource res;\n        bool ok\
    \ = true;\n    \n        try\n        {\n          res = retriever.get(mesh_file);\n\
    \        }\n        catch (resource_retriever::Exception& e)\n        {\n    \
    \      ROS_ERROR(\"%s\", e.what());\n          ok = false;\n        }\n    \n\
    \        if (ok)\n        {\n          if (res.size == 0)\n            ROS_WARN(\"\
    Retrieved empty mesh for resource '%s'\", mesh_file.c_str());\n          else\n\
    \          {\n            mesh = shapes::createMeshFromBinaryStlData(reinterpret_cast<char*>(res.data.get()),\
    \ res.size);\n            if (mesh == NULL)\n              ROS_ERROR(\"Failed\
    \ to load mesh '%s'\", mesh_file.c_str());\n            else\n              retval\
    \ = true;\n          }\n        }\n      }\n      else\n        ROS_WARN(\"Empty\
    \ mesh filename\");\n    \n      if(retval)\n      {\n        const bodies::Body\
    \ *body=new bodies::ConvexMesh(mesh);\n        body->computeBoundingCylinder(cyl);\n\
    \        ROS_INFO(\"Bounding cylinder has radius: %0.3f  length: %0.3f\", cyl.radius,\
    \ cyl.length);\n      }\n    \n      return retval;\n    }\n\n"
