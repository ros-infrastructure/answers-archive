    \ want to do is clear your own layer.  One option is to disable your layer using\
    \ dynamic reconfigure, like in the [Creating a New Layer tutorial](http://wiki.ros.org/costmap_2d/Tutorials/Creating%20a%20New%20Layer):\n\
    \n    #include<simple_layers/simple_layer.h>\n    #include <pluginlib/class_list_macros.h>\n\
    \    \n    PLUGINLIB_EXPORT_CLASS(simple_layer_namespace::SimpleLayer, costmap_2d::Layer)\n\
    \    \n    using costmap_2d::LETHAL_OBSTACLE;\n    \n    namespace simple_layer_namespace\n\
    \    {\n    \n    SimpleLayer::SimpleLayer() {}\n    \n    void SimpleLayer::onInitialize()\n\
    \    {\n      ros::NodeHandle nh(\"~/\" + name_);\n      current_ = true;\n  \
    \  \n      dsrv_ = new dynamic_reconfigure::Server<costmap_2d::GenericPluginConfig>(nh);\n\
    \      dynamic_reconfigure::Server<costmap_2d::GenericPluginConfig>::CallbackType\
    \ cb = boost::bind(\n          &SimpleLayer::reconfigureCB, this, _1, _2);\n \
    \     dsrv_->setCallback(cb);\n    }\n    \n    \n    void SimpleLayer::reconfigureCB(costmap_2d::GenericPluginConfig\
    \ &config, uint32_t level)\n    {\n      enabled_ = config.enabled;\n    }\n \
    \   \n    void SimpleLayer::updateBounds(double robot_x, double robot_y, double\
    \ robot_yaw, double* min_x,\n                                               double*\
    \ min_y, double* max_x, double* max_y)\n    {\n      if (!enabled_)\n        return;\n\
    \    \n      mark_x_ = robot_x + cos(robot_yaw);\n      mark_y_ = robot_y + sin(robot_yaw);\n\
    \    \n      *min_x = std::min(*min_x, mark_x_);\n      *min_y = std::min(*min_y,\
    \ mark_y_);\n      *max_x = std::max(*max_x, mark_x_);\n      *max_y = std::max(*max_y,\
    \ mark_y_);\n    }\n    \n    void SimpleLayer::updateCosts(costmap_2d::Costmap2D&\
    \ master_grid, int min_i, int min_j, int max_i,\n                            \
    \                  int max_j)\n    {\n      if (!enabled_)\n        return;\n\
    \      unsigned int mx;\n      unsigned int my;\n      if(master_grid.worldToMap(mark_x_,\
    \ mark_y_, mx, my)){\n        master_grid.setCost(mx, my, LETHAL_OBSTACLE);\n\
    \      }\n    }\n    \n    } // end namespace\n\nNotice that the `enabled_` flag\

    \ = new dynamic_reconfigure::Client<SimpleLayerConfig>(\"move_base/local_costmap/simple_layer\"\
    );\n    SimpleLayerConfig layer_config;\n    layer_client_local_->getCurrentConfiguration(layer_config,\
    \ ros::Duration(1.0));\n    layer_config.enabled = false;\n    layer_client_local_->setConfiguration(layer_config);\n\
