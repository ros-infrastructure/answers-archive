  body_markdown: "My experience when I tried to use the MeanFilter by itself, was\
    \ that there was something not working with the passing of the number_of_observations\
    \ parameter, as I tried every namespace combination with no luck. I ended having\
    \ to create a new class that extended the MeanFilter just to override the configure()\
    \ method so I could pass the parameter then. Not very elegant, but as I said,\
    \ I could not get the filter to work when instantiating it directly. \nYou may\
    \ also see this answer, although it didn't work for me:\nhttps://answers.ros.org/question/284761/setting-parameters-for-mean-filter/\n\
    \nSee example below:\n\n    #include <ros/ros.h>\n    #include <sensor_msgs/Imu.h>\n\
    \    #include <filters/mean.h>\n    \n    class MyMeanFilter: public filters::MeanFilter\
    \ <double>\n    {\n    public:\n      bool configure(int num_obs){\n        number_of_observations_\
    \ = num_obs;\n        data_storage_.reset(new filters::RealtimeCircularBuffer<double>(number_of_observations_,\
    \ temp_));\n        return true;\n      }\n    };\n    \n    class SimpleImuFilterNode\n\
    \    {\n    protected:\n      ros::NodeHandle nh_;\n      ros::Publisher output_pub_;\n\
    \      ros::Subscriber imu_sub_;\n      int num_obs_;\n      MyMeanFilter mean_filter_;\n\
    \    \n    public:\n      SimpleImuFilterNode() \n      : mean_filter_()\n   \
    \   {\n        ros::NodeHandle pnh_(\"~\");\n        output_pub_ = nh_.advertise<sensor_msgs::Imu>(\"\
    imu_filtered_out\", 10);\n        imu_sub_    = nh_.subscribe(\"imu_in\", 1, &SimpleImuFilterNode::callback,\
    \ this);\n        pnh_.param<int>(\"number_of_observations\", num_obs_, 5);\n\
    \        mean_filter_.configure(num_obs_);\n      }\n    \n      void callback(const\
    \ sensor_msgs::Imu::ConstPtr& msg_in)\n      {\n        sensor_msgs::Imu out_msg_;\n\
    \        out_msg_.header = msg_in->header;\n        mean_filter_.update (msg_in->linear_acceleration.x,\
    \ out_msg_.linear_acceleration.x);\n        output_pub_.publish(out_msg_);\n \
    \     }\n    };\n    \n    int main(int argc, char **argv)\n    {\n      ros::init(argc,\
    \ argv, \"imu_filter_node\");\n      SimpleImuFilterNode simple_imu_filter_node;\n\
    \      ros::spin();\n      return 0;\n    }"
