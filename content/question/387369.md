Subscriber:

    // Differential drive kinematics conversion to motor control inputs.
    
    #include <ros.h>
    #include <geometry_msgs/Twist.h>
    #include <std_msgs/Float64.h>
    #include <knet_bot_msgs/EncoderSpeed.h>
    
    #include "CytronMotorDriver.h"
    
    // Constant definitions
    #define PI 3.1415926535897932384626433832795
    
    // Robot parameters
    float gear_ratio = 98.78;
    float wheel_radius = 0.08;
    float base_length = 0.171;
    float power_supply_voltage = 24.0;
    
    // Motor parameters
    float max_motor_rpm = 10000;
    float max_motor_voltage = 12.0;
    float max_wheel_rpm = max_motor_rpm / gear_ratio;
    float max_serial_scale_speed = 250.0 / (power_supply_voltage / max_motor_voltage); 
    float max_wheel_linear_speed = max_wheel_rpm * 2.0 * PI * wheel_radius / 60.0; // m/sec
    
    // Feedback parameters
    float right_motor_rpm;
    float left_motor_rpm;
    
    // Configure the motor driver.
    CytronMD motor_right(PWM_DIR, 9, 8);  // PWM 1 = Pin 3, DIR 1 = Pin 4.
    CytronMD motor_left(PWM_DIR, 11, 13); // PWM 2 = Pin 9, DIR 2 = Pin
    
    // Initialize ROS parameters
    ros::NodeHandle nh;
    
    void motor_callback(const geometry_msgs::Twist& velocity_msg){
      nh.loginfo("Message to motor callback received.");
      
      // Velocity inputs are v_x and w_z. Others are 0.
      float diff_drive_right_speed = 1/wheel_radius * velocity_msg.linear.x + base_length/(2*wheel_radius) * velocity_msg.angular.z;
      float diff_drive_left_speed = 1/wheel_radius * velocity_msg.linear.x - base_length/(2*wheel_radius) * velocity_msg.angular.z;
      
      // Check speed saturation limits.
      if(abs(diff_drive_right_speed) > max_wheel_linear_speed){
          diff_drive_right_speed = diff_drive_right_speed/abs(diff_drive_right_speed) * max_wheel_linear_speed;
        }
      if(abs(diff_drive_left_speed) > max_wheel_linear_speed){
          diff_drive_left_speed = diff_drive_left_speed/abs(diff_drive_left_speed) * max_wheel_linear_speed;
        }
    
      // Convert wheel linear speeds to motor angular velocity in RPM
      float motor_right_speed_input_rpm = 60.0 * diff_drive_right_speed * gear_ratio / (2.0 * PI * wheel_radius);
      float motor_left_speed_input_rpm = 60.0 * diff_drive_left_speed * gear_ratio / (2.0 * PI * wheel_radius);
      
      // Convert to Arduino serial scale.
      int motor_right_serial_scale_speed_input = (int) motor_right_speed_input_rpm * max_serial_scale_speed / max_motor_rpm;
      int motor_left_serial_scale_speed_input = (int) motor_left_speed_input_rpm * max_serial_scale_speed / max_motor_rpm;
    
      // Voltage saturation limits.
      if(abs(motor_right_serial_scale_speed_input) > max_serial_scale_speed){
        motor_right_serial_scale_speed_input = motor_right_serial_scale_speed_input / abs(motor_right_serial_scale_speed_input) *  max_serial_scale_speed;
        }
      if(abs(motor_left_serial_scale_speed_input) > max_serial_scale_speed){
        motor_left_serial_scale_speed_input =  motor_left_serial_scale_speed_input / abs(motor_left_serial_scale_speed_input) * max_serial_scale_speed;
        }
    
      // Command motor speeds.
      char right_motor_command_str[8]; 
      dtostrf(motor_right_speed_input_rpm, 6, 2, right_motor_command_str);
      char left_motor_command_str[8];
      dtostrf(motor_left_speed_input_rpm, 6, 2, left_motor_command_str);
    
      //sprintf(log_msg,"points[0].positions[0] =%s", right_motor_command_str);
      nh.loginfo(right_motor_command_str);
      nh.loginfo(left_motor_command_str);
      
      motor_right.setSpeed(motor_right_serial_scale_speed_input);   
      motor_left.setSpeed(motor_left_serial_scale_speed_input); 
    }
    
    void odometry_callback(const knet_bot_msgs::EncoderSpeed& odometry_msg){
      right_motor_rpm = odometry_msg.right_wheel_rpm * gear_ratio;
      left_motor_rpm = odometry_msg.left_wheel_rpm * gear_ratio;
      }
    
    void setup()
    {
      nh.initNode();
      ros::Subscriber<geometry_msgs::Twist> vel_subscriber("velocity_command", &motor_callback);
      ros::Subscriber<knet_bot_msgs::EncoderSpeed> vel_subscriber("odometry/wheels_rpm", &odometry_callback);
      nh.subscribe(vel_subscriber);
      nh.getHardware()->setBaud(57600);
      }
    
    void loop()
    {
      nh.spinOnce();
      delay(100);
    }
