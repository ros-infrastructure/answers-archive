    \ you set and animation, and get a callback on completion.\n\nHere is the api:\n\

    \nWhat you can do is:\n\n    #include <boost/bind.hpp>\n    #include \"common/CommonTypes.hh\"\
    \n    #include \"common/Animation.hh\"\n    #include \"common/KeyFrame.hh\"\n\

    namespace gazebo

      class AnimationTest : public WorldPlugin
