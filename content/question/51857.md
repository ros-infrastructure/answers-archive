  body_markdown: "Below is a bit of python that will do what you want. I tried to\
    \ make it a bit generic for you so you can extend it.\n\n\n    #!/usr/bin/python\
    \                                                              \n    \n    import\
    \ roslib; roslib.load_manifest('test')\n    import rospy\n    import sys\n   \
    \ \n    from geometry_msgs.msg import Twist\n    \n    class Walk(object):\n \
    \   \n        def __init__(self):\n            rospy.init_node('walk_reader_node')\n\
    \            self._init_params()\n            self._init_pubsub()\n          \
    \  self.fh = open(self.file)\n    \n        def _init_pubsub(self):\n        \
    \    self.cmd_vel_pub = rospy.Publisher('cmd_vel', Twist)\n    \n        def _init_params(self):\n\
    \            self.file = rospy.get_param('~file', 'walk.dat')\n    \n        def\
    \ run(self):\n            r = rospy.Rate(10.0)\n            while not rospy.is_shutdown():\n\
    \                for line in self.fh:\n                    l = line.split()\n\
    \                    cmd = Twist()\n                    cmd.linear.x = float(l[0])\n\
    \                    cmd.angular.z = float(l[1])\n                    self.cmd_vel_pub.publish(cmd)\n\
    \                    r.sleep()\n                self.fh.close()\n            \
    \    exit()\n    \n    if __name__ == '__main__':\n        w = Walk()\n      \
    \  w.run()\n\n"
