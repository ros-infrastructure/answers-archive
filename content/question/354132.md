    \ your code here. This can possibly lead to a race condition\n\n[https://github.com/ros2/rclcpp/issues/206](https://github.com/ros2/rclcpp/issues/206)\n\
    \nThere are multiple possible solutions .Either run your rclcpp::spin in a separate\

    \nOR \n\nHow I have worked around is using the *std::future<T>::get()\n\n    auto\
    \ request = std::make_shared<turtlesim::srv::Spawn::Request>();\n    request->x\
    \ = x;\n    request->y = y;\n    request->theta = theta;\n    request->name =\
    \ name;\n\n    RCLCPP_WARN(this->get_logger(), \"Spawning '%s'  to '%d'   '%d'\"\
    , name.c_str(), x, y);\n    while (!client->wait_for_service(1s))\n    {\n   \
    \     if (!rclcpp::ok())\n        {\n            RCLCPP_ERROR(rclcpp::get_logger(\"\
    rclcpp\"), \"Interrupted while waiting for the service. Exiting.\");\n       \
    \     return;\n        }\n        RCLCPP_INFO(rclcpp::get_logger(\"rclcpp\"),\
    \ \"service not available, waiting again...\");\n    }\n\n    auto future = client->async_send_request(request);\n\
