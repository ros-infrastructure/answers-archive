    brew install caskroom/cask/xquartz
    brew tap osrf/simulation
    brew install gazebo9 --only-dependencies
    brew install mercurial

    hg clone https://bitbucket.org/osrf/gazebo /tmp/gazebo

    make -j4
    make install
