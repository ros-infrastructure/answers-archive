    pcm.!default {
        type hw
        card 0
    }

    ctl.!default {
            type hw
            card 0
    }

    pcm.sc {
        type hw
        card 1
        format S16_LE
        channels 1
        rate 16000
    }

    pcm.mic_channel0 {
      type file
      file "/tmp/matrix_micarray_channel_0"
      infile "/tmp/matrix_micarray_channel_0"
      format "raw"
      slave {
        pcm sc
      }
