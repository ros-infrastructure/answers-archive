                for x in range(0,10):

            else: 
                rospy.loginfo("Going Straight")
                for x in range(0,10):
                    self.cmd_vel.publish(move_cmd)
                    r.sleep()
	    # turn 90 degrees
 

	        rospy.loginfo("Turning")
                for x in range(0,10): 
                    self.cmd_vel.publish(turn_cmd)

    def shutdown(self):
