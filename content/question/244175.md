    \ do ultimately, but you might want to use [smach](http://wiki.ros.org/smach)\
    \ .\nIt is a state machine package in ROS.\n\nQuoting from the page:\n\n> When\
    \ should I use SMACH?\n> \n> SMACH is useful when you want a robot\n> to execute\
    \ some complex plan, where\n> all possible states and state\n> transitions can\
    \ be described\n> explicitly. This basically takes the\n> hacking out of hacking\
    \ together\n> different modules to make systems like\n> mobile robotic manipulators\
    \ do\n> interesting things.\n\nYou can also control the goal/behavior of the robot\
    \ depending on the result of the previous goal.\n\nCheck out the [tutorial](http://wiki.ros.org/smach/Tutorials)\n\
    \n------.\n\nHere is an example from \"Programming Robots with ROS\" book:\n\n\
    \    #!/usr/bin/env python\n    \n    import rospy\n    import actionlib\n   \
    \ from smach import State,StateMachine\n    from move_base_msgs.msg import MoveBaseAction,\
    \ MoveBaseGoal\n    \n    waypoints = [\n        ['one', (2.1, 2.2), (0.0, 0.0,\
    \ 0.0, 1.0)],\n        ['two', (6.5, 4.43), (0.0, 0.0, -0.984047240305, 0.177907360295)]\n\
    \    ]\n    \n    class Waypoint(State):\n        def __init__(self, position,\
    \ orientation):\n            State.__init__(self, outcomes=['success'])\n    \n\
    \            # Get an action client\n            self.client = actionlib.SimpleActionClient('move_base',\
    \ MoveBaseAction)\n            self.client.wait_for_server()\n    \n         \
    \   # Define the goal\n            self.goal = MoveBaseGoal()\n            self.goal.target_pose.header.frame_id\
    \ = 'map'\n            self.goal.target_pose.pose.position.x = position[0]\n \
    \           self.goal.target_pose.pose.position.y = position[1]\n            self.goal.target_pose.pose.position.z\
    \ = 0.0\n            self.goal.target_pose.pose.orientation.x = orientation[0]\n\
    \            self.goal.target_pose.pose.orientation.y = orientation[1]\n     \
    \       self.goal.target_pose.pose.orientation.z = orientation[2]\n          \
    \  self.goal.target_pose.pose.orientation.w = orientation[3]\n    \n        def\
    \ execute(self, userdata):\n            self.client.send_goal(self.goal)\n   \
    \         self.client.wait_for_result()\n            return 'success'\n    \n\
    \    \n    if __name__ == '__main__':\n        rospy.init_node('patrol')\n   \
    \ \n        patrol = StateMachine('success')\n        with patrol:\n         \
    \   for i,w in enumerate(waypoints):\n                StateMachine.add(w[0],\n\
    \                                 Waypoint(w[1], w[2]),\n                    \
    \             transitions={'success':waypoints[(i + 1) % \\\n                \
    \                 len(waypoints)][0]})\n    \n        patrol.execute()"
