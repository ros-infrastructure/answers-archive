    \ image_sub_;\n        message_filters::Subscriber<sensor_msgs::CameraInfo> camera_info_sub_;\n\
    \        typedef message_filters::TimeSynchronizer<sensor_msgs::Image, sensor_msgs::CameraInfo>\
    \ approx_sync_policy;\n        std::shared_ptr<approx_sync_policy> sync;\n\n \

    \ camera_info_sub_, 10));\n        sync->registerCallback(std::bind(&ImageConverter::MultiSubscriberCB,\
