  body_markdown: "The relevant topic names and message types have been shuffled a\
    \ bit since that tutorial was last updated. You'll want to replace `gps_common::GPSFix`\
    \ with `gps_common::NavSatFix` (or `sensor_msgs::NavSatFix` if you're using Diamondback),\
    \ and your `#include` should point to `gps_common/NavSatFix.h`. You'll also need\
    \ to have a `<depend package=\"gps_common\"/>` line in your package's manifest.xml.\
    \ If you're using Cturtle, set gpsd_client's `~report_as` parameter to \"navsat\"\
    .\n\n    #include <ros/ros.h>\n    // Change following lines to \"gps_common\"\
    \ if you're using Cturtle\n    #include <sensor_msgs/NavSatFix.h>\n    #include\
    \ <sensor_msgs/NavSatStatus.h>\n    using sensor_msgs;\n\n    void callback(const\
    \ NavSatFixConstPtr &fix) {\n      if (fix->status.status == NavSatStatus::STATUS_NO_FIX)\
    \ {\n        std::cout << \"Unable to get a fix on the location.\" << std::endl;\n\
    \        return;\n      }\n      \n      std::cout << \"Current Latitude: \" <<\
    \ fix->latitude << std::endl;\n      std::cout << \"Current Longitude \" << fix->longitude\
    \ << std::endl;\n    }\n\n    int main(int argc, char **argv) {\n      ros::init(argc,\
    \ argv, \"gps_subscriber\");\n      ros::NodeHandle nh;\n\n      ros::Subscriber\
    \ gps_sub = nh.subscribe(\"fix\", 1, callback);\n\n      ros::spin();\n      return\
    \ 0;\n    }\n"
