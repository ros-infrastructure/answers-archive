    painter\"/>\n\n    <group if=\"$(eval team == 'Red')\">\n      <group if=\"$(eval\
    \ type == 'painter')\">\n        <param name=\"robot_description\" \n        \
    \  command=\"$(find xacro)/xacro $(find robopaint)/urdf/red/paintbot_red.urdf.xacro\"\
    \ />\n      </group>\n      <group if=\"$(eval type == 'attacker')\">\n      \

    \   </group>\n\n    <group if=\"$(eval team == 'Blue')\">\n      <group if=\"\
