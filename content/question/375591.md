    \ an empty launch description:\n    ld = LaunchDescription()\n    \n    # define\
    \ the launch argument and launch configuration for the variable in question:\n\
    \    my_param = LaunchConfiguration('my_param')\n    my_param_arg = DeclareLaunchArgument('my_param',\
    \ default_value='val')\n    \n    # then when you want to access the value at\
    \ runtime, you can use the launch configuration (`my_param`); e.g.:\n    info_cmd\
    \ = LogInfo(msg=my_param)\n    node_cmd = Node(package='my_package', executable='my_executable',\
    \ output='screen', 'arguments=['--my-param', my_param])\n    \n    # and of course,\
    \ add everything to the launch description:\n    ld.add_action(my_param)\n   \
