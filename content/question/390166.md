    \n\n    #include <ArduinoHardware.h>\n    #include <ros.h>\n    #include <geometry_msgs/Twist.h>\n\
    \    \n    ros::NodeHandle nh;\n    \n    geometry_msgs::Twist msg;\n    \n  \
    \  ros::Publisher pub(\"husky/cmd_vel\", &msg);\n    \n    void setup()\n    {\n\
    \     nh.initNode();\n     nh.advertise(pub);\n    } void loop()\n    {\n    \
    \ if(digitalRead(8)==1)\n     msg.linear.x=-0.25;\n    \n    else if (digitalRead(4)==1)\n\
    \    msg.linear.x=0.25;\n    \n    else if (digitalRead(8)==0 && digitalRead(4)==0)\n\
    \    msg.linear.x=0;\n    \n    pub.publish(&msg);\n    nh.spinOnce();\n    }\n\
    \nAs you can see a `Twist` message is published to the `cmd_vel` topic. Twist\
    \ message is composed of Linear and Angular velocities as can be seen here:\n\n\
    ![image description](https://ftp.osuosl.org/pub/ros/download.ros.org/downloads/se_migration/ros/16362930861284841.png)\n\
    \nOnce code loaded in Arduino and publishing velocity commands, we can pass these\
    \ messages along into our ROS environment through `rosserial`\n\n\n"
