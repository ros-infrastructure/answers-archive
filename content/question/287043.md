  body_markdown: "You should not stay in the callbacks - just alter the values of\
    \ variables to indicate what should happen in the main loop and work out whether\
    \ it is time to toggle the led each time round if it is blinking\n\n   \n\n  \
    \  # include <ros.h>\n    # include <std_msgs/Empty.h>\n    \n    int blinking\
    \ = 0;\n    int ledState = 0;\n    long lastTime = 0;\n    long blinkInterval\
    \ = 5000;\n    \n    ros::NodeHandle  nh;\n    \n    void messageCb(const std_msgs::Empty&\
    \ toggle_msg){\n      blinking = 1;\n      lastTime = millis();\n      ledState\
    \ = 1;\n    }\n    \n    \n    void messageCb1(const std_msgs::Empty& end_msg){\n\
    \      blinking = 0;\n    }\n    \n    ros::Subscriber<std_msgs::Empty> sub(\"\
    toggle_led\", &messageCb);\n    ros::Subscriber<std_msgs::Empty> sub1(\"end_led\"\
    , &messageCb1);\n    \n    \n    void setup(){\n      pinMode(13, OUTPUT);\n \
    \     nh.initNode();\n      nh.subscribe(sub);\n      nh.subscribe(sub1);\n  \
    \  }\n    \n    void loop(){\n      nh.spinOnce();\n    \n      if (blinking){\n\
    \        long currentTime = millis();\n        if (currentTime - lastTime >= blinkInterval){\n\
    \          ledState = !ledState;\n          lastTime = currentTime;\n        }\n\
    \      }else{\n        ledState = 0;\n      }\n      digitalWrite(13, ledState);\n\
    \    }"
