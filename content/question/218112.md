    void MyClass::lcm_cb(const lcm::ReceiveBuffer *rbuf,
            const string &chan,
            const mav::pose_t *msg) {
         // DO OTHER STUFF
    }
    
    MyClass::run() {
        ros::spin();
    }
    
    int main(int argc, char **argv) {
        ros::init(argc, argv, "my_node");
        ros::NodeHandle nh;
        MyClass mc(nh);
