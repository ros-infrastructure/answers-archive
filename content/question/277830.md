    \ solution is a good idea with others:\n\n    import rospy\n    import math\n\
    \    from sensor_msgs.msg import LaserScan\n    from math import *\n    import\
    \ numpy as np\n    import copy\n\n    ranges_filter = []\n    intensities_filter\
    \ = []\n\n    #copy the range and intensities of \"/scan\" topic to \"ranges_filter\"\
    \ and \"intensities_filter\" \n    #you need to convert them to \"list\" as \"\
    data.ranges\" and \"data.intensities\" are \"tuple\"\n    def callback_scan(data):\n\
    \        global ranges_filter, intensities_filter\n\n        len(data.ranges)\
    \ #360\n        len(data.intensities) #360\n\n        ranges_filter = copy.copy(data.ranges)\n\
    \        intensities_filter = copy.copy(data.intensities)\n    \n        #convert\
    \ them to list\n        ranges_filter = list(ranges_filter)\n        intensities_filter\
    \ = list(intensities_filter)\n\n         #filtering those angles that I do not\
    \ want them (based on the question)\n        for x in range(45, 135):\n      \
    \      ranges_filter[x] = 0\n            intensities_filter[x] = 0\n        \n\
    \        for y in range(225, 315):\n            ranges_filter[y] = 0\n       \
    \     intensities_filter[y] = 0\n  \n    #define a new topic called \"filterScan\"\
    \ to store all laser scanner data\n    rospy.init_node('laser_scan_filter')\n\n\
    \    scan_pub = rospy.Publisher('filterScan', LaserScan, queue_size=50)\n\n  \
    \  rospy.Subscriber(\"/scan\", LaserScan, callback_scan) \n\n    #it is based\
    \ on the type of laser scanner (length of data.ranges)\n    num_readings = 360\n\
    \    laser_frequency = 60\n\n    count = 0\n    r = rospy.Rate(1.0)\n    while\
    \ not rospy.is_shutdown():\n        current_time = rospy.Time.now()\n\n      \
    \  filterScan = LaserScan()\n\n        filterScan.header.stamp = current_time\n\
    \        filterScan.header.frame_id = 'base_scan'\n        filterScan.angle_min\
    \ = 0 # start angle of the scan [rad]\n        filterScan.angle_max = math.pi\
    \ * 2  # end angle of the scan [rad]\n        filterScan.angle_increment = 0.0174532923847\
    \ # angular distance between measurements [rad]\n        filterScan.time_increment\
    \ = 2.98699997074e-05 # time between measurements [seconds]\n        filterScan.range_min\
    \ = 0.0 # minimum range value [m]\n        filterScan.range_max = 3.5 # maximum\
    \ range value [m]\n\n        filterScan.ranges = []\n        filterScan.intensities\
    \ = []\n\n\n        for i in range(0, num_readings-1):\n            filterScan.ranges\
    \ = copy.copy(ranges_filter)\n\n            filterScan.intensities = copy.copy(intensities_filter)\n\
    \n        scan_pub.publish(filterScan)\n        r.sleep()\n\nI have done it in\
