     #include <ros/ros.h>

    using namespace std;
    
    bool moveToGoal(double start_world_x, double start_world_y, double goal_world_x, double goal_world_y)
    {
        //define a publisher to publish start pose
        ros::NodeHandle nh_;
        //bool called_back = false;
        ros::Publisher pub_ = nh_.advertise<geometry_msgs::PoseWithCovarianceStamped> ("/initialpose", 1);
        //ros::Subscriber sub_ = nh_.subscribe<geometry_msgs::PoseWithCovarianceStamped> ("/initialpose", 1, [&]{ called_back = true; });
    
    
        ros::Rate loop_rate(10); // 10HZ = 0.1s
        ros::spinOnce();
    
        std::string fixed_frame = "map";
        geometry_msgs::PoseWithCovarianceStamped pose;
        pose.header.frame_id = fixed_frame;
        pose.header.stamp = ros::Time::now();
    
        // set start_world x,y coord
        pose.pose.pose.position.x = start_world_x;
        pose.pose.pose.position.y = start_world_y;
        pose.pose.pose.position.z = 0.0;
        // set theta
        
        // publish
        //ROS_INFO("x: %f, y: %f, z: 0.0, theta: %f",x,y,theta);
        ROS_INFO("start_world_x: %f, start_world_y: %f, z: 0.0",start_world_x,start_world_y);
        ROS_INFO("Publishing start location ...");
        pub_.publish(pose);
        //if(called_back)
            //ROS_INFO("The start location has been published");
    
    
    
        //define a client to send goal requests to the move_base server through a SimpleActionClient
        actionlib::SimpleActionClient<move_base_msgs::MoveBaseAction> ac("move_base", true);
    
        //wait for the action server to come up
        while(!ac.waitForServer(ros::Duration(5.0)))
        {
             ROS_INFO("Waiting for the move_base action server to come up");
        }
    
        move_base_msgs::MoveBaseGoal goal;
    
        //set up the frame parameters
        goal.target_pose.header.frame_id = "map";
        goal.target_pose.header.stamp = ros::Time::now();
    
        /* moving towards the goal*/
        goal.target_pose.pose.position.x =  goal_world_x;
        goal.target_pose.pose.position.y =  goal_world_y;
        goal.target_pose.pose.position.z =  0.0;
        goal.target_pose.pose.orientation.x = 0.0;
        goal.target_pose.pose.orientation.y = 0.0;
        goal.target_pose.pose.orientation.z = 0.0;
        goal.target_pose.pose.orientation.w = 1.0;
        
        ROS_INFO("goal_world_x: %f, goal_world_y: %f, z: 0.0",goal_world_x,goal_world_y);
        ROS_INFO("Sending goal location ...");
        ac.sendGoal(goal);
        
        ac.waitForResult();
        
        if(ac.getState() == actionlib::SimpleClientGoalState::SUCCEEDED)
        {
            ROS_INFO("You have reached the destination");
            return true;
        }
        else
        {
            ROS_INFO("The robot failed to reach the destination");
            return false;
        }
        
    }//end function moveToGoal
