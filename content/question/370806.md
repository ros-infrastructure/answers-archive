    \   last_odom = None\n    pose = [0.0,0.0,0.0]\n    a1 = 0.0\n    a2 = 0.0\n \
    \   a3 = 0.0\n    a4 = 0.0\n    base_frame = \"\"\n     \n    # a callback for\
    \ Gazebo odometry data\n    def callback(data):\n        global last_odom\n  \
    \      global base_frame\n        global pose\n        global a1\n        global\
    \ a2\n        global a3\n        global a4\n     \n        if(last_odom == None):\n\
    \            last_odom = data\n            pose[0] = data.pose.pose.position.x\n\
    \            pose[1] = data.pose.pose.position.y\n            q = [ data.pose.pose.orientation.x,\n\
    \                    data.pose.pose.orientation.y,\n                    data.pose.pose.orientation.z,\n\
    \                    data.pose.pose.orientation.w ]\n            (r, p, y) = tf.transformations.euler_from_quaternion(q)\n\
    \            pose[2] = y\n        else:\n            # calculate new odometry\
    \ pose using the motion model\n            dx = data.pose.pose.position.x - last_odom.pose.pose.position.x\n\
    \            dy = data.pose.pose.position.y - last_odom.pose.pose.position.y\n\
    \            trans = sqrt(dx*dx + dy*dy)\n            q = [ last_odom.pose.pose.orientation.x,\n\
    \                    last_odom.pose.pose.orientation.y,\n                    last_odom.pose.pose.orientation.z,\n\
    \                    last_odom.pose.pose.orientation.w ]\n            (r,p, theta1)\
    \ = tf.transformations.euler_from_quaternion(q)\n            q = [ data.pose.pose.orientation.x,\n\
    \                    data.pose.pose.orientation.y,\n                    data.pose.pose.orientation.z,\n\
    \                    data.pose.pose.orientation.w ]\n            (r,p, theta2)\
    \ = tf.transformations.euler_from_quaternion(q)\n            rot1 = atan2(dy,\
    \ dx) - theta1\n            rot2 = theta2-theta1-rot1\n     \n            sd_rot1\
    \ = a1*abs(rot1) + a2*trans\n            sd_rot2 = a1*abs(rot2) + a2*trans\n \
    \           sd_trans = a3*trans + a4*(abs(rot1) + abs(rot2))\n     \n        \
    \    trans +=  np.random.normal(0,sd_trans*sd_trans)\n            rot1 += np.random.normal(0,\
    \ sd_rot1*sd_rot1)\n            rot2 += np.random.normal(0, sd_rot2*sd_rot2)\n\
    \     \n            pose[0] += trans*cos(theta1+rot1)\n            pose[1] +=\
    \ trans*sin(theta1+rot1)\n            pose[2] = theta1 + rot1 + rot2\n       \
    \     last_odom = data\n     \n        # publish the tf\n        br = tf.TransformBroadcaster()\n\
    \        br.sendTransform((pose[0], pose[1], 0),\n                         tf.transformations.quaternion_from_euler(0,\
    \ 0, pose[2]),\n                         data.header.stamp,\n                \
    \         base_frame,\n                         \"odom\")\n     \n    if __name__\
    \ == '__main__':\n        rospy.init_node('noisy_odometry', anonymous=True)\n\

    \            a1 = rospy.get_param(\"~alpha1\")\n        else:\n            rospy.logwarn(\"\

    ~alpha2\")\n        else:\n            a2 = 15.0*pi/180.0\n            rospy.logwarn(\"\
    alpha2 is set to default\")\n        # alpha 3 is m/meter\n        if rospy.has_param(\"\
    ~alpha3\"):\n            a3 = rospy.get_param(\"~alpha3\")\n        else:\n  \
    \          a3 = 0.2\n            rospy.logwarn(\"alpha3 is set to default\")\n\
    \        # alpha 4 is m/degree\n        if rospy.has_param(\"~alpha4\"):\n   \

    \        if rospy.has_param(\"~old_odom_topic\"):\n            odom_topic = rospy.get_param(\"\

    \  base_frame = rospy.get_param(\"~base_frame\")\n        else:\n            base_frame\
    \ = \"/base_footprint\"\n     \n        rospy.Subscriber(odom_topic, Odometry,\
    \ callback)\n     \n        rospy.spin()"
