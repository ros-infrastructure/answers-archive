    \                elif p.geom_type in ['Point', 'LineString']:\n              \
    \      xd,yd = p.xy\n                elif p.geom_type == 'MultiLineString':\n\
    \                    for ea in p:\n                        xd,yd = ea.xy\n   \
    \             elif p.geom_type == 'Polygon':\n                    xd,yd = p.exterior.coords.xy\n\
