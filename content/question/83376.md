bool query()
{
	ROS_INFO("Waiting for query");
	while(ros::ok())
	{
		if(speech.find("yes")!=string::npos)

		ROS_INFO("Yes!");
		return true;

		}

		{
		ROS_INFO("NO!");
		return false;
		break;
		}
	ros::spinOnce();


	}

}
