    \ constraints) with Moveit in Python:\n\n    from moveit_msgs.msg import RobotState,\
    \ Constraints, OrientationConstraint\n\n    def init_upright_path_constraints(self,pose):\n\
    \    \n        self.upright_constraints = Constraints()\n        self.upright_constraints.name\
    \ = \"upright\"\n        orientation_constraint = OrientationConstraint()\n  \
    \      orientation_constraint.header = pose.header\n        orientation_constraint.link_name\
    \ = self.arm.get_end_effector_link()\n        orientation_constraint.orientation\
    \ = pose.pose.orientation\n        orientation_constraint.absolute_x_axis_tolerance\
    \ = 0.4\n        orientation_constraint.absolute_y_axis_tolerance = 0.4\n    \
    \    orientation_constraint.absolute_z_axis_tolerance = 0.4\n        #orientation_constraint.absolute_z_axis_tolerance\
    \ = 3.14 #ignore this axis\n        orientation_constraint.weight = 1\n    \n\
    \        self.upright_constraints.orientation_constraints.append(orientation_constraint)\n\
    \    \n    \n    def enable_upright_path_constraints(self):\n        self.arm.set_path_constraints(self.upright_constraints)\n\
    \    \n    def disable_upright_path_constraints(self):\n        self.arm.set_path_constraints(None)"
