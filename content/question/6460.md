    \    \tstruct termios cooked, raw;\n    \t\t  // get the console in raw mode \
    \                                                             \n    \t  tcgetattr(kfd,\
    \ &cooked);\n    \t  memcpy(&raw, &cooked, sizeof(struct termios));\n    \t  raw.c_lflag\
    \ &=~ (ICANON | ECHO);\n    \t  // Setting a new line, then end of file      \
    \                   \n    \t  raw.c_cc[VEOL] = 1;\n    \t  raw.c_cc[VEOF] = 2;\n\
    \    \t  tcsetattr(kfd, TCSANOW, &raw);\n    \t\tstruct pollfd pfd = {0,0,0};\
    \       /* poll() settings   */\n    \t  int pr;                            /*\
    \ poll() result     */\n    \t  \n    \t  pfd.fd = STDIN_FILENO;\n    \t  pfd.events\
    \ = POLLIN;\n    \t  pr = poll(&pfd, 1, 100);\n    \t  if(pr>0)\n    \t  {\n \
    \   \t\tif(read(kfd, &c, 1) < 0)\n    \t\t{\n    \t\t  perror(\"read():\");\n\
    \    \t\t  exit(-1);\n    \t\t}\n    \t\tdouble rad = 1;\n    \t\t\tswitch(c)\n\
    \    \t\t\t{\n    \t\t\t  case KEYCODE_L:\n    \t\t\t\tputs(\"LEFT\");\n    \t\
    \t\t\tthis->left_wheel_joint_->SetAngle(0,0);  //joint\n    \t\t\t\tbreak;\n \
    \   \t\t\t  case KEYCODE_D:\n    \t\t\t\tputs(\"DOWN\");\n    \t\t\t\tthis->left_wheel_joint_->SetAngle(0,-rad);\
    \ //joint\n    \t\t\t\tbreak;\n    \t\t\t}\n    \t   }\n    \t   else{\n    \t\
    \t//no key\n    \t\t}\n    }"
