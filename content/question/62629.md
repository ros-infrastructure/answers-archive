    \ tutorial.\n\n    #include <ros/ros.h>\n    #include \"roscopter/RC.h\"\n   \
    \ \n    int main(int argc, char **argv)\n    {\n        ros::init(argc, argv,\
    \ \"rc_sender\");\n        ros::NodeHandle n;\n        ros::Publisher rc_pub =\
    \ n.advertise<roscopter::RC>(\"send_rc\", 1000);\n        ros::Rate loop_rate(10);\n\
    \        int count = 0;\n        while (ros::ok())\n        {\n    \troscopter::RC\
    \ msg;\n    \trc_pub.publish(msg);\n    \tros::spinOnce();\n    \tloop_rate.sleep();\n\
    \    \t++count;\n        }\n        return 0;\n    }\n    \nMy *manifest.xml*\
    \ (or *package.xml* in Groovy) declares the roscopter package as a dependency.\
    \  Also, my *CMakeLists.txt* file is set to compile this file.  Running the produced\
    \ executable indeed publishes an empty `roscopter/RC` message at 10 Hz.\n"
