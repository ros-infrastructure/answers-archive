  body_markdown: " here is an example using MultiArray\n\n    #if (ARDUINO >= 100)\n\
    \     #include <Arduino.h>\n    #else\n     #include <WProgram.h>\n    #endif\n\
    \    \n    #include <Servo.h> \n    #include <ros.h>\n    #include \"std_msgs/MultiArrayLayout.h\"\
    \n    #include \"std_msgs/MultiArrayDimension.h\"\n    \n    #include \"std_msgs/UInt16MultiArray.h\"\
    \n    \n    ros::NodeHandle  nh;\n    \n    Servo servo1;\n    Servo servo2;\n\
    \    \n    void servo_cb( const std_msgs::UInt16MultiArray&  cmd_msg){\n     \
    \ servo1.write(cmd_msg.data[0]); //set servo angle, should be from 0-180  \n \
    \     servo2.write(cmd_msg.data[1]); \n      digitalWrite(13, HIGH-digitalRead(13));\
    \  //toggle led  \n    }\n    \n    \n    ros::Subscriber<std_msgs::UInt16MultiArray>\
    \ sub(\"servo\", servo_cb);\n    \n    void setup(){\n      pinMode(13, OUTPUT);\n\
    \    \n      nh.initNode();\n      nh.subscribe(sub);\n    \n      \n      servo1.attach(9);\
    \ //attach it to pin 9\n      servo2.attach(10);//attach it to pin10\n    }\n\
    \    \n    void loop(){\n      nh.spinOnce();\n      delay(1);\n    }\n\nPublisher:\n\
    \n     rostopic pub servo std_msgs/UInt16MultiArray '{data: [<angle1>, <angle2>]}'\n\
    \n"
