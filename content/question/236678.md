  body_markdown: " You can use [ros::master::check()](http://docs.ros.org/api/roscpp/html/namespaceros_1_1master.html#a6428e7e3468e4e9ab69b75f75d63510d).\n\
    \n    #include \"ros/ros.h\"\n    #include <iostream>\n    \n    int main(int\
    \ argc, char *argv[])\n    {\n    \tros::init(argc, argv, \"master_test\");\n\
    \    \tros::Time::init();\n    \twhile (!ros::master::check())\n    \t{\n    \t\
    \tstd::cout << \"waiting...\" << std::endl;\n    \t\tros::Duration(0.5).sleep();\n\
    \    \t}\n    \tstd::cout << \"master started!\" << std::endl;\n    \treturn 0;\n\
    \    }\n"
