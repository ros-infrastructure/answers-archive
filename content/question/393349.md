    \ velocity can be set higher than the physical robot. \n\nThe main function:\n\
    \n    void Turtlebot3Fake::commandVelocityCallback(const geometry_msgs::TwistConstPtr\
    \ cmd_vel_msg)\n    {\n      last_cmd_vel_time_ = ros::Time::now();\n    \n  \
    \    goal_linear_velocity_  = cmd_vel_msg->linear.x;\n      goal_angular_velocity_\
    \ = cmd_vel_msg->angular.z;\n    \n      wheel_speed_cmd_[LEFT]  = goal_linear_velocity_\
    \ - (goal_angular_velocity_ * wheel_seperation_ / 2);\n      wheel_speed_cmd_[RIGHT]\
    \ = goal_linear_velocity_ + (goal_angular_velocity_ * wheel_seperation_ / 2);\n\
    \    }\nhttps://github.com/ROBOTIS-GIT/turtlebot3_simulations/blob/master/turtlebot3_fake/src/turtlebot3_fake.cpp\n\
    \nDoesn't check for:\n\n    #define MAX_LINEAR_VELOCITY             0.22   //\
    \ m/s\n    #define MAX_ANGULAR_VELOCITY            2.84   // rad/s\n\nfound at:\
    \ https://github.com/ROBOTIS-GIT/turtlebot3_simulations/blob/master/turtlebot3_fake/include/turtlebot3_fake/turtlebot3_fake.h\n\
    \nTo simulate accurately, it's suggested to just a check for maximum velocity\
    \ like in the `teleop program`:\n\n    def checkLinearLimitVelocity(vel):\n  \
    \      if turtlebot3_model == \"burger\":\n          vel = constrain(vel, -BURGER_MAX_LIN_VEL,\
    \ BURGER_MAX_LIN_VEL)\n        elif turtlebot3_model == \"waffle\" or turtlebot3_model\
    \ == \"waffle_pi\":\n          vel = constrain(vel, -WAFFLE_MAX_LIN_VEL, WAFFLE_MAX_LIN_VEL)\n\
    \        else:\n          vel = constrain(vel, -BURGER_MAX_LIN_VEL, BURGER_MAX_LIN_VEL)\n\
    \    \n        return vel\n\n"
