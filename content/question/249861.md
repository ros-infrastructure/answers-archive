    \ is my code:\n<pre><code>\nimport android.util.Log;\n\nimport org.ros.android.view.visualization.Color;\n\
    import org.ros.android.view.visualization.VisualizationView;\nimport org.ros.android.view.visualization.layer.SubscriberLayer;\n\
    import org.ros.android.view.visualization.layer.TfLayer;\nimport org.ros.message.MessageListener;\n\
    import org.ros.namespace.GraphName;\nimport org.ros.node.ConnectedNode;\nimport\
    \ org.ros.rosjava_geometry.FrameTransform;\nimport org.ros.rosjava_geometry.Transform;\n\
    import org.ros.rosjava_geometry.Vector3;\n\nimport java.nio.ByteBuffer;\nimport\
    \ java.nio.ByteOrder;\nimport java.nio.FloatBuffer;\nimport java.util.ArrayList;\n\
    \nimport javax.microedition.khronos.opengles.GL10;\n\nimport nav_msgs.Odometry;\n\
    \n/**\n * Created by tony on 16-12-8.\n */\n\npublic class TrajectoryOdomLayer\
    \ extends SubscriberLayer<nav_msgs.Odometry> implements TfLayer {\n\n    private\
    \ static final Color COLOR = Color.fromHexAndAlpha(\"ff0000\", 1.0f);\n\n    private\
    \ static final float POINT_SIZE = 5.0f;\n\n    private FloatBuffer trajectoryBuffer;\n\
    \n    private float[] vertexs;\n\n    private ArrayList<Float> arrayList = new\
    \ ArrayList<Float>();\n\n    private boolean ready;\n\n    private GraphName frame;\n\
    \n    private float lastX = 0.0f;\n    private float lastY = 0.0f;\n    private\
    \ float lastZ = 0.0f;\n\n    public TrajectoryOdomLayer(String topic) {\n    \
    \    this(GraphName.of(topic));\n    }\n\n    public TrajectoryOdomLayer(GraphName\
    \ topic) {\n        super(topic, Odometry._TYPE);\n        ready = false;\n  \
    \  }\n\n    @Override\n    public GraphName getFrame() {\n        return frame;\n\
    \    }\n\n    @Override\n    public void draw(VisualizationView view, GL10 gl)\
    \ {\n        if (ready) {\n            gl.glEnableClientState(GL10.GL_VERTEX_ARRAY);\n\
    \            gl.glVertexPointer(3, GL10.GL_FLOAT, 0, trajectoryBuffer);\n    \
    \        COLOR.apply(gl);\n            gl.glPointSize(POINT_SIZE);\n         \
    \   gl.glDrawArrays(GL10.GL_POINTS, 0, trajectoryBuffer.limit() / 3);\n      \
    \      gl.glDisableClientState(GL10.GL_VERTEX_ARRAY);\n        }\n    }\n\n  \
    \  @Override\n    public void onStart(final VisualizationView view, final ConnectedNode\
    \ connectedNode) {\n        super.onStart(view, connectedNode);\n        frame\
    \ = GraphName.of(\"map\");\n        ByteBuffer goalVertexByteBuffer = ByteBuffer.allocateDirect(1024\
    \ * 1024);\n        goalVertexByteBuffer.order(ByteOrder.nativeOrder());\n   \
    \     trajectoryBuffer = goalVertexByteBuffer.asFloatBuffer();\n\n        getSubscriber().addMessageListener(new\
    \ MessageListener<nav_msgs.Odometry>() {\n            @Override\n            public\
    \ void onNewMessage(nav_msgs.Odometry message) {\n                Log.e(\"Trajectory\"\
    , \"onNewMessage\");\n                updateVertexBuffer(view, message);\n   \
    \             ready = true;\n            }\n        });\n    }\n\n    //\u8F6C\
    \u6362odom frame\u5230map frame\n    private void updateVertexBuffer(VisualizationView\
    \ view, nav_msgs.Odometry message) {\n        int count = message.getHeader().getSeq();\n\
    \        if (count % 5 == 0) {\n            FrameTransform frameTransform = view.getFrameTransformTree().transform(GraphName.of(message.getHeader().getFrameId()),\
    \ GraphName.of(\"map\"));\n            if (frameTransform != null) {\n       \
    \         Transform transform = Transform.fromPoseMessage(message.getPose().getPose());\n\
    \                Vector3 vector3 = frameTransform.getTransform().multiply(transform).getTranslation();\n\
    \                float x = (float) vector3.getX();\n                float y =\
    \ (float) vector3.getY();\n                float z = (float) vector3.getZ();\n\
    \                if (Math.sqrt(Math.pow(x - lastX, 2) + Math.pow(y - lastY, 2)\
    \ + Math.pow(z - lastZ, 2)) > 0.02) {\n                    arrayList.add(x);\n\
    \                    arrayList.add(y);\n                    arrayList.add(z);\n\
    \                    vertexs = new float[arrayList.size()];\n                \
    \    for (int i = 0; i < vertexs.length; i++) {\n                        vertexs[i]\
    \ = arrayList.get(i);\n                    }\n                    trajectoryBuffer.put(vertexs);\n\
    \                    lastX = x;\n                    lastY = y;\n            \
    \        lastZ = z;\n                }\n            }\n\n        }\n        trajectoryBuffer.position(0);\n\
    \    }\n}\n\n</code>\n</pre>"
