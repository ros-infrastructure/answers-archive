    \ the position of the upper-left corner of the image in rviz.\nsizeSq is a scaling\
    \ factor.\n\n\nvoid sendImageToRviz(std::string frame_id, int id, ros::Publisher*\
    \ marker_pub, cv::Mat src, double sizeSq, std::pair<double, double> LUcorner)\
    \ {\n\n\tvisualization_msgs::Marker image;\n\timage.header.frame_id = frame_id;\n\
    \timage.header.stamp = ros::Time::now();\n\timage.ns = \"image\";\n\timage.id\
    \ = id;\n\timage.action = visualization_msgs::Marker::ADD;\n\timage.type = visualization_msgs::Marker::TRIANGLE_LIST;\n\
    \timage.scale.x = 1;\n\timage.scale.y = 1;\n\timage.scale.z = 1;\n\n\tdouble pix\
    \ = sizeSq / src.rows;\n        geometry_msgs::Point p;\n\tstd_msgs::ColorRGBA\
    \ crgb;\n\n\tfor(int r = 0; r < src.rows; ++r) {\n\t\tfor(int c = 0; c < src.rows;\
    \ ++c) {\n\t\t  cv::Vec3b intensity = src.at<cv::Vec3b>(r, c);\n\t\t  crgb.r =\
    \ intensity.val[2] / 255.0;\n\t\t  crgb.g = intensity.val[1] / 255.0;\n\t\t  crgb.b\
    \ = intensity.val[0] / 255.0;\n\t\t  crgb.a = 1.0;\n\n\t\t  p.z = 0;\n\t\t  p.x\
    \ = LUcorner.first + r * pix;\n\t\t  p.y = LUcorner.second + c * pix;\n\t\t  image.points.push_back(p);\n\
    \t\t  image.colors.push_back(crgb);\n\t\t  p.x = LUcorner.first + (r + 1) * pix;\n\
    \t\t  p.y = LUcorner.second + c * pix;\n\t\t  image.points.push_back(p);\n\t\t\
    \  image.colors.push_back(crgb);\n\t\t  p.x = LUcorner.first + r * pix;\n\t\t\
