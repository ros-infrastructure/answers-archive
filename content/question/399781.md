    void imageCallback(const sensor_msgs::ImageConstPtr &msg)
    {
        cv::Mat inputIm;
        cv::Mat outputIm;
        std::vector<int> markerIds;
        std::vector<std::vector<cv::Point2f>> markerCorners, rejectedCandidates;
        cv::aruco::DetectorParameters parameters;
    
        try
        {
            std::cout<<"------------------------------------imageoCallback------------------------------------"<<std::endl;
            cv::imshow("Sub_Image",cv_bridge::toCvShare(msg, "bgr8")->image);
            cv::waitKey(30);
    
            inputIm = cv_bridge::toCvShare(msg,"bgr8")->image;
    
            while(true)
                {
                inputIm.copyTo(outputIm);
                cv::aruco::detectMarkers(inputIm, cv::aruco::getPredefinedDictionary(cv::aruco::DICT_4X4_250), markerCorners, markerIds);
                if(markerIds.size() > 0)
                    {
                    cv::aruco::drawDetectedMarkers(outputIm,markerCorners,markerIds);
                    cv::imshow("Detected Markers",outputIm);
                    char ch = cv::waitKey(10);
                    if(ch == 27)
                        break;
                    }
                }
            inputIm.release();
        }
        catch(cv_bridge::Exception e)
        {
            ROS_ERROR("COULDN'T CONVERT!");
        }
    }
    void camInfoCallback(const sensor_msgs::CameraInfoConstPtr &msg)
    {
        cv::Mat camMatrix;
        cv::Mat_<double> distCoeff;
        image_geometry::PinholeCameraModel cam_model;
    
        try
        {
            cam_model.fromCameraInfo(msg);
            camMatrix = cv::Mat(cam_model.fullIntrinsicMatrix());
            distCoeff = cv::Mat(cam_model.distortionCoeffs());
            std::cout<<"---------camera matrix----------"<<camMatrix<<std::endl;
            std::cout<<"---------distortion matrix---------"<<distCoeff<<std::endl;
        }
        catch(cv_bridge::Exception e)
        {
            ROS_ERROR("cv_bridge exception %s",e.what());
            return;
        }
    }
    
    int main(int argc, char **argv)
    {   
        ros::init(argc,argv, "marker_detection");

        ros::Subscriber cam_info_sub = nh.subscribe("/camera/camera_info",1,camInfoCallback);
        ros::spin();
        cv::destroyWindow("Sub_Image");
    
        return 0;
