    \ std_msgs::EmptyConstPtr /* msg */)\n    {\n      ROS_INFO_STREAM(\"Kobuki :\
    \ Resetting the odometry. [\" << name << \"].\");\n      joint_states.position[0]\
    \ = 0.0; // wheel_left\n      joint_states.velocity[0] = 0.0;\n      joint_states.position[1]\
    \ = 0.0; // wheel_right\n      joint_states.velocity[1] = 0.0;\n      odometry.resetOdometry();\n\
    \      kobuki.resetOdometry();\n      return;\n    } \n\nHope this helps."
