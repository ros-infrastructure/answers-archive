          vec.data.push_back(temp);
    
      publisher.publish(vec);

      for(;itr != vec1->data.end();++itr)

          ROS_INFO("%d ",*itr);
        }

    ...
    ros::Subscriber sub_1 = nh.subscribe("Topic_1",1000,callback_1);
    ...

There are no errors when I compile these nodes.
