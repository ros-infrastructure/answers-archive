    \n    # Check what is going to be removed \n\n    dpkg -l | grep '^ii.*\\(libignition\\\

    \ -l | grep '^ii.*\\(libignition\\|sdformat\\|gazebo\\).*~focal' | awk '{ print\
    \ $2 };') ]]; do sudo dpkg -r $(dpkg -l | grep '^ii.*\\(libignition\\|sdformat\\\
    |gazebo\\).*~focal' | awk '{ print $2 }';); done\n\n    # Install new Gazebo on\
