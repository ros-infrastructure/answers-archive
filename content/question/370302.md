    \ \n\nEDIT:\n\nYou'll need also to spin briefly to let the discovery happen and\
    \ see the topics. For example the code could look like\n\n    rclcpp::Rate r(100);\n\
    \    int i = 0;\n    while (rclcpp::ok() && i < 100) {\n      rclcpp::spin_some(this->get_node_base_interface());\n\
    \      r.sleep();\n      i++;\n    }\n    // Now you can call the function to\
    \ get topics\n    std::map<std::string, std::vector<std::string>> topic_infos\
    \ = this->get_topic_names_and_types();\n    for (const auto& topic_it : topic_infos)\
    \ {\n      std::string topic_name = topic_it.first;\n      std::vector<std::string>\
    \ topic_types = topic_it.second;\n      // iterate over all topic types\n    \
    \  for (const auto& topic_type : topic_types) \n        RCLCPP_INFO(this->get_logger(),\
    \ \"Topic: %s, type: %s\", topic_name.c_str(), topic_type.c_str());\n    }"
