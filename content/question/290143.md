    \ same time, in one bag.\n([Ref 1](http://wiki.ros.org/rosbag/Code%20API#Python_API),\
    \ [Ref 2](http://wiki.ros.org/rosbag/Cookbook))\n    \n    import rosbag\n   \
    \ import rospy\n    from sensor_msgs.msg import LaserScan\n    from math import\
    \ floor\n\n    # Bag Names\n    old_bag_name = 'old_bag.bag'        \n    new_bag_name\
    \ = 'new_bag.bag'\n\n    # List of topic names in the bag(have one topic that\
    \ is of type LaserScan)\n    topics = ['scan_topic']\n\n    # List of New Topic\
    \ Names\n    new_topics = ['new_scan_topic_1', 'new_scan_topic_2']     \n\n  \
    \  # Open Bags for writing and reading\n    old_bag = rosbag.Bag(old_bag_name)\n\
    \    new_bag = rosbag.Bag(new_bag_name, 'w')\n\n    # Array to store Messages\
    \ in chronological order\n    scan_array = []\n\n    # Read Bag and append messages\
    \ from topics accordingly\n    for topic, msg, t in old_bag.read_messages(topics=topics):\n\
    \        # Based on the topic name, read into an array(has only one topic right\
    \ now)\n        if(topic == \"scan_topic\"):\n            scan_array.append(msg)\n\
    \    bag.close()\n\n\n    # Write message into new bag for each topic, while updating\
    \ the time stamp\n    t = 0\n    half_time = floor(len(scan_array/2))\n    while\
    \ t < half_time:\n\n        # Update Time for required messages\n        current_time\
    \ = rospy.Time.now()\n        scan_array[t].header.stamp = current_time\n    \
    \    scan_array[time_offest+t].header.stamp = current_time\n\n        # Write\
    \ messages onto bag for the current_time\n        new_bag.write(new_topics[0],\
    \ scan_array[t], current_time)\n        new_bag.write(new_topics[1], scan_array[time_offest+t],\
    \ current_time)\n        t = t + 1"
