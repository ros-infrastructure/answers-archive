  body_markdown: "I don't know of a native way to do this, but I wrote the following\
    \ conversion functions a while ago that I've been using. A call to `msg_to_se3`\
    \ can return a 4x4 SE(3) matrix for a few different ROS message types.\n\n   \
    \ import tf.transformations as tr\n    \n    from geometry_msgs.msg import Point\n\
    \    from geometry_msgs.msg import Pose\n    from geometry_msgs.msg import PoseStamped\n\
    \    from geometry_msgs.msg import Quaternion\n    from geometry_msgs.msg import\
    \ Transform\n    from geometry_msgs.msg import TransformStamped\n    from geometry_msgs.msg\
    \ import Vector3\n    \n    import numpy as np\n\n    def pose_to_pq(msg):\n \
    \       \"\"\"Convert a C{geometry_msgs/Pose} into position/quaternion np arrays\n\
    \    \n        @param msg: ROS message to be converted\n        @return:\n   \
    \       - p: position as a np.array\n          - q: quaternion as a numpy array\
    \ (order = [x,y,z,w])\n        \"\"\"\n        p = np.array([msg.position.x, msg.position.y,\
    \ msg.position.z])\n        q = np.array([msg.orientation.x, msg.orientation.y,\n\
    \                      msg.orientation.z, msg.orientation.w])\n        return\
    \ p, q\n    \n    \n    def pose_stamped_to_pq(msg):\n        \"\"\"Convert a\
    \ C{geometry_msgs/PoseStamped} into position/quaternion np arrays\n    \n    \
    \    @param msg: ROS message to be converted\n        @return:\n          - p:\
    \ position as a np.array\n          - q: quaternion as a numpy array (order =\
    \ [x,y,z,w])\n        \"\"\"\n        return pose_to_pq(msg.pose)\n    \n    \n\
    \    def transform_to_pq(msg):\n        \"\"\"Convert a C{geometry_msgs/Transform}\
    \ into position/quaternion np arrays\n    \n        @param msg: ROS message to\
    \ be converted\n        @return:\n          - p: position as a np.array\n    \
    \      - q: quaternion as a numpy array (order = [x,y,z,w])\n        \"\"\"\n\
    \        p = np.array([msg.translation.x, msg.translation.y, msg.translation.z])\n\
    \        q = np.array([msg.rotation.x, msg.rotation.y,\n                     \
    \ msg.rotation.z, msg.rotation.w])\n        return p, q\n    \n    \n    def transform_stamped_to_pq(msg):\n\
    \        \"\"\"Convert a C{geometry_msgs/TransformStamped} into position/quaternion\
    \ np arrays\n    \n        @param msg: ROS message to be converted\n        @return:\n\
    \          - p: position as a np.array\n          - q: quaternion as a numpy array\
    \ (order = [x,y,z,w])\n        \"\"\"\n        return transform_to_pq(msg.transform)\n\
    \    \n    \n    def msg_to_se3(msg):\n        \"\"\"Conversion from geometric\
    \ ROS messages into SE(3)\n    \n        @param msg: Message to transform. Acceptable\
    \ types - C{geometry_msgs/Pose}, C{geometry_msgs/PoseStamped},\n        C{geometry_msgs/Transform},\
    \ or C{geometry_msgs/TransformStamped}\n        @return: a 4x4 SE(3) matrix as\
    \ a numpy array\n        @note: Throws TypeError if we receive an incorrect type.\n\
    \        \"\"\"\n        if isinstance(msg, Pose):\n            p, q = pose_to_pq(msg)\n\
    \        elif isinstance(msg, PoseStamped):\n            p, q = pose_stamped_to_pq(msg)\n\
    \        elif isinstance(msg, Transform):\n            p, q = transform_to_pq(msg)\n\
    \        elif isinstance(msg, TransformStamped):\n            p, q = transform_stamped_to_pq(msg)\n\
    \        else:\n            raise TypeError(\"Invalid type for conversion to SE(3)\"\
    )\n        norm = np.linalg.norm(q)\n        if np.abs(norm - 1.0) > 1e-3:\n \
    \           raise ValueError(\n                \"Received un-normalized quaternion\
    \ (q = {0:s} ||q|| = {1:3.6f})\".format(\n                    str(q), np.linalg.norm(q)))\n\
    \        elif np.abs(norm - 1.0) > 1e-6:\n            q = q / norm\n        g\
    \ = tr.quaternion_matrix(q)\n        g[0:3, -1] = p\n        return g"
