  body_markdown: "With this code, you can get the name of all the running services\
    \ and copy them to an array.\n\n    ros::init(argc, argv, \"node\");\n    \tros::NodeHandle\
    \ n;\n    \tXmlRpc::XmlRpcValue req = \"/node\";\n    \tXmlRpc::XmlRpcValue res;\n\
    \    \tXmlRpc::XmlRpcValue pay;\n    \n    \tif (ros::ok() == true)\n    \t{\n\
    \    ros::master::execute(\"getSystemState\",req ,res ,pay ,true);\n    \tstd::string\
    \ state[res.size()];\n    \t  for(int x=0 ; x<res[2][2].size() ; x++)\n    \t\
    \  {\n    \t\t  std::string gh = res[2][2][x][0].toXml().c_str();\n    \t\t  gh.erase(gh.begin(),\
    \ gh.begin()+7);\n    \t\t  gh.erase(gh.end()-8, gh.end());\n    \t\t  state[x]\
    \ = gh;\n    \t  }\n    }"
