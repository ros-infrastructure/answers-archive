    \ to get you started:\n\n    #!/usr/bin/env python\n\n    import roslib; roslib.load_manifest('example')\n\
    \    import rospy\n\n    from std_msgs.msg import String\n\n    from PySide.QtCore\
    \ import * \n    from PySide.QtGui import * \n    import sys\n\n    class GUI:\n\
    \        def __init__(self): \n            # A publisher for messages\n      \
    \      self.pub = rospy.Publisher('example', String)\n\n            # A subscriber\
    \ for our own messages\n            self.sub = rospy.Subscriber('example', String,\
    \ self.sub_callback)\n\n            self.button = QPushButton('push\\nme')\n \
    \           self.button.clicked.connect(self.pub_callback)\n            self.button.show()\n\
    \n        def pub_callback(self):\n            self.pub.publish('Thank You!')\n\
    \        \n        def sub_callback(self, msg):\n            print 'Got message:',\
    \ msg.data\n        \n\n    if __name__ == '__main__':\n        # Initialize the\
    \ node\n        rospy.init_node('example')\n\n        # Initialize Qt\n      \
    \  app = QApplication(sys.argv)\n\n        gui = GUI()\n\n        app.exec_()\n\
    \nYou'll also need the following lines in your manifest.xml\n\n    <depend package=\"\
