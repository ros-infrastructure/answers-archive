  body_markdown: "The final friction is a result of the friction parameters for both\
    \ bodies in contact. In your case, you should also change the friction for the\
    \ ground plane. \n\nAlso, note that bodies moving very slowly are \"automatically\
    \ disabled\" on ODE. You can turn off that behavior setting the `<allow_auto_disable>`\
    \ flag to false. Or you can use a different physics engine, such as Bullet.\n\n\
    Try this world for example:\n\n    <?xml version=\"1.0\" ?>\n    <sdf version=\"\
    1.5\">\n      <world name=\"default\">\n        <physics type=\"ode\">\n     \
    \     <gravity>-0.1 0 -9.8</gravity>\n        </physics>\n        <!-- A global\
    \ light source -->\n        <include>\n          <uri>model://sun</uri>\n    \
    \    </include>\n        <!-- A ground plane -->\n        <model name=\"ground_plane\"\
    >\n          <static>true</static>\n          <link name=\"link\">\n         \
    \   <collision name=\"collision\">\n              <geometry>\n               \
    \ <plane>\n                  <normal>0 0 1</normal>\n                  <size>100\
    \ 100</size>\n                </plane>\n              </geometry>\n          \
    \    <surface>\n                <friction>\n                  <ode>\n        \
    \            <mu>0</mu>\n                    <mu2>0</mu2>\n                  </ode>\n\
    \                </friction>\n              </surface>\n            </collision>\n\
    \            <visual name=\"visual\">\n              <geometry>\n            \
    \    <plane>\n                  <normal>0 0 1</normal>\n                  <size>100\
    \ 100</size>\n                </plane>\n              </geometry>\n          \
    \  </visual>\n          </link>\n        </model>\n        <!-- box -->\n    \
    \    <model name=\"box\">\n          <allow_auto_disable>false</allow_auto_disable>\n\
    \          <pose>0 0 0.5 0 0 0</pose>\n          <link name=\"link\">\n      \
    \      <collision name=\"collision\">\n              <geometry>\n            \
    \    <box>\n                  <size>1 1 1</size>\n                </box>\n   \
    \           </geometry>\n              <surface>\n                <friction>\n\
    \                  <ode>\n                    <mu>0</mu>\n                   \
    \ <mu2>0</mu2>\n                  </ode>\n                </friction>\n      \
    \        </surface>\n            </collision>\n            <visual name=\"visual\"\
    >\n              <geometry>\n                <box>\n                  <size>1\
    \ 1 1</size>\n                </box>\n              </geometry>\n            </visual>\n\
    \          </link>\n        </model>\n      </world>\n    </sdf>"
