  body_markdown: "An alternative to writing   \na Gazebo Plugin that publishes force/torque\
    \ data to a ROS Topic \nis to write   \na ROS Node that listens to the Gazebo\
    \ Transport layer and republishes force/torque data to a ROS Topic.\n\nUses the\
    \ [Gazebo Force/Torque Sensor](http://gazebosim.org/tutorials?tut=force_torque_sensor&cat=sensors).\n\
    \n@sloretz's answer is probably better. I did it this way - in my case - because\
    \ I wanted to access some options that were not available in the gazebo_ros force/torque\
    \ plugin. \n\nExample:\n\n----------\nmy_robot.urdf\n\n    ...\n      <gazebo\
    \ reference=\"joint_name\">\n        <sensor name=\"joint_name_force_torque\"\
    \ type=\"force_torque\">\n          <always_on>true</always_on>\n          <update_rate>20.0</update_rate>\n\
    \          <visualize>true</visualize>\n          <force_torque>\n           \
    \ <frame>sensor</frame>\n            <measure_direction>child_to_parent</measure_direction>\n\
    \          </force_torque>\n        </sensor>\n      </gazebo>\n    ...\n\ngazebo_transport_to_ros_topic.cpp\
    \ (place in src directory of ROS package)\n\n    /*\n    *  Name: gazebo_transport_to_ros_topic.cpp\n\
    \    *  Author: Joseph Coombe\n    *  Date: 11/22/2017\n    *  Edited: 11/27/2017\n\
    \    *  Description:\n    *   Subscribe to a Gazebo transport topic and publish\
    \ to a ROS topic\n    */\n    \n    // Gazebo dependencies\n    #include <gazebo/transport/transport.hh>\n\
    \    #include <gazebo/msgs/msgs.hh>\n    #include <gazebo/gazebo.hh>\n    #include\
    \ <gazebo/gazebo_client.hh>\n    \n    // ROS dependencies\n    #include <ros/ros.h>\n\
    \    #include <geometry_msgs/WrenchStamped.h>\n    \n    #include <iostream>\n\
    \    \n    ros::Publisher pub;\n    \n    void torquesCb(const ConstWrenchStampedPtr\
    \ &_msg)\n    {\n      std::cout << \"Received msg: \" << std::endl;\n      std::cout\
    \ << _msg->DebugString() << std::endl;\n      geometry_msgs::WrenchStamped msgWrenchedStamped;\n\
    \      // try WrenchStamped msgWrenchedStamped;\n      msgWrenchedStamped.header.stamp\
    \ = ros::Time::now();\n      msgWrenchedStamped.wrench.force.x = _msg->wrench().force().x();\n\
    \      msgWrenchedStamped.wrench.force.y = _msg->wrench().force().y();\n     \
    \ msgWrenchedStamped.wrench.force.z = _msg->wrench().force().z();\n      msgWrenchedStamped.wrench.torque.x\
    \ = _msg->wrench().torque().x();\n      msgWrenchedStamped.wrench.torque.y = _msg->wrench().torque().y();\n\
    \      msgWrenchedStamped.wrench.torque.z = _msg->wrench().torque().z();\n   \
    \   pub.publish(msgWrenchedStamped);\n    }\n    \n    int main(int argc, char\
    \ **argv)\n    {\n      ROS_INFO(\"Starting gazebo_transport_to_ros_topic node\"\
    );\n    \n      // Load Gazebo\n      gazebo::client::setup(argc, argv);\n   \
    \ \n      // Load ROS\n      ros::init(argc, argv, \"gazebo_transport_to_ros_topic\"\
    );\n    \n      // Create Gazebo node and init\n      gazebo::transport::NodePtr\
    \ node(new gazebo::transport::Node());\n      node->Init();\n    \n      // Create\
    \ ROS node and init\n      ros::NodeHandle n;\n    \n      // Get ROS params\n\
    \      std::string gazebo_transport_topic_to_sub;\n      std::string ros_topic_to_pub;\n\
    \      double ros_rate;\n      ros::param::get(\"~gazebo_transport_topic_to_sub\"\
    , gazebo_transport_topic_to_sub);\n      ros::param::get(\"~ros_rate\", ros_rate);\n\
    \    \n      pub = n.advertise<geometry_msgs::WrenchStamped>(ros_topic_to_pub,\
    \ 100);\n    \n      // Listen to Gazebo force_torque sensor topic\n      gazebo::transport::SubscriberPtr\
    \ sub = node->Subscribe(gazebo_transport_topic_to_sub, torquesCb);\n      ros::Rate\
    \ loop_rate(ros_rate); // 100 hz\n      while(ros::ok())\n      {\n        ros::spinOnce();\n\
    \        loop_rate.sleep();\n      }\n      gazebo::shutdown();\n      return\
    \ 0;\n    }\n\nMake sure to add the following sections to your CMakeLists.txt\
    \ (sorry this is a bit messy):\n\n    ...\n    ## Compile as C++11, supported\
    \ in ROS Kinetic and newer\n    add_compile_options(-std=c++11)\n    # Find Gazebo\n\
    \    find_package(gazebo REQUIRED)\n    \n    set(CMAKE_CXX_FLAGS \"-std=c++11\
    \ ${CMAKE_CXX_FLAGS}\")\n    \n    link_directories(${GAZEBO_LIBRARY_DIRS})\n\
    \    include_directories(include)\n    include_directories( ${catkin_INCLUDE_DIRS}\n\
    \                         ${Boost_INCLUDE_DIR}\n                         ${GAZEBO_INCLUDE_DIRS}\n\
    \    )\n    \n    target_link_libraries(\n      ${PROJECT_NAME}_gazebo_transport_to_ros_topic\n\
    \      ${catkin_LIBRARIES}\n      ${GAZEBO_LIBRARIES}\n    )\n    ...\n\nmy_robot.launch\n\
    \n    ...\n      <node name=\"joint_name\" pkg=\"my_pkg_gazebo\" type=\"gazebo_transport_to_ros_topic\"\
    \n            output=\"screen\" launch-prefix=\"xterm -e\" >\n        <param name=\"\
    gazebo_transport_topic_to_sub\" type=\"str\" value=\"/gazebo/default/robot_name/joint_name/joint_name_force_torque/wrench\"\
    \ />\n        <param name=\"ros_topic_to_pub\" type=\"str\" value=\"force_torque/robot_name/joint_name_force_torque\"\
    \ />\n        <param name=\"ros_rate\" type=\"double\" value=\"100.0\" />\n  \
    \    </node>\n    ..."
