    int main(int argc, char** argv)
{
    ros::init(argc,argv,"registration");
    ros::NodeHandle node;
    registration reg;

    sensor_msgs::CameraInfo camInt;

            // reading the camera intrinsic parametrs from local directory

                    LIST1.push_back(v1);
