    \      const ros::M_string& header = event.getConnectionHeader();\n      ros::Time\
    \ receipt_time = event.getReceiptTime();\n      \n      const sensor_msgs::PointCloud2&\
    \ input = event.getMessage();\n\n      ROS_INFO(\"Got a cloud on topic %s in frame\

    \ \n               input.width*input.height);\n    }"

      sub_cloud_ = nh_.subscribe<sensor_msgs::PointCloud2>

                    boost::bind(&HapticNode::cloudCallback, this, _1, _2));
    
      void HapticNode::cloudCallback(const sensor_msgs::PointCloud2ConstPtr &input,
