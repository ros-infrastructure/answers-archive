    \ the opensource_tracking.launch file looks like this:\n\n    <launch>\n    <arg\
    \ name=\"offline\"          default=\"false\"/>\n    <include unless=\"$(arg offline)\"\
    \ \n        file=\"$(find realsense2_camera)/launch/rs_camera.launch\">\n    \
    \    <arg name=\"align_depth\" value=\"true\"/>\n        <arg name=\"linear_accel_cov\"\
    \ value=\"1.0\"/>\n        <arg name=\"enable_gyro\" value=\"true\"/>\n      \
    \  <arg name=\"enable_accel\" value=\"true\"/>\n        <arg name=\"unite_imu_method\"\
    \ value=\"linear_interpolation\"/>\n        <arg name=\"enable_infra1\" value=\"\
    true\"/>\n        <arg name=\"enable_infra2\" value=\"true\"/>\n    </include>\n\
    \    \n    <node pkg=\"imu_filter_madgwick\" type=\"imu_filter_node\" name=\"\
    ImuFilter\">\n        <param name=\"use_mag\" type=\"bool\" value=\"false\" />\n\
