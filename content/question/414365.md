  body_markdown: "Yes it is possible. The most important thing to remember is namespaces.\
    \ [Check out Multiple robots simulation and navigation](https://answers.ros.org/question/41433/multiple-robots-simulation-and-navigation/).\n\
    \nHere is an example launch file with 3 turtlebot3 robots in gazebo. You will\
    \ have to visualize your RViz topics.\n\n    <launch>\n      <!-- <param name=\"\
    /use_sim_time\" value=\"true\"/> -->\n      <arg name=\"model\" default=\"$(env\
    \ TURTLEBOT3_MODEL)\" doc=\"model type [burger, waffle, waffle_pi]\"/>\n     \
    \ <arg name=\"first_tb3\"  default=\"tb3_0\"/>\n      <arg name=\"second_tb3\"\
    \ default=\"tb3_1\"/>\n      <arg name=\"third_tb3\"  default=\"tb3_2\"/>  \n\
    \    \n      <!-- 3 in the same room: -->\n      <arg name=\"first_tb3_x_pos\"\
    \ default=\"3.0\"/>\n      <arg name=\"first_tb3_y_pos\" default=\"4.0\"/>\n \
    \     <arg name=\"first_tb3_z_pos\" default=\" 0.0\"/>\n      <arg name=\"first_tb3_yaw\"\
    \   default=\" 0.0\"/>\n    \n      <arg name=\"second_tb3_x_pos\" default=\"\
    \ 3.0\"/>\n      <arg name=\"second_tb3_y_pos\" default=\"1.0\"/>\n      <arg\
    \ name=\"second_tb3_z_pos\" default=\" 0.0\"/>\n      <arg name=\"second_tb3_yaw\"\
    \   default=\" 0.0\"/>\n    \n      <arg name=\"third_tb3_x_pos\" default=\" 3.0\"\
    />\n      <arg name=\"third_tb3_y_pos\" default=\" 3.0\"/>\n      <arg name=\"\
    third_tb3_z_pos\" default=\" 0.0\"/>\n      <arg name=\"third_tb3_yaw\"   default=\"\
    \ 0.0\"/>\n    \n    \n      <include file=\"$(find gazebo_ros)/launch/empty_world.launch\"\
    >\n        <arg name=\"world_name\" value=\"$(find turtlebot3_gazebo)/worlds/turtlebot3_house.world\"\
    /> \n        <arg name=\"paused\" value=\"false\"/>\n        <arg name=\"use_sim_time\"\
    \ value=\"true\"/>\n        <!-- <arg name=\"gui\" value=\"false\"/> -->\n   \
    \     <arg name=\"gui\" value=\"true\"/>\n        <arg name=\"headless\" value=\"\
    false\"/>\n        <arg name=\"debug\" value=\"false\"/>\n      </include>  \n\
    \    \n      <group ns = \"$(arg first_tb3)\">\n        <param name=\"robot_description\"\
    \ command=\"$(find xacro)/xacro $(find turtlebot3_description)/urdf/turtlebot3_$(arg\
    \ model).urdf.xacro\" />\n    \n        <node pkg=\"robot_state_publisher\" type=\"\
    robot_state_publisher\" name=\"robot_state_publisher\" output=\"screen\">\n  \
    \        <param name=\"publish_frequency\" type=\"double\" value=\"50.0\" />\n\
    \          <param name=\"tf_prefix\" value=\"$(arg first_tb3)\" />\n        </node>\n\
    \        \n        <node name=\"spawn_urdf\" pkg=\"gazebo_ros\" type=\"spawn_model\"\
    \ args=\"-urdf -model $(arg first_tb3) -x $(arg first_tb3_x_pos) -y $(arg first_tb3_y_pos)\
    \ -z $(arg first_tb3_z_pos) -Y $(arg first_tb3_yaw) -param robot_description\"\
    \ />\n      </group>\n    \n      <group ns = \"$(arg second_tb3)\">\n       \
    \ <param name=\"robot_description\" command=\"$(find xacro)/xacro $(find turtlebot3_description)/urdf/turtlebot3_$(arg\
    \ model).urdf.xacro\" />\n    \n        <node pkg=\"robot_state_publisher\" type=\"\
    robot_state_publisher\" name=\"robot_state_publisher\" output=\"screen\">\n  \
    \        <param name=\"publish_frequency\" type=\"double\" value=\"50.0\" />\n\
    \          <param name=\"tf_prefix\" value=\"$(arg second_tb3)\" />\n        </node>\n\
    \    \n        <node name=\"spawn_urdf\" pkg=\"gazebo_ros\" type=\"spawn_model\"\
    \ args=\"-urdf -model $(arg second_tb3) -x $(arg second_tb3_x_pos) -y $(arg second_tb3_y_pos)\
    \ -z $(arg second_tb3_z_pos) -Y $(arg second_tb3_yaw) -param robot_description\"\
    \ />\n      </group>\n    \n      <group ns = \"$(arg third_tb3)\">\n        <param\
    \ name=\"robot_description\" command=\"$(find xacro)/xacro $(find turtlebot3_description)/urdf/turtlebot3_$(arg\
    \ model).urdf.xacro\" />\n    \n        <node pkg=\"robot_state_publisher\" type=\"\
    robot_state_publisher\" name=\"robot_state_publisher\" output=\"screen\">\n  \
    \        <param name=\"publish_frequency\" type=\"double\" value=\"50.0\" />\n\
    \          <param name=\"tf_prefix\" value=\"$(arg third_tb3)\" />\n        </node>\n\
    \    \n        <node name=\"spawn_urdf\" pkg=\"gazebo_ros\" type=\"spawn_model\"\
    \ args=\"-urdf -model $(arg third_tb3) -x $(arg third_tb3_x_pos) -y $(arg third_tb3_y_pos)\
    \ -z $(arg third_tb3_z_pos) -Y $(arg third_tb3_yaw) -param robot_description\"\
    \ />\n      </group>\n    \n    </launch>\n\n"
