  body_markdown: "Hi, \n\nThis small (and untested) example subscribes on a topic\
    \ and return success if 2 is received on this topic in the next 30 seconds.\n\n\
    \    #!/usr/bin/env python\n    import roslib; roslib.load_manifest('smach_example')\n\
    \    import rospy\n    import smach\n    import smach_ros\n    from std_msgs.msg\
    \ import Int16\n    import time\n    import threading\n        \n    class WaitForTwo(smach.State):\n\
    \        def __init__(self):\n            smach.State.__init__(self, outcomes=['success',\
    \ 'in_progress', 'failed'])\n    \n            self.mutex = threading.Lock()\n\
    \            self.two_received = False\n    \n            self.subscriber = rospy.Subscriber('/test',\
    \ Int16, self.callback)\n    \n        def callback(self, data):\n           \
    \ self.mutex.acquire()\n            if data.data == 2:\n                self.two_received\
    \ = True\n            self.mutex.release()\n    \n        def execute(self):\n\
    \            #wait for a maximum of 30 seconds \n            for i in range(0,\
    \ 300):\n                self.mutex.acquire()\n                if self.two_received:\n\
    \                    #ok we received 2\n                    return 'success'\n\
    \                \n                self.mutex.release()\n    \n              \
    \  time.sleep(.1)\n                #still waiting\n                return 'in_progress'\n\
    \            #we didn't get 2 in the 30 sec\n            return 'failed'\n"
