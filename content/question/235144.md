    {

    }
    
    void thread1(ros::NodeHandle &n)
    {
    
    	ros::Subscriber sub1 = n.subscribe("msg1", 1000, chatterCallback);
    }

    {
    
    	ros::Subscriber sub2 = n.subscribe("msg2", 1000, chatterCallback);

     get_speedData_thread.join();	
     get_encoderData_thread.join();
     
     return 0;
    }
