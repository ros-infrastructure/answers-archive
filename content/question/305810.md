    \ have attached part of my codes for PC2 and Marker at here as references for\
    \ beginner like me.\n\nFor PointCloud2 :\n\n    # PointCloud2 is working\n   \
    \ # PointCloud2 : The following part is to gdm_data X, Y and mean/var\n    gdm_X\
    \ = X.flatten()\n    gdm_Y = Y.flatten()\n    gdm_Z = numpy.zeros_like(gdm_X)\n\
    \    gdm_Mean = mean_map.flatten()\n    gdm_data = numpy.column_stack((gdm_X,gdm_Y,gdm_Z,gdm_Mean))\n\
    \    # PointCloud2 : The following algo is for visualize pointcloud2\n    header\
    \ = std_msgs.msg.Header()\n    header.stamp = rospy.Time.now()\n    header.frame_id\
    \ = 'map'\n    field=[PointField(name='x', offset=0, datatype=PointField.FLOAT32,\
    \ count=1),PointField(name='y', offset=4, datatype=PointField.FLOAT32, count=1),PointField(name='z',\
    \ offset=8, datatype=PointField.FLOAT32, count=1),PointField(name='rgb', offset=12,\
    \ datatype=PointField.UINT32, count=1)]\n    gdm_data = pcl2.create_cloud(header,\
    \ field, gdm_data)\n    pc2_Testing.publish(gdm_data)\n\n For Marker :\n\n   \
    \ # MarkerArray is Working\n    gdm_X = X.flatten()\n    gdm_Y = Y.flatten()\n\
    \    gdm_Z = numpy.zeros_like(gdm_X)\n    gdm_Mean = mean_map.flatten()\n    gdm_Mean\
    \ = numpy.interp(gdm_Mean,[min(gdm_Mean),max(gdm_Mean)],[0.0,1.0])\n    gdm_data\
    \ = numpy.column_stack((gdm_X,gdm_Y,gdm_Z,gdm_Mean))\n    triplePoints = []\n\
    \    colorsGroup = []\n    markerArray = MarkerArray()\n    for (x,y,z,a) in gdm_data:\n\
    \    \tp = Point() \n    \tp.x = x\n    \tp.y = y\n    \tp.z = 0\n    \tmarker\
    \ = Marker()\n    \tmarker.header.frame_id = \"/map\"\n    \tmarker.type = marker.POINTS\n\
    \    \tmarker.action = marker.ADD\n    \tmarker.scale.x = 0.2\n    \tmarker.scale.y\
    \ = 0.2\n    \tmarker.scale.z = 0.2\n    \tcolorsGroup.append(ColorRGBA(0,0,a,1))\n\
    \    \tmarker.pose.orientation.w = 1.0\n    \tt = rospy.Duration()\n    \tmarker.lifetime\
    \ = t\n    \ttriplePoints.append(p)\n    marker.points = triplePoints\n    marker.colors\
    \ = colorsGroup\n    markerArray.markers.append(marker)\n    marker_pub.publish(markerArray)"
