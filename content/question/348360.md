I am building a differential drive robot with encoders and a RPLidar on-board. I am getting the wheel encoder data and publishing it to /odom and /tf using this python script below. All the codes compile ok and I can view the /tf, /scan on rviz. When I run gmapping, it creates first iteration of the map. But as i drive the robot, the laser scan moves with it in rviz, but the map does not get added. I am using ROS Kinetic.

Python code to get wheels encoder data and do /tf broadcast

    import math
	from math import sin, cos, pi
	import sys
	#import serial
	#import time
	import rospy
	import tf
	from nav_msgs.msg import Odometry
	from geometry_msgs.msg import Point, Pose, Quaternion, Twist, Vector3

	rospy.init_node('pub_odom')

	odom_pub = rospy.Publisher("odom", Odometry, queue_size=50)
	odom_broadcaster = tf.TransformBroadcaster()

	x = 0.0
	y = 0.0
	th = 0.0

	vx = 0.0
	vy = 0
	vth = 0.0
	_PreviousLeftEncoderCounts = 0
	_PreviousRightEncoderCounts = 0

	DistancePerCount = 0.00133  #24 tick per rev, 12.3 ratio| 24x12.3=295.2 ticks for Pix0.125=0.3927m distance 
	lengthBetweenTwoWheels = 0.4


	current_time = rospy.Time.now()
	last_time = rospy.Time.now()

	r = rospy.Rate(20.0)

	while not rospy.is_shutdown():
	    
	    with open("/home/omar/catkin_ws/src/read_position/scripts/publish_odom.txt", "r") as f: 
			content = f.readlines()  #encoder values are stored in this txt file
	    
	    
	    try:
		    data_odom = content[0].split(",")
	    	tick_x = int(data_odom [1]) #left encoder value
	    	tick_y = int (data_odom [0]) #right encoder value
	    	
	    	deltaLeft = tick_x - int(_PreviousLeftEncoderCounts) #delta distance
	    	deltaRight = tick_y - _PreviousRightEncoderCounts
	    	current_time = rospy.Time.now()
	    	dt = (current_time - last_time).to_sec()	

	    	omega_left = (deltaLeft * DistancePerCount) / dt #left speed
	    	omega_right = (deltaRight * DistancePerCount) / dt #right speed

	    	vx = (omega_left + omega_right) / 2 #average speed
	    	vth = (omega_left - omega_right)/lengthBetweenTwoWheels #heading angle

	    	delta_x = (vx * math.cos(th)) * dt
	    	delta_y = (vx * math.sin(th)) * dt
	    	delta_th = vth * dt

	    	x += delta_x
	    	y += delta_y
	    	th += delta_th

	    	# since all odometry is 6DOF we'll need a quaternion created from yaw
	    	odom_quat = tf.transformations.quaternion_from_euler(0, 0, th)
	    
	    	# first, we'll publish the transform over tf
	    	odom_broadcaster.sendTransform(
			(x, y, 0.),
			odom_quat,
			current_time,
			"base_link",
			"odom"
	    	)

	    	# next, we'll publish the odometry message over ROS
	    	odom = Odometry()
	    	odom.header.stamp = current_time
	    	odom.header.frame_id = "odom"

	    	# set the position
	    	odom.pose.pose = Pose(Point(x, y, 0.), Quaternion(*odom_quat))

	    	# set the velocity
	    	odom.child_frame_id = "base_link"
	    	odom.twist.twist = Twist(Vector3(vx, vy, 0), Vector3(0, 0, vth))

	    	# publish the message
	    	odom_pub.publish(odom)
	    

	    	_PreviousLeftEncoderCounts = tick_x
	    	_PreviousRightEncoderCounts = tick_y
	    	last_time = current_time


	    except:
	    	pass    


	    
	    r.sleep()
