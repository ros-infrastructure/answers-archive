I want to create random coordinates using the GNSS RTK topic to simulate positions in the world only, however I don't know how to do it, I know that the topic and the message used is the one they make available on the anavs official site, can someone help me fix this code ?

    #!/usr/bin/env python2.7
    from nis import match
    import rospy 
    from rospy import Time
    from std_msgs.msg import String
    from sensor_msgs.msg import Image
    from geometry_msgs.msg import PointStamped
    from tf.transformations import quaternion_from_euler,euler_from_quaternion
    import math
    
    def get_RTK_point():
    
        RTKPoint = PointStamped()
        RTKPoint.header.stamp = rospy.Time.now()
        RTKPoint.header.frame_id = "map"
    
        RTKPoint.pose.position.x = 0.0
        RTKPoint.pose.position.y = 0.0
        RTKPoint.pose.position.z = -1
    
    
    pub_rtk = rospy.Publisher('relativ_test', PointStamped, queue_size=10)
    
    
    def callback_RTK(data):
        global  pub_rtk
        array = get_RTK_point(data)
        pub_rtk.publish(array)
    
    def talker():
        rospy.Subscriber('/anavs/solution/pos_xyz', PointStamped, callback_RTK, queue_size = 10)
        rospy.spin()
    
    
    if __name__ == '__main__':
        try:
            talker()
        except rospy.ROSInterruptException:
            pass
