  body_markdown: "Regarding your first question, you want to set the `setMaxIntervalDuration`.\
    \ As far as I understand from a quick look at the sources, a higher value for\
    \ `setAgePenalty` makes the algorithm more sloppy for older messages (to get them\
    \ out of the queue faster) and `setInterMessageLowerBound` allows to set a minimum\
    \ delay between messages of a single input of your filter (messages arriving faster\
    \ than that will be dropped).\n\nAs for the second question, you can just feed\
    \ a SimpleFilter as an input to your Synchronizer:\n\n    #include <ros/ros.h>\n\
    \    #include <image_transport/subscriber_filter.h>\n    #include <message_filters/pass_through.h>\n\
    \    #include <message_filters/simple_filter.h>\n    #include <message_filters/subscriber.h>\n\
    \    #include <message_filters/sync_policies/approximate_time.h>\n    #include\
    \ <sensor_msgs/Image.h>\n    \n    \n    void callback(const sensor_msgs::Image::ConstPtr&\
    \ imagemessage1,\n        const sensor_msgs::Image::ConstPtr& imagemessage2) {\n\
    \    \n    }\n    \n    \n    int main(int argc, char ** argv) {\n      ros::init(argc,\
    \ argv, \"test_synchronizer\");\n      ros::NodeHandle nodehandle;\n    \n   \
    \   image_transport::ImageTransport imagetransport{nodehandle};\n    \n      image_transport::SubscriberFilter\
    \ imagesubscriber1;\n      imagesubscriber1.subscribe(imagetransport, \"/image1\"\
    , 32);\n    \n      image_transport::SubscriberFilter imagesubscriber2;\n    \
    \  imagesubscriber2.subscribe(imagetransport, \"/image2\", 32);\n    \n      message_filters::PassThrough<sensor_msgs::Image>\
    \ passthrough(imagesubscriber2);\n      message_filters::SimpleFilter<sensor_msgs::Image>\
    \ & simplefilter = passthrough;\n    \n      using MyPolicy = message_filters::sync_policies::ApproximateTime<\n\
    \          sensor_msgs::Image, sensor_msgs::Image>;\n      MyPolicy mypolicy(32);\n\
    \      mypolicy.setMaxIntervalDuration(ros::Duration(1.e-3));\n      message_filters::Synchronizer<MyPolicy>\
    \ synchronizer{\n          static_cast<const MyPolicy &>(mypolicy), imagesubscriber1,\
    \ passthrough};\n      synchronizer.registerCallback(boost::bind(&callback, _1,\
    \ _2));\n    \n      ros::spin();\n    \n      return EXIT_SUCCESS;\n    }"
