  body_markdown: "Write yourself a little publisher like:\n\n    #include <iostream>\n\
    \    \n    #include <list>\n    #include <vector>\n    \n    #include <sensor_msgs/Image.h>\n\
    \    #include <cv_bridge/cv_bridge.h>\n    #include <opencv2/highgui/highgui.hpp>\n\
    \    \n    \n    int main ( int argc, char** argv )\n    {\n        if ( argc\
    \ < 2 )\n        {\n          std::cout << \"no image names given\" << std::endl;\n\
    \          return -1;\n        }\n    \n      ros::init( argc, argv, \"image_publisher\"\
    \ );\n      \n      ros::NodeHandle lo_nh;\n      \n      try\n      {\n     \
    \   \n        std::vector<cv::Mat> lv_images;\n     \n        ROS_INFO( \"Loading\
    \ images\" );\n        \n        for ( int lo_i = 1; lo_i < argc; lo_i++ )\n \
    \       {\n          lv_images.push_back( cv::imread(std::string( argv[ lo_i ]\
    \ ), CV_LOAD_IMAGE_COLOR ) );\n        }\n        \n        ros::Publisher lo_pub\
    \ = lo_nh.advertise<sensor_msgs/Image>( \u201C/image\u201D, 5 );\n        \n \
    \         int lo_rate = 10;\n          \n          ROS_INFO_STREAM( \"Publishing\
    \ images at \" << lo_rate << \"hz\" );\n          ros::Rate lo_rate1( lo_rate\
    \ );\n          \n          std::vector<cv::Mat>::iterator lo_iter_imgs( lv_images.begin()\
    \ ), lo_iter_imgs_end( lv_images.end() );\n          \n          for ( ; lo_iter_imgs\
    \ != lo_iter_imgs_end ; ++lo_iter_imgs )\n          {\n    \tcv_bridge::CvImage\
    \ lo_img;\n    \tlo_img.encoding = \"bgr8\";\n    \tlo_img.image = *lo_iter_imgs;\t\
    \n    \t\n    \tlo_pub.publish(lo_img.toImageMsg() );\n    \t\n    \tros::spinOnce();\n\
    \    \tlo_rate1.sleep();\n          }\n          \n      }\n      catch ( const\
    \ std::exception& lo_e )\n      {\n        std::cout << \"Exception in main: \"\
    \ << lo_e.what() << std::endl;\n        \n        return -1;\n      }\n      catch(\
    \ ... )\n      {\n        std::cout << \"Unknown exception in main\" << std::endl;\n\
    \        \n        return -1;\n      }\n      \n      return 0;\n    }\n\nAnd\
    \ call it with all images in the current folder using a shell script like:\n\n\
    \    #!/bin/bash\n    echo \"Working ...\"\n    \n    FILES= \n    \n    for i\
    \ in $(ls *.png); do\n     FILES=\"${FILES} ${i}\"\n    done\n    \n    rosrun\
    \ my_pkg my_image_publisher $FILES\n    \n    echo \"Done\"\n\n\n"
