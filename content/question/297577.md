    #!/usr/bin/env python
    
    import math
    
    # OpenVR package
    import openvr
    
    # ROS packages
    import rospy
    import tf
    
    # ROS messages
    from sensor_msgs.msg import Joy
    from geometry_msgs.msg import Point, Pose, Quaternion
    
    # Constants
    topic_name = "openvr"
    
    # --- adapted from triad_openvr --- #
    #Convert the standard 3x4 position/rotation matrix to a x,y,z location and the appropriate Quaternion
    def convert_to_quaternion(pose_mat):
        # Per issue #2, adding a abs() so that sqrt only results in real numbers
        r_w = math.sqrt(abs(1+pose_mat[0][0]+pose_mat[1][1]+pose_mat[2][2]))/2
        r_x = (pose_mat[2][1]-pose_mat[1][2])/(4*r_w)
        r_y = (pose_mat[0][2]-pose_mat[2][0])/(4*r_w)
        r_z = (pose_mat[1][0]-pose_mat[0][1])/(4*r_w)
    
        x = pose_mat[0][3]
        y = pose_mat[1][3]
        z = pose_mat[2][3]
        return [x,y,z,r_w,r_x,r_y,r_z]
    
    def openvr_state_to_ros_joy(pControllerState, timestamp, device_name):
    	""" 
    	Converts an OpenVR pControllerState (https://github.com/ValveSoftware/openvr/wiki/IVRSystem::GetControllerState)
    	for the HTC Vive Controller (https://docs.unity3d.com/560/Documentation/Manual/OpenVRControllers.html)
    	to a 
    	ROS Joy MSG (http://docs.ros.org/api/sensor_msgs/html/msg/Joy.html)
    	"""
    
    	# See http://docs.ros.org/api/sensor_msgs/html/msg/Joy.htm
    	joy_msg = Joy()
    
    	# Standard ROS header msg http://docs.ros.org/api/std_msgs/html/msg/Header.html
    	joy_msg.header.frame_id = device_name
    	joy_msg.header.stamp = timestamp
    
        # the x axis represents the trigger axis (NOTE there are other axes, for the touchpad, but we are only concerned with the touchpad)
    	trigger_axis = pControllerState.rAxis[1].x
    	button_trackpad = pControllerState.ulButtonPressed
    
    	# third bit marks grip button
    	button_grip = bool(pControllerState.ulButtonPressed >> 2 & 1)
    
    	# Add the trigger axis to the Jog msg
    	joy_msg.axes = [
    		trigger_axis
    	]
    
    	d = {}
    	d['unPacketNum'] = pControllerState.unPacketNum
    	# on trigger .y is always 0.0 says the docs
    	d['trigger'] = pControllerState.rAxis[1].x
    	# 0.0 on trigger is fully released
    	# -1.0 to 1.0 on joystick and trackpads
    	d['trackpad_x'] = pControllerState.rAxis[0].x
    	d['trackpad_y'] = pControllerState.rAxis[0].y
    	# These are published and always 0.0
    	# for i in range(2, 5):
    	#     d['unknowns_' + str(i) + '_x'] = pControllerState.rAxis[i].x
    	#     d['unknowns_' + str(i) + '_y'] = pControllerState.rAxis[i].y
    	d['ulButtonPressed'] = pControllerState.ulButtonPressed
    	d['ulButtonTouched'] = pControllerState.ulButtonTouched
    	# To make easier to understand what is going on
    	# Second bit marks menu button
    	d['menu_button'] = bool(pControllerState.ulButtonPressed >> 1 & 1)
    	# 32 bit marks trackpad
    	d['trackpad_pressed'] = bool(pControllerState.ulButtonPressed >> 32 & 1)
    	d['trackpad_touched'] = bool(pControllerState.ulButtonTouched >> 32 & 1)
    	# third bit marks grip button
    	d['grip_button'] = bool(pControllerState.ulButtonPressed >> 2 & 1)
    	# System button can't be read, if you press it
    	# the controllers stop reporting
    
    	# Add buttons to the Joy msg
    	joy_msg.buttons = [
    		d['trigger'] == 1.0,
            d['trackpad_touched'],
            d['trackpad_pressed'],
            d['menu_button'],
            d['grip_button']
    	]
    
    	return joy_msg
    
    class HTCViveTracking():
    	"""Handles publishing of OpenVR device poses"""
    
    	def __init__(self, vr):
    		"""Initialise Vive publisher"""
    		print("Initialising Vive tracker")
    
    		print("Setting up VR scene")
    
    		self.vr = vr
    
    		print("Initially getting devices")
    		self.poses_t = openvr.TrackedDevicePose_t * openvr.k_unMaxTrackedDeviceCount
    
    		self.update_vr()
    		self.check_devices()
    
    		self.publisher = {}
    		self.num_packet = {}
    		
    		print("Waiting until 2 controllers are connected")
    		# Loop until two controllers connected
    		while True:
    			if len(self.object_names["Controller"]) == 2:
    				print("Both controllers set up")
    				break
    			else:
    				self.check_devices()
    		print("2 controllers connected. Setting up publishers")
    
    		# Create publishers (controllers also have an additional joy topic)
    		for device_name in self.devices:
    			self.publisher[device_name] = rospy.Publisher("/{}/{}/pose".format(topic_name, device_name), Pose, queue_size=10)
    			
    			if "controller" in device_name:
    				self.publisher[device_name+"_joy"] = rospy.Publisher("/{}/{}/joy".format(topic_name, device_name), Joy, queue_size=10)
    				self.num_packet[device_name] = 0
    
    		print("Initialisation finished")
    
    		
    	def check_devices(self):
    		"""Identify connected devices"""
    		self.devices = {}
    		self.object_names = {"Tracking Reference":[],"HMD":[],"Controller":[],"Tracker":[]}
    
    		for i in range(openvr.k_unMaxTrackedDeviceCount):
    			# Get this device's class https://github.com/ValveSoftware/openvr/wiki/IVRSystem::GetTrackedDeviceClass
    			device_class = self.vr.getTrackedDeviceClass(i)
    
    			# If device is still connected See https://github.com/ValveSoftware/openvr/blob/d35c04ca3d7ddb762afa4a1c6c4a9d5de992cda7/headers/openvr.h#L189
    			if self.poses[i].bPoseIsValid and self.poses[i].bDeviceIsConnected:
    				
    				# If device is a controller
    				if (device_class == openvr.TrackedDeviceClass_Controller):
    					device_num = len(self.object_names["Controller"])+1
    					device_name = "controller_{}".format(device_num)
    					self.object_names["Controller"].append(device_name)
    					self.devices[device_name] = i
    
    				# Or device is and HMD
    				elif (device_class == openvr.TrackedDeviceClass_HMD):
    					device_num  = len(self.object_names["HMD"])+1
    					device_name = "hmd_{}".format(device_num)
    					self.object_names["HMD"].append(device_name)
    					self.devices[device_name] = i
    
    	def update_vr(self):
    		"""Update the poses, as recommended by (TODO: openvr github comment)"""
    
    		# Reset pose list
    		self.poses = self.poses_t()
    
    		# Get pose states from openvr https://github.com/ValveSoftware/openvr/wiki/IVRSystem::GetDeviceToAbsoluteTrackingPose
    		self.vr.getDeviceToAbsoluteTrackingPose(
                openvr.TrackingUniverseStanding,
                0,
                len(self.poses),
                self.poses)
    
    	def publish_data(self):
    		"""Publishes pose data (and state data for controllers) of OpenVR devices to ROS topics"""
    
    		# Update device and VR state
    		self.update_vr()
    		self.check_devices()
    
    		# What time is it?
    		now = rospy.Time.now()
    
    		# Loop through name of each connected device
    		for device_name in self.devices:
    
    			# Get device, device's pose and publisher
    			device = self.devices[device_name]
    			publisher = self.publisher[device_name]
    			pose = self.poses[device].mDeviceToAbsoluteTracking
    
    			# Get device pose as a quaternion
    			[x, y, z, qx, qy, qz, qw] = convert_to_quaternion(pose)
     
    			# Publish controller state data
    			if "controller" in device_name:
    				# Rotate Vive Trackers 180, so Z+ comes out of the top of the Tracker
    				[qx, qy, qz, qw] = tf.transformations.quaternion_multiply([qx, qy, qz, qw], tf.transformations.quaternion_from_euler(math.pi, 0.0, -math.pi/2.0))
    
    				# Get state https://github.com/ValveSoftware/openvr/wiki/IVRSystem::GetControllerState
    				result, state = self.vr.getControllerState(device)
    				
    				# ID state by packet number
    				state_packet = state.unPacketNum
    
    				# Check that this is a different state packet
    				if state_packet != self.num_packet[device_name]:
    					# Convert state to ROS joystick message
    					state_joy = openvr_state_to_ros_joy(state, now, device_name)
    
    					# Publish controller state
    					self.publisher[device_name+"_joy"].publish(state_joy)
    
    				# Set last packet number to current
    				self.num_packet[device_name] = state_packet
    				
    			# Build Pose
    			pose = Pose(Point(x, y, z), Quaternion(qx,qy,qz,qw))
    
    			# Publish pose
    			publisher.publish(pose)
    
    	def shutdown(self):
    		"""Shuts down the OpenVR application"""
    
    		openvr.shutdown()
