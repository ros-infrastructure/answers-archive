  body_markdown: "You need to [create a publisher](http://wiki.ros.org/ROS/Tutorials/WritingPublisherSubscriber%28python%29)\
    \ and publish a [geometry_msgs/Pose](http://docs.ros.org/melodic/api/geometry_msgs/html/msg/Pose.html)\
    \ message.\n\n    #!/usr/bin/env python\n    import rospy\n    from geometry_msgs.msg\
    \ import Pose\n    \n    def publisher():\n        pub = rospy.Publisher('pose',\
    \ Pose, queue_size=1)\n        rospy.init_node('pose_publisher', anonymous=True)\n\
    \        rate = rospy.Rate(2) # Hz\n        while not rospy.is_shutdown():\n \
    \           p = Pose()\n            p.position.x = 0.5\n            p.position.y\
    \ = -0.1\n            p.position.z = 1.0\n            # Make sure the quaternion\
    \ is valid and normalized\n            p.orientation.x = 0.0\n            p.orientation.y\
    \ = 0.0\n            p.orientation.z = 0.0\n            p.orientation.w = 1.0\n\
    \            pub.publish(p)\n            rate.sleep()\n    \n    if __name__ ==\
    \ '__main__':\n        try:\n            publisher()\n        except rospy:\n\
    \            pass"
