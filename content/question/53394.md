  body_markdown: "An example might look like this (roughly):\n\n    import rospy\n\
    \    from std_msgs.msg import Int32\n    \n    class Echo(object):\n        def\
    \ __init__(self):\n            self.value = 0\n    \n            rospy.init_node('echoer')\n\
    \                   \n            self.pub = rospy.Publisher('/out_value', Int32,\
    \ latch=True)\n            rospy.Subscriber('/in_value', Int32, self.update_value)\n\
    \        \n        def update_value(self, msg):\n            self.value = msg.data\n\
    \    \n        def run(self):\n            r = rospy.Rate(10)\n            while\
    \ not rospy.is_shutdown():\n                self.pub.publish(self.value)\n   \
    \             r.sleep()\n\n"
