  body_markdown: "in your CMakeLists.txt:\n\n        find_package( CUDA REQUIRED \n\
    \                                                    \n     # set CUDA_NVCC_FLAGS\
    \ as you would do with CXX/C FLAGS                        \n      set(CUDA_NVCC_FLAGS\
    \ \"${CUDA_NVCC_FLAGS} -DMY_DEF=1\")\n      set( CMAKE_CXX_FLAGS \"${CMAKE_CXX_FLAGS}\
    \ -DMY_DEF=1\" )\n      set( CMAKE_C_FLAGS \"${CMAKE_C_FLAGS} -DMY_DEF=1\" )\n\
    \    \n     # only necessary if you wish to create a lib\n      set(CUDA_ATTACH_VS_BUILD_RULE_TO_CUDA_FILE\
    \ OFF)\n      set(BUILD_SHARED_LIBS ON)\n    \n    # cuda compile GPU_CODE for\
    \ executable\n      cuda_compile(  HELLO_CU_O src/hello.cu )\n    \n    ## create\
    \ an executable\n    ## can add other (none CUDA) cpp files here\n      cuda_add_executable(\
    \ hello_exec ${LINE_PROFILER_CU_O} )\n      target_link_libraries( hello_exec\
    \ ${catkin_LIBRARIES} )\n      add_dependencies( hello_lib ${PROJECT_NAME}_generate_messages_cpp\
    \  ${LINE_PROFILER_CU_O} )\n    \n    ## if message headers needed \n      add_dependencies(\
    \ hello_lib ${PROJECT_NAME}_generate_messages_cpp )\n    \n    ### or create a\
    \ lib \n      cuda_add_library( ${PROJECT_NAME} src/lib_cuda_code.cu src/libe_none_cuda_code.cpp\
    \ )\n      target_link_libraries( ${PROJECT_NAME} ${catkin_LIBRARIES}  )\n   \
    \ \n    ## if message headers needed \n      add_dependencies( ${PROJECT_NAME}\
    \ ${PROJECT_NAME}_generate_messages_cpp )\n\n"
