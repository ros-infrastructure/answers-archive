    \ /camera/rgb/camera_info /camera/rgb/image_rect_color\n\nDon't forget to use\
    \ these topics in your launchfile for rgbdslam (and set config/topic_points to\
    \ an empty string), e.g.:\n\n\n    <launch>\n      <node pkg=\"rgbdslam\" type=\"\
    rgbdslam\" name=\"rgbdslam\" cwd=\"node\" required=\"true\" output=\"log\"> \n\
    \        <!-- Input data settings-->\n        <param name=\"config/topic_image_mono\"\
    \              value=\"/camera/rgb/image_rect_color\"/> <!--could also be color\
    \ -->\n        <param name=\"config/topic_image_depth\"             value=\"/camera/depth/image_rect\"\
    />\n        <param name=\"config/topic_points\"                  value=\"\"/>\
    \ <!--if empty, poincloud will be reconstructed from image and depth -->\n   \
    \ \n        <param name=\"config/feature_detector_type\"         value=\"SURF\"\
    />\n        <param name=\"config/feature_extractor_type\"        value=\"SURF\"\
    />\n        <param name=\"config/max_keypoints\"                 value=\"600\"\
