    I resolved both issues. Obstacle bumping and following global path.
    For following global path:
    I used the TEB parameters of global_plan_viapoint_sep set to a positive value and set the weight_viapoint at a slightly lower value to the weight_obstacle.

    For obstacle bumping issue:
    Two things of note:
    The global planner does not consider your robot radius or footprint. It will assume it is a point. Which is why the inflation settings exist. Make sure that the inflation radius is slightly larger than your robot radius. This way, the global planner will plan with sufficient padding.
    The TEB local planner uses it's own settings instead of the settings from your local_costmap. I played with inflation_dist and min_obstacle_dist settings to get the optimal behavior I needed.

    If anyone is stuck dealing with ackermann robots, reach out to me. I think I learned a lot by failing.

The output of the navigation2 package is cmd_vel. The Upsquared Robomaker prokit takes ackermann_msg as input. Hence, I run the following script to create a ROS node that converts cmd_vel to ackermann_msg.
<pre><code>
#!/usr/bin/env python3

import rclpy, math
import numpy
import sys
from geometry_msgs.msg import Twist
from ackermann_msgs.msg import AckermannDriveStamped
from rclpy.qos import QoSHistoryPolicy
from rclpy.qos import QoSProfile
from rclpy.exceptions import ROSInterruptException

def convert_trans_rot_vel_to_steering_angle(v, omega, wheelbase):
  if omega == 0 or v == 0:
    return 0.0

  radius = v / omega
  return math.atan(wheelbase / radius)


def cmd_callback(data):
  global wheelbase
  global ackermann_cmd_topic
  global frame_id
  global pub
  global cmd_angle_instead_rotvel

  v = data.linear.x

  if cmd_angle_instead_rotvel:
    #steering = data.angular.z
    #steering = data.angular.z*1.55 #option1
    data.angular.z = numpy.clip(data.angular.z, -0.36, 0.36)
    steering = data.angular.z * (1.0 + (0.36 - abs(data.angular.z)))
  else:
    steering = convert_trans_rot_vel_to_steering_angle(v, data.angular.z, wheelbase) * 3 #Worked decently up to a point.

  msg = AckermannDriveStamped()
  msg.header.stamp = node.get_clock().now().to_msg() # TODO get current time from Node.
  msg.header.frame_id = frame_id

  msg.drive.steering_angle = steering
  msg.drive.speed = v

  pub.publish(msg)


if __name__ == '__main__':
  try:

    rclpy.init(args=sys.argv)
    global node
    node = rclpy.create_node('twist_to_ackermann_node')

    twist_cmd_topic = node.declare_parameter('~twist_cmd_topic', 'cmd_vel').value
    ackermann_cmd_topic = node.declare_parameter('~ackermann_cmd_topic', 'ackermann_cmd').value
    wheelbase = node.declare_parameter('~wheelbase', 0.16).value
    frame_id = node.declare_parameter('~frame_id', 'odom').value
    cmd_angle_instead_rotvel = node.declare_parameter('cmd_angle_instead_rotvel', False).value

    qos_profile = QoSProfile(depth=1, history=QoSHistoryPolicy.KEEP_LAST)
    node.create_subscription(Twist, twist_cmd_topic, cmd_callback, qos_profile)
    pub = node.create_publisher(AckermannDriveStamped, ackermann_cmd_topic, qos_profile)

    node.get_logger().info("Node 'twist_to_ackermann_node' started.\nListening to /cmd_vel, publishing to %s. Frame id: %s, wheelbase: %f" % (ackermann_cmd_topic, frame_id, wheelbase))

    rclpy.spin(node)

  except ROSInterruptException:
    pass
