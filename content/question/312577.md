    \ minimal example of building a ROS 1 package from source.\nJust swap melodic\
    \ for kinetic if that is the distro you still require; the following includes:\
    \  \nCloning source code into workspace  \nInstalling package dependencies  \n\
    Building package using catkin tools  \nSourcing workspace into entrypoint  \n\
    And launching package nodes\n\n\n    FROM ros:melodic\n    \n    # install build\
    \ tools\n    RUN apt-get update && apt-get install -y \\\n          python-catkin-tools\
    \ \\\n        && rm -rf /var/lib/apt/lists/*\n    \n    # clone ros package repo\n\
    \    ENV ROS_WS /opt/ros_ws\n    RUN mkdir -p $ROS_WS/src\n    WORKDIR $ROS_WS\n\
    \    RUN git -C src clone \\\n          -b $ROS_DISTRO-devel \\\n          https://github.com/ros/ros_tutorials.git\n\
    \    \n    # install ros package dependencies\n    RUN apt-get update && \\\n\
    \        rosdep update && \\\n        rosdep install -y \\\n          --from-paths\
    \ \\\n            src/ros_tutorials/roscpp_tutorials \\\n          --ignore-src\
    \ && \\\n        rm -rf /var/lib/apt/lists/*\n    \n    # build ros package source\n\
    \    RUN catkin config \\\n          --extend /opt/ros/$ROS_DISTRO && \\\n   \
    \     catkin build \\\n          roscpp_tutorials\n    \n    # source ros package\
    \ from entrypoint\n    RUN sed --in-place --expression \\\n          '$isource\
    \ \"$ROS_WS/devel/setup.bash\"' \\\n          /ros_entrypoint.sh\n    \n    #\
    \ run ros package launch file\n    CMD [\"roslaunch\", \"roscpp_tutorials\", \"\
