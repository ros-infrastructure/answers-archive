    \ 's example project, I was able to solve the issue. I basically generated two\
    \ projects from the same package and Cmake. \nI am aware it's not best practice\
    \ - BUT I had to do it because I also had to pack it into a debian(Which can only\
    \ pack one project).\n\nFollowing is a template for such a project if anyone needs\
    \ it in the future:\n\n    find_package(rosidl_default_generators REQUIRED)\n\
    \    ...\n    \n    set(library_name mynode)\n    \n    rosidl_generate_interfaces(${PROJECT_NAME}\n\
    \      \"msg/msgFile.msg\"\n    )\n    \n    add_library(${library_name} SHARED\n\
    \            ${INCLUDES}\n            ${SOURCES}\n            )\n    \n    rosidl_get_typesupport_target(cpp_typesupport_target\
    \ \"${PROJECT_NAME}\" \"rosidl_typesupport_cpp\")\n    target_link_libraries(${library_name}\
    \ \"${cpp_typesupport_target}\")\n    \n    ament_target_dependencies(${library_name}\n\
    \    ${dependencies}\n    )\n    \n    target_link_libraries(${library_name}\n\
    \    ...\n            )\n    \n    install(TARGETS ${library_name} \n    ARCHIVE\
    \ DESTINATION lib\n    LIBRARY DESTINATION lib\n    RUNTIME DESTINATION bin\n\
    \    )\n    \n    rclcpp_components_register_node(${library_name}\n      PLUGIN\
    \ \"Myclass::ActiveNode\"\n      EXECUTABLE activenode\n    )\n    \n    target_include_directories(activenode\
    \ PUBLIC\n      $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>\n     \
    \ $<INSTALL_INTERFACE:include>\n    ...\n            )\n    target_compile_features(activenode\
    \ PUBLIC c_std_99 cxx_std_17)  # Require C99 and C++17\n    \n    \n    ## Specify\
    \ libraries to link a library or executable target against\n    target_link_libraries(activenode\n\
    \    ...\n            )\n    \n    ament_target_dependencies(activenode\n    ${dependencies}\n\
    \    )\n    \n      # Install binaries\n    install(TARGETS activenode\n    ARCHIVE\
    \ DESTINATION lib\n    LIBRARY DESTINATION lib\n    RUNTIME DESTINATION bin\n\
    \    )\n    \n    # Install headers\n    install(\n    DIRECTORY include/\n  \
    \  DESTINATION include\n    )\n    \n    # Install launch files\n    install(DIRECTORY\
    \ \n      launch\n      DESTINATION share/${PROJECT_NAME}\n      )\n    \n   \
    \ install(DIRECTORY\n      config\n      DESTINATION share/${PROJECT_NAME}\n \
    \   )\n    \n    if(BUILD_TESTING)\n      find_package(ament_lint_auto REQUIRED)\n\

    \   # comment the line when a copyright and license is added to all source files\n\

    \ files\n      set(ament_cmake_cpplint_FOUND TRUE)\n      ament_lint_auto_find_test_dependencies()\n\
    \    endif()\n    \n    ament_export_include_directories(include)\n    ament_export_libraries(${library_name})\n\
    \    ament_export_dependencies(${dependencies} )\n    \n    ament_package()\n\n"
