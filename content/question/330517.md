    \ \"std_msgs/Bool.h\"\n\n    ros::Publisher chatter_pub;\n\n\n    void chatterCallback(const\
    \ std_msgs::UInt8::ConstPtr& msg)\n    {\n      ROS_INFO(\"I heard: [%d]\", msg->data);\n\
    \n      std_msgs::Bool msgOut;\n      msgOut.data = false;\n      if (msg->data\
    \ > 15)\n         msgOut.data = true;\n\n      chatter_pub.publish(msgOut);\n\n\
    \    }\n\n    int main(int argc, char **argv)\n    {\n      ros::init(argc, argv,\
    \ \"sub_and_pub_node\");\n\n      ros::NodeHandle nh;\n\n      ros::Subscriber\
    \ sub = nh.subscribe(\"active_listener\", 1, chatterCallback);\n      chatter_pub\
    \ = nh.advertise<std_msgs::Bool>(\"chatter_within_subscriber\", 1);\n\n      ros::spin();\n\
    \n      return 0;\n    }\n\nHope it helps and is helpful enough despite the removed\
