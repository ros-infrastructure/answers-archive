  body_markdown: "That's not ROS related but only C++, I would advise to start learning\
    \ the language first and then come back to ROS.\n\nAs for your question if you\
    \ want to have an object of type `moveit::planning_interface::MoveGroupInterface`\
    \ you will need to declare an attribute pointing to an object of this type because\
    \ if you don't the object will be destroyed at the end of the function (in your\
    \ case the constructor). With a pointer the memory is allocated until you call\
    \ `delete` :\n\n    #include <moveit/move_group_interface/move_group_interface.h>\n\
    \    \n    class Move_operations_class \n    {\n        public:\n            //Constructor\n\
    \            Move_operations_class(const std::string &move_group_name) \n    \
    \        {\n                //Now DEFINE your pointer to MoveGroupInterface object\n\
    \                move_group_ptr_ = new moveit::planning_interface::MoveGroupInterface(move_group_name);\n\
    \            };\n    \n           //Destructor to dealocate memory\n         \
    \   ~Move_operations_class()\n           {\n               delete move_group_ptr_;\n\
    \           }\n\n            //DECLARE the pointer\n            moveit::planning_interface::MoveGroupInterface\
    \ *move_group_ptr_;\n    };\n    \n    int main(int argc, char** argv)\n    {\n\
    \        //ROS init\n    \n        Move_operations_class move_group_main(\"arm_torso\"\
    );\n        move_group_main.move_group_ptr_->setPoseReferenceFrame(\"base_footprint\"\
    );\n\n        return 0;\n    }"
