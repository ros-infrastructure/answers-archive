    \ (pclPCacumulada, pclmsg, pclPCacumulada);\n    \tpcl_conversions::fromPCL(pclPCacumulada,\
    \ PCacumulada);\t\n    }\n    \n    int main (int argc, char ** argv)\n    {\n\
    \    \tros::init(argc,argv,\"nodeProcesador\"); \n    \tros::NodeHandle ns; \n\
    \    \tros::NodeHandle np; \n    \tros::Subscriber subscriptor = ns.subscribe(\"\
