                // GL Viewer
                zed.retrieveMeasure(point_cloud, sl::MEASURE::XYZRGBA, sl::MEM::GPU, pc_resolution);
                zed.getPosition(cam_w_pose, sl::REFERENCE_FRAME::WORLD);
                viewer.updateData(point_cloud, objects.object_list, cam_w_pose.pose_data);
            }
        }
