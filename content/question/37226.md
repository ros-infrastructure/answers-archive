    \ Instead, create a overlay in your home directory. To create an overlay for electric,\
    \ just execute the following commands:\n\n    sudo apt-get install python-pip\n\
    \    sudo pip install -U rosinstall\n    rosws init ~/ros_overlay /opt/ros/electric\n\
    \    mkdir ~/ros_overlay/sandbox\n    source ~/ros_overlay/setup.bash\n    rosws\
    \ set ~/ros_overlay/sandbox\n    \nNow open your ~/.bashrc and change the line\n\
    \n    source /opt/ros/electric/setup.bash\n\nto\n\n    source ~/ros_overlay/setup.bash\n\
    \nNow close and re-open your terminal and execute the command `roscreate-pkg`\
    \ in ~/ros_workspace/sandbox.\n"
