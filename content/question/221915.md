    \ package.\nThis is an example to change the PointCloud type from PointXYZ to\
    \ PointXYZRGB\n\n    #include <iostream>\n    #include <string>\n    #include\
    \ <ros/ros.h>\n    #include <pcl_ros/point_cloud.h>\n    #include <pcl/point_types.h>\n\
    \    #include <pcl_conversions/pcl_conversions.h>\n    #include <pcl_ros/io/pcd_io.h>\n\
    \        \n    typedef pcl::PointXYZRGB PointT;\n    typedef pcl::PointCloud<PointT>\
    \ PointCloud;\n    using namespace std;\n    \n    int main(int argc, char** argv)\n\
    \    {\n        PointCloud::Ptr cloud(new PointCloud);\n        ros::init (argc,\
    \ argv, \"my_pcl_tutorial\");\n        ros::NodeHandle nh;\n        ros::Publisher\
    \ pub = nh.advertise<PointCloud> (\"points\", 1);\n        pcl::io::loadPCDFile<PointT>\
    \ (\"yourpcd.pcd\", *cloud);\n    \n        PointCloud::Ptr msg(new PointCloud);\n\
    \        msg->header.frame_id = \"map\";\n        msg->height = cloud->height;\n\
    \        msg->width = cloud->width;\n    \n        PointT p;\n        for (size_t\
    \ i = 0; i < cloud->size(); i++) {\n            p.x = cloud->points[i].x;\n  \
    \          p.y = cloud->points[i].y;\n            p.z = cloud->points[i].z;\n\
    \            p.r = cloud->points[i].r;\n            p.g = cloud->points[i].g;\n\
    \            p.b = cloud->points[i].b;\n            msg->points.push_back(p);\n\
    \        }\n    \n        ros::Rate loop_rate(4);\n    \n        while (nh.ok())\n\
    \        {\n            pub.publish (msg);\n            ros::spinOnce ();\n  \
    \          loop_rate.sleep ();\n        }\n    }"
