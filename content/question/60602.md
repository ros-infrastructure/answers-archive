    \ It comes with turtlebot stack. You need to create a launch file say kinect_laser.launch\

    \ output=\"screen\" respawn=\"true\" args=\"manager\"/>\n \n      <!-- throttling\
    \ -->\n      <node pkg=\"nodelet\" type=\"nodelet\" name=\"pointcloud_throttle\"\
    \ args=\"load pointcloud_to_laserscan/CloudThrottle openni_manager\">\n      \
    \  <param name=\"max_rate\" value=\"2\"/>\n        <remap from=\"cloud_in\" to=\"\
    /camera/depth/points\"/>\n        <remap from=\"cloud_out\" to=\"cloud_throttled\"\
    />\n      </node>\n \n      <!-- fake laser -->\n      <node pkg=\"nodelet\" type=\"\
    nodelet\" name=\"kinect_laser\" args=\"load pointcloud_to_laserscan/CloudToScan\
    \ openni_manager\">\n        <param name=\"output_frame_id\" value=\"/camera_depth_frame\"\
    />\n        <remap from=\"cloud\" to=\"cloud_throttled\"/>\n      </node>\n  \
