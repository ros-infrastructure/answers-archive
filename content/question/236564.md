    moveit_msgs::Constraints PC;

    OC.header.frame_id = "mico_link_base";
    OC.link_name = "mico_link_7";

    OC.absolute_x_axis_tolerance = 0.2;
    OC.absolute_y_axis_tolerance = 2*PI;
