    \ relative to the frame that you pass in the header. So as you already have the\

    \ set the header.frame_id to you first frame:\n\n    visualization_msgs::Marker\
    \ marker;\n    marker.header.frame_id = \"frame1\";\n    marker.type = visualization_msgs::Marker::CYLINDER;\n\
    \    marker.pose.position = [translation of other frame]/2\n    marker.pose.orientation\
    \ = orientation of frame2 relative to frame1\n    marker.scale.x = marker.scale.y\
    \ = 0.1; // diameter of cylinder\n    marker.scale.z = translation.z;  \n    marker.color.a\
