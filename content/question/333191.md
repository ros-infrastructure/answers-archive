  body_markdown: "Thx to gvdhoorn's comment, I managed to write the following simple\
    \ script to do what I want\n\n\n    #!/usr/bin/python2\n    \n    import yaml\n\
    \    import rosbag\n    import argparse\n    import re\n    \n    parser = argparse.ArgumentParser(description=\"\
    Greps /rosout for a specific message and reports its timestamps\")\n    parser.add_argument(\"\
    bagfile\", nargs=1, help=\"input bag file\")\n    parser.add_argument(\"regex\"\
    , nargs=1, help=\"regex pattern to search for\")\n    args = parser.parse_args()\n\
    \    \n    bagfile = args.bagfile[0]\n    regex = args.regex[0]\n    \n    info_dict\
    \ = yaml.load(rosbag.Bag(bagfile, 'r')._get_yaml_info())\n    \n    for topic,\
    \ msg, t in rosbag.Bag(bagfile).read_messages():\n        if topic == \"rosout\"\
    :\n            if re.search(regex, msg.msg) or re.search(regex, msg.name):\n \
    \               seconds_from_start = t.to_sec() - info_dict[\"start\"]\n     \
    \           print(\"[\" + str(seconds_from_start) + \"] \" + msg.msg)\n\n"
