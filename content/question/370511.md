Hi,

I have a rather basic python2 node (it's the interface to the motor controllers)

- a main loop running (theoretically) at 10Hz (using rate.sleep() )
- 2 subscribers for velocities commands (only one is used at a given time, depending of the navigation mode) : the frequency should be about 10Hz on the active one (and 0 on the other one). Both have a queu size of 1
- 1 subscriber for proximity sensors (about 10Hz). Queue_size=1
- 2 publishers (publishing done from the main loop

Most of the time everything runs OK.
However, sometimes, the main loop seems to be "stuck" for sometimes more than 1 second in a random place, while the message callbacks seem to be handled fine.


My guess is that sometimes, the main thread isn't executed for about 1 second, while the callback thread is still running.

From my understanding (please correct me if I'm wrong, I'm quite unsure about it), in Python there is only one thread running at a given time (even if multiple CPU cores are available). From what I understood in python (contrary to C++), the message callbacks are handled on a separate (python) thread).

So it seems that the callback thread gets executed but not the main thread.

NB : this even is rather execeptionnal (about once per minute under very heavy CPU load from other nodes, far less under lighter load).



For this specific use case, The main loop is more important than the message callbacks.

Is there a way to make sure the main loop gets executed (dropping messages in the input queues if needed) rather than processing all incomming messages and not executting fat enough the main loop?

A few potential ideas (but I'm not sure if they are and if so how) :

- increase the main thread priority (or decrease the message handling thread) : from my understanding it is not possible because python is handling itself it's threads)

- disable the message processing thread and une something like the C++ spinOnce function

- have a way to explicitly aks for the latest data (ie not having any callbacks at all). If no new data arrived, getting "None" or getting the last recieved data are both fine to me


Thanks a lot in advance

Felix

    distances_US=[255,255,255,255,255,255,255,255,255,255,255]

    FRONT_RIGHT=6

    class RoboclawNode:
    	def __init__(self):
    		self.mode=1
    		self.time_last_receive_cmd_vel = time.time()
    		self.omega_cmd_g = 0
    		self.omega_cmd_d = 0

    		self.vth_pub = rospy.Publisher('vth', Float64, queue_size=10)
    		self.battery_motors_voltage_pub = rospy.Publisher('battery_motors_voltage', Float32, queue_size=1)
    		rospy.Subscriber('cmd_vel', Twist, self.cmdVelCallback, queue_size=1)
    		rospy.Subscriber('cmd_vel_joystick', Twist, self.joystickCallback, queue_size=1)
    		rospy.Subscriber('capteur_proximite', capteur_proximite_msg, self.distanceCallback, queue_size=1)
    		rospy.init_node('roboclaw_node', anonymous=True)
    		
    		# We have a single 'roboclaw' object handling serial communications.
    		# We're about to launch different threads that each want to talk.
    		# 1 - Diagnostics thread calling into our self.check_vitals
    		# 2 - '/cmd_vel' thread calling our self.cmd_vel_callback
    		# 3 - our self.run that publishes to '/odom'
    		# To prevent thread collision in the middle of serial communication
    		# (which causes sync errors) all access to roboclaw from now

    	def cmdVelCallback(self, cmd_vel_msg):

      	def distanceCallback(self, distance_msg):

    		distances_US[7] = distance_msg.data7
    		distances_US[8] = distance_msg.data8
    		distances_US[9] = distance_msg.data9
    
      	def joystickCallback(self, cmd_vel_msg):

      		self.vth_cmd = cmd_vel_msg.angular.z
    		self.mode=2
    		self.obstaclesLimitSpeed()
      		self.omega_cmd_g = int((2 * self.vx_cmd - chi * e * self.vth_cmd) / (2 * r) * 2400 * rapport_codeur / (2 * PI))
     		self.omega_cmd_d = int((2 * self.vx_cmd + chi * e * self.vth_cmd) / (2 * r) * 2400 * rapport_codeur / (2 * PI))
     		self.time_last_receive_cmd_vel = time.time()
    
    	def constrain_abs_value(self, value, abs_limit):
    		ret=value
    		if value < -abs_limit:
    			ret= -abs_limit
    		if value > abs_limit:
    			ret= abs_limit
    		return ret
    
    	def contraint_abs_value_pair_proportionnal(self, value_1, abs_limit_1, value_2, abs_limit_2):	#on limite les valeurs absolues, en gardant le rapport de proportionnalite
    		abs_1=abs(value_1)
    		abs_2=abs(value_2)
    		factor=1
    		if abs_1 != 0 and abs_1 > abs_limit_1:
    			factor = min(factor, abs_limit_1/abs_1)
    		
    		if abs_2 != 0 and abs_2 > abs_limit_2:
    			factor = min(factor, abs_limit_2/abs_2)
    
    		rospy.loginfo("factor:")
    		rospy.loginfo(factor)
    		return (value_1*factor, value_2*factor)
    
    	def obstaclesLimitSpeed(self):
    		global distances_US
    		distance_in_front=min(distances_US[FRONT_LEFT], distances_US[FRONT_RIGHT])
    		distance_all_arround=min(distances_US[FRONT_LEFT], distances_US[FRONT_RIGHT],distances_US[LEFT_FRONT])
    		distance_all_arround_except_back_and_left_back=min(distances_US[FRONT_LEFT], distances_US[FRONT_MIDDLE], distances_US[FRONT_RIGHT], distances_US[LEFT_FRONT], distances_US[LEFT_MIDDLE], distances_US[RIGHT_MIDDLE], distances_US[RIGHT_BACK])
    
    		
    		rospy.loginfo(distance_in_front)
    		rospy.loginfo(distance_all_arround)
    		rospy.loginfo("initial:")
    		rospy.loginfo(self.vx_cmd)
    		rospy.loginfo(self.vth_cmd)
    
    		tance_all_arround<60:
    				rospy.logwarn("distance_all_arround <60")
    				if self.vth_cmd > 0.2:		#si on veut tourner : on tourne lentement et on n'avance pas
    					self.vx_cmd=0
    					self.vth_cmd=self.constrain_abs_value(self.vth_cmd,0.2)
    				else:	#on ne tourne pas ou presque
    					self.vx_cmd=self.constrain_abs_value(self.vx_cmd,0.5)	#NB : on n'est pas trop strict, mais on n'a pas ici de return : il se peut qu'on soit plus stricte une fois qu'on a vu la distance devant le robot
    					self.vth_cmd=self.vth_cmd	#vitesse angulaire deja faible : on ne la change pas	
    			
    			if distance_in_front < 150:
    				rospy.logwarn("front <150")
    				self.vx_cmd=self.constrain_abs_value(self.vx_cmd,0.2)			
    			if distance_in_front < 80:
    				rospy.logwarn("front <80")
    				self.vx_cmd=0
    		else :
    			if (self.mode==2):	#joystick
    				rospy.loginfo("mode2")
    				rospy.loginfo(distance_all_arround_except_back_and_left_back)
    				if distance_all_arround_except_back_and_left_back <30:
    					rospy.loginfo("all arround <30")
    					(self.vx_cmd, self.vth_cmd) = self.contraint_abs_value_pair_proportionnal(self.vx_cmd, 0.1, self.vth_cmd, 0.1)
    				#if distance_all_arround_except_back_and_left_back<40:
    				#	rospy.loginfo("all arround <40")
    				#	(self.vx_cmd, self.vth_cmd) = self.contraint_abs_value_pair_proportionnal(self.vx_cmd, 0.2, self.vth_cmd, 0.2)
    				if distance_in_front < 60:
    					rospy.loginfo("distance_in_front < 60")
    					self.vx_cmd=self.constrain_abs_value(self.vx_cmd,0.1)
    				if distance_in_front < 100:
    					rospy.loginfo("distance_in_front < 60")
    					self.vx_cmd=self.constrain_abs_value(self.vx_cmd,0.25)	
    
    		rospy.loginfo("final:")
    		rospy.loginfo(self.vx_cmd)
    		rospy.loginfo(self.vth_cmd)
    		rospy.loginfo("")

    				if (time.time() - self.time_last_receive_cmd_vel > 0.5) :	#timeout : on ralenti progressivement

    					roboclaw.SpeedAccelM1M2(ADDRESS_FRONT, self.accelMax*3/4, self.omega_cmd_g, self.omega_cmd_d)

    	def getAndPublishMotorBatery(self):
    		(success,front_voltage)=roboclaw.ReadMainBatteryVoltage(ADDRESS_FRONT)
    		#back_voltage=roboclaw.ReadMainBatteryVoltage(ADDRESS_BACK)
    		if success:
    			voltage_msg=Float32()
    			voltage_msg.data=front_voltage/10.
    			self.battery_motors_voltage_pub.publish(voltage_msg)
    		else:
    			rospy.logwarn("reading battery voltage from roboclaw failed")
    
    
    	def getSpeed(self):
    		with self.mutex:
    			buffer_g_front = roboclaw.ReadSpeedM1(ADDRESS_FRONT)
    	       		buffer_d_front = roboclaw.ReadSpeedM2(ADDRESS_FRONT)
    			buffer_g_back = roboclaw.ReadSpeedM1(ADDRESS_BACK)
    	       		buffer_d_back = roboclaw.ReadSpeedM2(ADDRESS_BACK)
    			if buffer_g_front[0]:
    				self.omega_mes_g_front = buffer_g_front[1]
    			else:
    				self.omega_mes_g_front = 0
    			if buffer_d_front[0]:
    				self.omega_mes_d_front = buffer_d_front[1]
    			else:
    				self.omega_mes_d_front = 0
    
    			if buffer_g_back[0]:
    				self.omega_mes_g_back = buffer_g_back[1]
    			else:
    				self.omega_mes_g_back = 0
    			if buffer_d_back[0]:
    				self.omega_mes_d_back = buffer_d_back[1]
    			else:
    				self.omega_mes_d_back = 0
    
    
    	def publishSpeed(self):
    		vx_msg = Float64()
    		vth_msg = Float64()
    		vx_msg.data = (self.omega_mes_g_front + self.omega_mes_g_back + self.omega_mes_d_front + self.omega_mes_d_back) / 2 * r / 2 * 2 * PI / (2400 * rapport_codeur) 
      		vth_msg.data = (self.omega_mes_d_front + self.omega_mes_d_back - self.omega_mes_g_front - self.omega_mes_g_back) / 2 * r / (chi * e) * 2 * PI / (2400 * rapport_codeur)
     		self.vx_pub.publish(vx_msg)
     		self.vth_pub.publish(vth_msg)
    
    
    if __name__ == '__main__':
    
    	try:
            	roboclaw_node = RoboclawNode()
            	roboclaw_node.runRoboClawNode()
    
        	except rospy.ROSInterruptException:
           	 	pass
