    \ customizations can just be added to the python logger object directly. To get\
    \ this level of customization for how the log files are written, you should write\
    \ your own log handler that would probably inherit from the [FileHandler](https://docs.python.org/3/library/logging.handlers.html#filehandler)\
    \ class.\n\nYour code would look something like this:\n\n    #!/usr/bin/env python\n\
    \    \n    import rospy\n    import logging\n    \n    rospy.init_node('log_test')\n\
    \    \n    logger = logging.getLogger(\"rosout\")\n    \n    class CustomFileHandler(logging.FileHandler):\n\
    \        def emit(self, record):\n            # handle logging records, I would\
    \ check the python documentation linked above\n        def close(self):\n    \
    \        # close the file\n    \n    logger.addHandler(CustomFileHandler('path/to/file.log'))"
