    \ after connection, otherwise the server will close the connection. See http://robotwebtools.org/jsdoc/roslibjs/current/Ros.html\n\
    \n    authenticate(mac, client, dest, rand, t, level, end)\n    Sends an authorization\
    \ request to the server.\n    Parameters:\n    Name \tType \tDescription\n   \
    \ mac \t\tMAC (hash) string given by the trusted source.\n    client \t\tIP of\
    \ the client.\n    dest \t\tIP of the destination.\n    rand \t\tRandom string\
    \ given by the trusted source.\n    t \t\tTime of the authorization request.\n\
    \    level \t\tUser level as a string given by the client.\n    end \t\tEnd time\
    \ of the client's session."

      var ros = new ROSLIB.Ros({
        url : 'ws://127.0.0.1:9090'

        console.log('Connected to websocket server.');

      });
