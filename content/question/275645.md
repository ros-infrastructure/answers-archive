 Hello,

I'm using `roscpp` to create ROS node and found that various callback methods for topic subscription are possible.
http://wiki.ros.org/roscpp/Overview/Publishers%20and%20Subscribers#Callback_Signature

I wonder which order is called on these methods whose names are the same.
I did a brief experiment by writing the node as followings:

    #include <ros/ros.h>
    #include <std_msgs/String.h>
    
    class Foo
    {
    public:
      Foo(): nh_("") {
        sub_ = nh_.subscribe("input", 1, &Foo::callback, this);
      }
      ~Foo(){}
    
      void callback(const std_msgs::String::ConstPtr&) {
        ROS_WARN_STREAM("callback1");
      }
      void callback(std_msgs::String::ConstPtr){
        ROS_WARN_STREAM("callback2");
      }
      void callback(const std_msgs::String&){
        ROS_WARN_STREAM("callback3");
      }
      void callback(std_msgs::String){
        ROS_WARN_STREAM("callback4");
      }
      void callback(const ros::MessageEvent<std_msgs::String const>&){
        ROS_WARN_STREAM("callback5");
      }
    
      ros::NodeHandle nh_;
      ros::Subscriber sub_;
    };
    
    int main(int argc, char** argv) {
      ros::init(argc, argv, "sandbox");
    
      Foo n = Foo();
    
      ros::spin();
    }

And then executed:

    $ rosrun sandbox sandbox
    [ WARN] [1510473407.806693318] [/sandbox:ros.sandbox]: callback1

The result is only the first callback was called.
I'd like to make it be called `callback5` where `ros::MessageEvent` is used first and if this is not defined then `callback1` is called.
Is it possible?

Thank you very much in advance!
