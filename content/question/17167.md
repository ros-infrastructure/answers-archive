    \   \nI managed, with some small corrections, to achieve this working code  :\
    \ \n    \n      double heightMapZ(double x, double y)  {\n\n         // getting\
    \ the pointer to the HeightmapShape\n         private: physics::WorldPtr world\
    \ = this->model->GetWorld();\n         physics::ModelPtr model = world->GetModel(\"\
    worldHightmap\");\n         physics::CollisionPtr collision = model->GetLink(\"\
    link\")->GetCollision(\"collision\");\n         physics::HeightmapShapePtr heightmap\
    \ = boost::dynamic_pointer_cast<physics::HeightmapShape>(collision->GetShape());\n\
    \n\n         // coordinate transform from regular Word (x,y) to the HeightmapShape\
    \ (index_x,index_y) \n         math::Vector3 size = this->heightmap->GetSize();\
    \ \n         math::Vector2i vc = this->heightmap->GetVertexCount();\n        \
    \ int index_x = (  ( (x + size.x/2)/size.x ) * vc.x - 1 ) ;\n         int index_y\
    \ = (  ( (-y + size.y/2)/size.y ) * vc.y - 1 ) ;\n\n         //  getting the height\
    \ :\n         double z =  this->heightmap->GetHeight( index_x , index_y ) ; \n\
    \     \n         return z;\n         }"
