    \ setting a bool parameter on a remote node:\n\n      void set_bool_param(std::string\
    \ param, bool value)\n      {\n        if (!param_client_) {\n          param_client_\
    \ = std::make_shared<rclcpp::AsyncParametersClient>(this, \"remote_node\");\n\
    \        }\n    \n        if (param_client_->service_is_ready()) {\n         \
    \ RCLCPP_INFO_STREAM(get_logger(), \"set \" << param << \" to \" << value);\n\
    \          param_client_->set_parameters({rclcpp::Parameter(param, value)},\n\
    \            [this](std::shared_future<std::vector<rcl_interfaces::msg::SetParametersResult>>\
    \ future)\n            {\n              future.wait();\n              auto results\
    \ = future.get();\n              if (results.size() != 1) {\n                RCLCPP_ERROR_STREAM(get_logger(),\
    \ \"expected 1 result, got \" << results.size());\n              } else {\n  \
    \              if (results[0].successful) {\n                  RCLCPP_INFO(get_logger(),\
    \ \"success\");\n                } else {\n                  RCLCPP_ERROR(get_logger(),\
    \ \"failure\");\n                }\n              }\n            });\n       \
    \ } else {\n          RCLCPP_ERROR(get_logger(), \"remote parameter server is\
    \ not ready\");\n        }\n      }\n\n"
