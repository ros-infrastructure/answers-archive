  body_markdown: "The gazebo::msgs::JointWrench has changed. \n\nFor force:\n\n  \
    \  contacts.contact(i).wrench(j).body_1_wrench().force()\n\nand for torque:\n\n\
    \    contacts.contact(i).wrench(j).body_1_wrench().torque()"
