  body_markdown: "I write a python script that subscribe to `/image_raw/compressed`\
    \ and save the video as `output_video.mp4`. here is the script  \n\n```\nimport\
    \ rclpy\nfrom rclpy.node import Node\nfrom sensor_msgs.msg import CompressedImage\n\
    import cv_bridge\nimport cv2\nimport numpy as np\n\n\nclass ImageToVideoConverter(Node):\n\
    \    def __init__(self):\n        super().__init__('image_to_video_converter')\n\
    \        self.bridge = cv_bridge.CvBridge()\n        self.subscription = self.create_subscription(\n\
    \            CompressedImage,\n            '/image_raw/compressed',\n        \
    \    self.image_callback,\n            10\n        )\n        self.video_writer\
    \ = None\n\n\n    def image_callback(self, msg):\n        try:\n            np_arr\
    \ = np.frombuffer(msg.data, np.uint8)\n            cv_image = cv2.imdecode(np_arr,\
    \ cv2.IMREAD_COLOR)\n            if self.video_writer is None:\n             \
    \   self.init_video_writer(cv_image)\n            self.video_writer.write(cv_image)\n\
    \        except Exception as e:\n            self.get_logger().error('Error processing\
    \ image: %s' % str(e))\n\n    def init_video_writer(self, image):\n        try:\n\
    \            height, width, _ = image.shape\n            video_format = 'mp4'\
    \  # or any other video format supported by OpenCV\n            video_filename\
    \ = 'output_video.' + video_format\n            fourcc = cv2.VideoWriter_fourcc(*'mp4v')\n\
    \            fps = 30  # Frames per second\n            self.video_writer = cv2.VideoWriter(video_filename,\
    \ fourcc, fps, (width, height))\n        except Exception as e:\n            self.get_logger().error('Error\
    \ initializing video writer: %s' % str(e))\n\n    def destroy_node(self):\n  \
    \      if self.video_writer is not None:\n            self.video_writer.release()\n\
    \        super().destroy_node()\n\n\ndef main(args=None):\n    rclpy.init(args=args)\n\
    \    image_to_video_converter = ImageToVideoConverter()\n    rclpy.spin(image_to_video_converter)\n\
    \    image_to_video_converter.destroy_node()\n    rclpy.shutdown()\n\n\nif __name__\
    \ == '__main__':\n    main()\n\n```\n"
