    \ with Husky, currently is an infinite loop, but you can modify accordingly:\n\
    \nLaunch the following in separate terminals:\n\n    $ roslaunch husky_gazebo\
    \ husky_empty_world\n    $ roslaunch husky_viz view_robot.launch\n    $ roslaunch\
    \ husky_navigation move_base_mapless_demo.launch\n\nRun the following python program:\
    \ `navigation_husky.py`\n\n    #! /usr/bin/env python3\n    \n    import rospy\n\
    \    import actionlib\n    from move_base_msgs.msg import MoveBaseAction, MoveBaseResult,\
    \ MoveBaseGoal\n    \n    class MoveHusky(object):\n      def __init__(self):\n\
    \        self.client = actionlib.SimpleActionClient('/move_base', MoveBaseAction)\n\
    \        self.client.wait_for_server()\n        self.goal = MoveBaseGoal()\n \
    \       \n      def moveToDest(self, x, y):\n        self.goal.target_pose.header.frame_id\
    \ = \"odom\"\n        self.goal.target_pose.pose.position.x = x\n        self.goal.target_pose.pose.position.y\
    \ = y\n        self.goal.target_pose.pose.orientation.z = 0.7071\n        self.goal.target_pose.pose.orientation.w\
    \ = 0.7071\n        rate = rospy.Rate(10)\n        self.client.send_goal(self.goal)\n\
    \        result = self.client.get_result()\n        while result == None:\n  \
    \        result = self.client.get_result()\n          status = self.client.get_state()\n\
    \          rospy.loginfo(\"Status: {}\".format(status))\n          rospy.loginfo(\"\
    Result: {}\".format(result))\n          rate.sleep()\n    \n    if __name__ ==\
    \ \"__main__\":\n      rospy.init_node(\"move_base_client\")\n      moveHusky\
    \ = MoveHusky()\n    \n      while not rospy.is_shutdown():\n        moveHusky.moveToDest(-4.0,\
    \ 0.0)\n        moveHusky.moveToDest(-4.0, -2.0)\n        moveHusky.moveToDest(0.0,\
    \ 0.0)\n\n"
