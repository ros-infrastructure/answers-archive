    \ with groovy-libg2o, I found out that in /opt/ros/groovy/share/libg2o the folder\
    \ cmake is missing.\nI copied the cmake folder from my fuerte distro and it worked.\
    \ \nYou can get the relevant files here: \n\n> www.dropbox.com/s/l0q4oc7yuk00t92/if_g2o_is_missed.zip\n\
    \nIn addition, my compiler had problems finding the cs.h.\nTherefore I used the\
    \ FindSuiteSparse.cmake script (It is included in if_g2o_is_missed.zip).\n\nInside\
    \ CMakeLists.txt I handled libg2o like a common catkin-package and added this:\n\
    \n    find_package(SuiteSparse)\n    if(CSPARSE_FOUND)\n      message(STATUS \"\
    CSPARSE_IS_FOUND\")\n    else(CSPARSE_FOUND)\n      message(STATUS \"CSPARSE_NOTFOUND\"\
    )\n    endif(CSPARSE_FOUND)\n    \n    FIND_PACKAGE(libg2o REQUIRED)\n\n    include_directories(\n\
    \      include\n      ${catkin_INCLUDE_DIRS}\n      ${CSPARSE_INCLUDE_DIR}\n \
    \     ${CHOLMOD_INCLUDE_DIR}\n      ${libg2o_INCLUDE_DIRS}\n    )\n\n     target_link_libraries(node\n\
    \       ${catkin_LIBRARIES}\n       ${CHOLMOD_LIBRARIES}\n       ${CSPARSE_LIBRARIES}\n\
    \        /usr/lib/libcxsparse.so # not included in findSUITESPARSE cmake macro\n\
    \       ${libg2o_LIBRARIES}\n     )\n\nMaybe a better solution would be to create\
    \ a FindG2O.cmake\nHope this helps."

  body_markdown: "I know this is an old thread but I found a solution on the rgbd-slamv2\
    \ code (http://felixendres.github.io/rgbdslam_v2/)\n\nIndeed you need a create\
    \ a FindG2O.cmake file and put the code below. \n\n    # Locate the g2o libraries\n\
    \    # A general framework for graph optimization.\n    #\n    # This module defines\n\
    \    # G2O_FOUND, if false, do not try to link against g2o\n    # G2O_LIBRARIES,\
    \ path to the libg2o\n    # G2O_INCLUDE_DIR, where to find the g2o header files\n\
    \    #\n    # Niko Suenderhauf <niko@etit.tu-chemnitz.de>\n    # Adapted by Felix\
    \ Endres <endres@informatik.uni-freiburg.de>\n    \n    IF(UNIX)\n    \n     \
    \ IF(G2O_INCLUDE_DIR AND G2O_LIBRARIES)\n        # in cache already\n        SET(G2O_FIND_QUIETLY\
    \ TRUE)\n      ENDIF(G2O_INCLUDE_DIR AND G2O_LIBRARIES)\n    \n      MESSAGE(STATUS\
    \ \"Searching for g2o ...\")\n      FIND_PATH(G2O_INCLUDE_DIR\n        NAMES core\
    \ math_groups types\n        PATHS /usr/local /usr\n        PATH_SUFFIXES include/g2o\
    \ include)\n    \n      IF (G2O_INCLUDE_DIR)\n        MESSAGE(STATUS \"Found g2o\
    \ headers in: ${G2O_INCLUDE_DIR}\")\n      ENDIF (G2O_INCLUDE_DIR)\n    \n   \
    \   FIND_LIBRARY(G2O_CORE_LIB             \n        NAMES g2o_core \n        PATHS\
    \ /usr/local /usr \n        PATH_SUFFIXES lib)\n      FIND_LIBRARY(G2O_STUFF_LIB\
    \            \n        NAMES g2o_stuff \n        PATHS /usr/local /usr \n    \
    \    PATH_SUFFIXES lib)\n      FIND_LIBRARY(G2O_TYPES_SLAM3D_LIB     \n      \
    \  NAMES g2o_types_slam3d \n        PATHS /usr/local /usr \n        PATH_SUFFIXES\
    \ lib)\n      FIND_LIBRARY(G2O_SOLVER_CHOLMOD_LIB   \n        NAMES g2o_solver_cholmod\
    \ \n        PATHS /usr/local /usr \n        PATH_SUFFIXES lib)\n      FIND_LIBRARY(G2O_SOLVER_PCG_LIB\
    \       \n        NAMES g2o_solver_pcg \n        PATHS /usr/local /usr \n    \
    \    PATH_SUFFIXES lib)\n      FIND_LIBRARY(G2O_SOLVER_CSPARSE_LIB   \n      \
    \  NAMES g2o_solver_csparse \n        PATHS /usr/local /usr \n        PATH_SUFFIXES\
    \ lib)\n      FIND_LIBRARY(G2O_INCREMENTAL_LIB      \n        NAMES g2o_incremental\
    \ \n        PATHS /usr/local /usr \n        PATH_SUFFIXES lib)\n      FIND_LIBRARY(G2O_CSPARSE_EXTENSION_LIB\n\
    \        NAMES g2o_csparse_extension\n        PATHS /usr/local /usr\n        PATH_SUFFIXES\
    \ lib)\n    \n      SET(G2O_LIBRARIES ${G2O_CSPARSE_EXTENSION_LIB}\n         \
    \               ${G2O_CORE_LIB}           \n                        ${G2O_STUFF_LIB}\
    \          \n                        ${G2O_TYPES_SLAM3D_LIB}   \n            \
    \            ${G2O_SOLVER_CHOLMOD_LIB} \n                        ${G2O_SOLVER_PCG_LIB}\
    \     \n                        ${G2O_SOLVER_CSPARSE_LIB} \n                 \
    \       ${G2O_INCREMENTAL_LIB}                        \n                     \
    \   )\n     \n      IF(G2O_LIBRARIES AND G2O_INCLUDE_DIR)\n        SET(G2O_FOUND\
    \ \"YES\")\n        IF(NOT G2O_FIND_QUIETLY)\n          MESSAGE(STATUS \"Found\
    \ libg2o: ${G2O_LIBRARIES}\")\n        ENDIF(NOT G2O_FIND_QUIETLY)\n      ELSE(G2O_LIBRARIES\
    \ AND G2O_INCLUDE_DIR)\n        IF(NOT G2O_LIBRARIES)\n          IF(G2O_FIND_REQUIRED)\n\
    \            message(FATAL_ERROR \"Could not find libg2o!\")\n          ENDIF(G2O_FIND_REQUIRED)\n\
    \        ENDIF(NOT G2O_LIBRARIES)\n    \n        IF(NOT G2O_INCLUDE_DIR)\n   \
    \       IF(G2O_FIND_REQUIRED)\n            message(FATAL_ERROR \"Could not find\
    \ g2o include directory!\")\n          ENDIF(G2O_FIND_REQUIRED)\n        ENDIF(NOT\
    \ G2O_INCLUDE_DIR)\n      ENDIF(G2O_LIBRARIES AND G2O_INCLUDE_DIR)\n    \n   \
    \ ENDIF(UNIX)\n\nAlso make module and make sure you add it to your CMAKE_MODULE_PATH\
    \ env variable in cmake. (you can use this line \n\n    set(CMAKE_MODULE_PATH\
    \ ${CMAKE_MODULE_PATH} \"${CMAKE_CURRENT_SOURCE_DIR}/cmake-modules/\")\n\nI hope\
    \ this helps. "
