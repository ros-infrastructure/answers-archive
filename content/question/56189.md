    I got it working with this:

         if(NOT OpenCV_DIR)
         find_path(OpenCV_DIR "OpenCVConfig.cmake" DOC "Root directory of OpenCV" HINTS    "/opt/ros/fuerte/share/OpenCV/")
        endif(NOT OpenCV_DIR)
        FIND_PACKAGE(OpenCV REQUIRED)
