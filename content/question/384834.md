>  <node pkg="slam_toolbox" type="merge_maps_kinematic" name="merge_maps_kinematic" output="screen"/>

and this config

odom_frame: base_link

map_frame: map

base_frame: base_link

scan_topic: /scan

mode: mapping #localization
