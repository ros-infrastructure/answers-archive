    \ sync policies. To handle your case you would have to implement a new sync policy\
    \ https://github.com/ros/ros_comm/tree/melodic-devel/utilities/message_filters/include/message_filters/sync_policies\n\
    \nA simpler way would probably to only sync the two high frequency topics, buffer\
    \ messages on the third topic and dequeue messages from the buffer in the two-topic\
    \ callback, with some timestamp checking of course.\n\nEdit: Here is some pseudocode\
    \ to get you started but you need to put some effort into solving your problem\
    \ instead of searching for an existing implementation. Please mark this question\
    \ as answered if this has helped.\n\n    void processTwo(const T1 &t1, const T2\
    \ &t2);\n    void processThree(const T1 &t1, const T2 &t2, const T3 &t3);\n  \
    \  \n    std::deque<T3> buffer_t3;\n    const double sync_tolerance_seconds =\
    \ 0.1;  // Some window of tolerance\n    \n    void callbackTwoTopics(const T1\
    \ &t1, const T2 &t2) {\n      if (buffer_t3.empty()) { \n        // You could\
    \ also do timestamp checks here and decide to buffer your \n        // messages\
    \ and wait for t3\n        processTwo(t1, t2);\n        return;\n      }\n   \
    \ \n      auto&& t3 = buffer_t3.front();\n      ros::Time&& stamp1 = t1.header.stamp;\n\
    \      ros::Time&& stamp3 = t3.header.stamp;\n      double time_diff_seconds =\
    \ stamp1.toSeconds() - stamp3.toSeconds();\n      if (abs(time_diff_seconds) <\
    \ sync_tolerance_seconds) {\n        // t3 is usable\n        buffer_t3.pop_front();\n\
    \        processThree(t1, t2, t3);\n      } else if (time_diff_seconds > 0) {\n\
    \        // t3 is too far in the future, leave it in the buffer for later\n  \
    \    } else if (time_diff_seconds < 0) {\n        // t3 is too far in the past,\
    \ drop it\n        buffer_t3.pop_front();\n      }\n    }\n    \n    void callbackOtherTopic(const\
    \ T3 &t3) {\n      buffer_t3.push_back(t3);\n      process();\n    }"
