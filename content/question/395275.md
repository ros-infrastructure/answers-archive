Hello,

I am trying to write a rosservice that constantly waiting for a request for path planning. The client node will send out a desired end-effector Cartesian position to the service node. The service node will then take the position and perform path planning internally, as well as executing the control. 

My code looks like the following:

    #include "ros/ros.h"
    #include <moveit/move_group_interface/move_group_interface.h>
    #include <moveit/planning_scene_interface/planning_scene_interface.h>
    #include <ar3_pick/pick_up_pose.h>

    class MoveGripper{
      private:
    moveit::planning_interface::MoveGroupInterface &move_group_;
      public:

    // Constructor
    MoveGripper(moveit::planning_interface::MoveGroupInterface &move_group):
    move_group_(move_group_)
    {

    }
    
    bool move_to_picking_location(ar3_pick::pick_up_pose::Request &req,
                            ar3_pick::pick_up_pose::Response &res);
   
    };


    bool MoveGripper::move_to_picking_location(ar3_pick::pick_up_pose::Request &req,
                            ar3_pick::pick_up_pose::Response &res){
    // Desired End-effector location
    geometry_msgs::Pose target_pose;
    target_pose.orientation.w = 0.5;
    target_pose.orientation.x = -0.5;
    target_pose.orientation.y = -0.5;
    target_pose.orientation.z = 0.5;
    target_pose.position.x = req.x;
    target_pose.position.y = req.y;
    target_pose.position.z = req.z;
    ROS_INFO("Received end-effector position %f,%f,%f", target_pose.position.x,target_pose.position.y,target_pose.position.z);

    move_group_.setPoseTarget(target_pose);

    moveit::planning_interface::MoveGroupInterface::Plan my_plan;
    
    bool success = (move_group_.plan(my_plan) == moveit::planning_interface::MoveItErrorCode::SUCCESS);
    
    ROS_INFO_NAMED("tutorial", "Visualizing plan 1 (pose goal) %s", success ? "" : "FAILED");
    move_group_.move();

    return true;
    }


    int main(int argc, char **argv)
    {
      ros::init(argc, argv, "ar3_pick");
      ros::NodeHandle node_handle;
      ros::AsyncSpinner spinner(0);
      spinner.start();

  
      const std::string planning_group = "manipulator";
      moveit::planning_interface::MoveGroupInterface move_group(planning_group);
      MoveGripper move_gripper_obj(move_group);

      // Raw pointers are frequently used to refer to the planning group for improved performance.
      const robot_state::JointModelGroup* joint_model_group =
        move_group.getCurrentState()->getJointModelGroup(planning_group);

      ros::ServiceServer service = node_handle.advertiseService("pick_up_pose", &MoveGripper::move_to_picking_location,&move_gripper_obj);                                     
      ROS_INFO("Ready to move to pick up location");
      //ros::spin();
      ros::waitForShutdown();
      return 0;
     }

I have defined a custom ROS srv `pick_up_pose` that contains `float64 x,
float64 y,float64 z` for describing the desired end-effector position. However, the planner did not execute and the robot did not move in the simulator. On the client side, I have the following code that sends out the end-effector position to the service node:

    #include "ros/ros.h"
    #include <ar3_pick/pick_up_pose.h>
    #include <stdlib.h>


    int main(int argc, char **argv)
    {
        ros::init(argc, argv, "send_target_pose_client");
  
      ros::NodeHandle n;
      ros::ServiceClient client = n.serviceClient<ar3_pick::pick_up_pose>("pick_up_pose");
      ar3_pick::pick_up_pose srv;
  
      srv.request.x = 0;
      srv.request.y = -0.4;
      srv.request.z = 0.3;


      if (client.call(srv))
      {
        ROS_INFO("Requesting robot arm to move to target position");
      }
      else
      {
        ROS_ERROR("Failed to request robot arm to move to target position");
        return 1;
      }

      return 0;

    }

To debug the problem, I made sure the service node did receive the `srv.request.x = 0; srv.request.y = -0.4; srv.request.z = 0.3;` from this line of code `ROS_INFO("Received end-effector position %f,%f,%f", target_pose.position.x,target_pose.position.y,target_pose.position.z);` in my service node. However, I still receive the promt `"Failed to request robot arm to move to target position"` because  `client.call(srv)` is `False` for some reasons. In addition, the service node automatically quit after receiving the request. I could not confirm whether the Moveit ran correctly. To further identify the issue, I manually tried to print some random message using`ROS_INFO` before and after `move_group_.setPoseTarget(target_pose);`. The code looks like this

    move_group_.setPoseTarget(target_pose);

    ROS_INFO("I ran here 2");
