    \ simply save the latest message. Then use a `rospy.Timer` with a separate function\
    \ that calls your processing code with the latest message on a fixed schedule.\n\
    \n    class Node:\n        ...\n        self.subscriber = rospy.Subscriber(sub_topic,\
    \ MessageType, self.sub_callback)\n        self.publisher = rospy.Publisher(pub_topic,\
    \ MessageType)\n        rospy.Timer(rospy.Duration(1), self.timer_callback)\n\
    \    \n        def sub_callback(self, message):\n            self.latest_message\
    \ = message\n    \n        def timer_callback(self, _):\n            self.processing_code(self.latest_message)\n\
    \    \n        def processing_code(self, message):\n            ...\n        \
    \    self.publisher.pub(processed_message)"
