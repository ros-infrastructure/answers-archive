    import rospy

    from std_srvs.srv import Empty
    
    driver = ModbusTcpClient("ip", port)
    
    def srv_cb(req):

        driver.close()

        node_name = 'node'
        rospy.init_node(node_name)
    
        serv  = rospy.Service('srv', Empty, srv_cb)
