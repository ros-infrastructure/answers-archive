  body_markdown: "The first answer to [Significance of ros::spinOnce()](http://answers.ros.org/question/11887/significance-of-rosspinonce/)\
    \ will probably help you. Also: [roscpp/Overview/Callbacks and Spinning](http://wiki.ros.org/roscpp/Overview/Callbacks%20and%20Spinning)\
    \ from `roscpp`.\n\n---\n\nEdit:\n\n    int main(int argc, char** argv)\n    {\
    \   \n        ...\n\n        ros::spin();\n    \n        return 0;\n    }\n\n\
    The `ros::spin()` will never return, and is probably why your other callbacks\
    \ aren't being serviced. As you can read in the [Significance of ros::spinOnce()](http://answers.ros.org/question/11887/significance-of-rosspinonce/)\
    \ answer, replacing it with something like:\n\n    int main(int argc, char** argv)\n\
    \    {\n        ...\n\n        ros::Rate loop_rate(10);\n\n        while (ros::ok()\
    \ && smart_ray_ok())\n        {\n            // let the smart ray event loop do\
    \ its thing\n            process_smart_ray_events()\n        \n            //\
    \ let ros do its things\n            ros::spinOnce();\n\n            // sleep\
    \ some (only if needed)\n            loop_rate.sleep();\n        }\n\nWill give\
    \ the smart ray SDK a chance to process its incoming data (and call the registered\
    \ *UserCB*).\n\nAs I have no experience with that particular SDK, I cannot say\
    \ what `smart_ray_ok()` and `process_smart_ray_events()` should really be (or\
    \ if they even exist), that is something you'll have to figure out yourself.\n\
    \nAs an alternative to the above, you could use an `AsyncSpinner`, which would\
    \ remove the need for the `while` loop and the `ros::spinOnce()`. See the linked\
    \  [roscpp/Overview/Callbacks and Spinning](http://wiki.ros.org/roscpp/Overview/Callbacks%20and%20Spinning)\
    \ page for more information on that.\n"

    int CBstatusMsg (CAMDESC* cd, int msgType, int msgdata1, int msgdata2, char *msg)

    	int Laserlight;

    	int digIn[4];
    	int digOut[2];
    	int FPS;
    
    	if(msg==NULL) //No message data available
    		return -1;
    
    	if (msgType == MSGTYPE_CONNECTION) //Ethernet connection messages
    	{
    		if(msgdata1 == CAMERA_CONNECTED) // TCP/IP connection established
    		{
    			printf("%s.\n",msg);
    			sensor.connectionstate = STATE_CAMERA_CONNECTED;
    			return(0);
    		}
    		else // disconnected
    		{
    			printf("%s.\n",msg);
    			sensor.connectionstate = STATE_CAMERA_CLOSED;

    		}
    	}
    	else if (msgType == MSGTYPE_INFO) 	// Info messages

    		digIn[1] = cd->digio_in[1];

    int startSmartRayCapture(void)
    {
    	int res;
    
    	if (sensor.connectionstate != STATE_CAMERA_CONNECTED)
    	{
    		printf("The camera is not connected.\n");
    		return -1;
    	}
    
    	res = SR_API_StartCam (&sensor);
    	//check whether the sensor can start
    	if(res == 0)
    	{
    		printf("Camera Started.\n");
    		return 0;
    	}
    	else
    	{
    		printf("Error starting the sensor.\n");
    		return -2;
    	}
    }
    
    int stopSmartRaySensor(void)
    {
    	SR_API_StopCameraConnectionManagement(&sensor);
    	SR_API_StopCam(&sensor);
    	SR_API_Exit();
    	return(0);
    }
