    \ a couple of hours.\n\nI also encountered problems with rostopic hz because I\
    \ did not have a list of topic names beforehand.  Run the sample ros node below\
    \ to understand how to retrieve all of this data.  The *myhz()* method will run\

    \ *mygraph()* function will print all nodes and edges (with to and from information).\n\
    \n\n    \n\n    #!/usr/bin/env python import roslib; \n        roslib.load_manifest('ssnode')\
    \ import \n        rospy import roslib \n        import roslib.message\n     \
    \   import rosgraph\n        import rostopic \n        from rostopic import ROSTopicHz\
    \ \n        from rosgraph.impl import graph\n        \n        def mygraph():\n\
    \            g = graph.Graph()\n            try:\n                while 1:\n \
    \                 g.update()\n        \n                  if not g.nn_nodes and\
    \ not g.srvs:\n                      print(\"empty\")\n                  else:\n\
    \                      print('\\n')\n                  if g.nn_nodes:\n      \
    \                print('Nodes:')\n                      for n in g.nn_nodes:\n\
    \                          prefix = n + '|'\n                          print('\
    \  ' + n + ' :')\n                          print('    Inbound:')\n          \
    \                for k in g.nn_edges.edges_by_end.iterkeys():\n              \
    \                if k.startswith(prefix):\n                                  for\
    \ c in g.nn_edges.edges_by_end[k]:\n                                      print('\
    \      ' + c.start + ' label = '\n    + c.label)\n                          print('\
    \    Outbound:')\n                          for k in g.nn_edges.edges_by_start.iterkeys():\n\
    \                              if k.startswith(prefix):\n                    \
    \              for c in g.nn_edges.edges_by_start[k]:\n                      \
    \                print('      ' + c.end + ' label = ' + c.label)\n           \
    \       if g.srvs:\n                      print('Services:')\n               \
    \       for s in g.srvs:\n                          print('  ' + s)\n        \
    \    except KeyboardInterrupt:\n                pass\n                \n     \
    \            def myhz(topic, window_size=-1, filter_expr=None):\n            rospy.loginfo(\"\
    check if roscore is up\")\n            while !rospy.is_shutdown(): \t\tthegraph();\
    \ \t\tros::sleep(10);\n                return\n            rospy.init_node(\"\
    myname\", anonymous=True)\n            rospy.loginfo(\"registered node name\"\
    )\n            rt = ROSTopicHz(window_size, filter_expr=filter_expr)\n       \
    \     \n            # have to subscribe with topic_type\n            sub = rospy.Subscriber(topic,\
    \ rospy.AnyMsg, rt.callback_hz)        \n            print(\"subscribed to [%s]\"\
    %topic)\n            while not rospy.is_shutdown():\n                rospy.loginfo(rt.times)\n\
    \        \n        \n        if __name__ == '__main__':\n            try:\n  \
    \            mygraph()\n              #myhz(\"sometopic\")\n            except\
    \ rospy.ROSInterruptException: pass"
