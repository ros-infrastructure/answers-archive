  body_markdown: "Here is a simple example that subscribes to a geometry_msgs/Pose2D\
    \ topic, toggles a LED every time it receives a message and stores the data in\
    \ separate variables.\n\n    #include <ros.h>\n    #include <geometry_msgs/Pose2D.h>\n\
    \    \n    ros::NodeHandle  nh;\n    \n    void poseCb( const geometry_msgs::Pose2D&\
    \ pose_msg){\n      digitalWrite(13, HIGH-digitalRead(13));   // blink the led\n\
    \      float x = pose_msg.x;\n      float y = pose_msg.y;\n      float th = pose_msg.theta;\n\
    \    }\n    \n    ros::Subscriber<geometry_msgs::Pose2D> sub(\"pose\", &poseCb\
    \ );\n    \n    void setup()\n    { \n      pinMode(13, OUTPUT);\n      nh.initNode();\n\
    \      nh.subscribe(sub);\n    }\n    \n    void loop()\n    {  \n      nh.spinOnce();\n\
    \      delay(1);\n    }\n     "
