    Solved

    libraries to be linked are saved on /usr/lib/x86_64-linux-gnu/cmake

    there I found Qt5Xml library and add this entry to CMakelists,txt

    set(EXTERNAL_DEPENDS
      Qt5Widgets
      Qt5PrintSupport
      Qt5Multimedia
      #OpenCV
      Qt5Xml
        )

    set(EXTERNAL_DEPENDS_LIB
      Qt5::Widgets
      Qt5::PrintSupport
      Qt5::Core
      Qt5::Xml
       #${OpenCV_LIBS}
      )
