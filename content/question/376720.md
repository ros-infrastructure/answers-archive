    \n> # perform remap so both turtles listen to the same command topic\n    forward_turtlesim_commands_to_second_turtlesim_node\
    \ = Node(\n            package='turtlesim',\n            executable='mimic',\n\
    \            name='mimic',\n            remappings=[\n                ('/input/pose',\
    \ '/turtlesim1/turtle1/pose'),\n                ('/output/cmd_vel', '/turtlesim2/turtle1/cmd_vel'),\n\
    \            ]\n        )"
