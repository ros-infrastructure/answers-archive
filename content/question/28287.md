    def target_tracking(self):

                target.position.x,
                target.position.y,
                target.position.z])

            TargetDiff = self.target_pxyz - self.state_pxyz
            NormTargetDiff = np.linalg.norm(TargetDiff)

            TargetDiff *= SigmoidLike(NormTargetDiff, self.R_trg, self.d_trg)
            return (TargetDiff*self.freq)
        else:
            return np.zeros(3)
