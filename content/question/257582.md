    #include <iostream>
    #include <string>
    #include <sstream>
    
    #include <opencv2/core.hpp>
    #include <opencv2/highgui.hpp>
    #include <opencv2/videoio.hpp>
    
    using namespace cv;
    
    int main(int argc, char **argv) {
    
        VideoCapture cap(0);
        namedWindow("video", WINDOW_NORMAL);
        while (1) {
            Mat frame;
            cap >> frame;
            imshow("video", frame);
            if (waitKey(30) >= 0) break;
        }
        cap.release();
        return 0;
    }
