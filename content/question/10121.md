    \ nodelet with the correct     parameters. Here is a working launch file.\n\n\
    \    <launch>\n    \n      <group ns=\"/pcl_filters\">\n        <!-- PCL Manager\
    \ -->\n        <node pkg=\"nodelet\" type=\"nodelet\" name=\"box_filter_pcl_manager\"\
    \ args=\"manager\" output=\"screen\"/>\n    \n        <!-- Run a passthrough filter\
    \ to delimit in x direction -->\n        <node pkg=\"nodelet\" type=\"nodelet\"\
    \ name=\"psx\" args=\"load pcl/PassThrough box_filter_pcl_manager\" output=\"\
    screen\">\n          <remap from=\"~input\" to=\"/camera/depth/points\" />\n \
    \         <rosparam>\n            filter_field_name: x\n            filter_limit_min:\
    \ -.8\n            filter_limit_max: 0.5\n          </rosparam>\n        </node>\n\
    \    \n        <!-- Run a passthrough filter to delimit in y direction -->\n \
    \       <node pkg=\"nodelet\" type=\"nodelet\" name=\"psy\" args=\"load pcl/PassThrough\
    \ box_filter_pcl_manager\" output=\"screen\">\n          <remap from=\"~input\"\
    \ to=\"psx/output\" />\n          <rosparam>\n            filter_field_name: y\n\
    \            filter_limit_min: -1.\n            filter_limit_max: .3\n       \
    \   </rosparam>\n        </node>\n\n        <!-- Run a passthrough filter to delimit\
    \ in z direction -->\n        <node pkg=\"nodelet\" type=\"nodelet\" name=\"psz\"\
    \ args=\"load pcl/PassThrough box_filter_pcl_manager\" output=\"screen\">\n  \
    \        <remap from=\"~input\" to=\"psy/output\" />\n          <rosparam>\n \
    \           filter_field_name: z\n            filter_limit_min: 0.0\n        \
    \    filter_limit_max: 2.0\n          </rosparam>\n        </node>\n      </group>\n\
    \    </launch>\n"
