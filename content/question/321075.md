    sis.start()

    # Open the container
    with sm_top:

        smach.StateMachine.add('BAS', Bas(),

        sm_con = smach.Concurrence(outcomes=['outcome4','outcome5'],

        with sm_con:

        smach.StateMachine.add('CON', sm_con,
                               transitions={'outcome4':'CON',
                                            'outcome5':'outcome6'})

    outcome = sm_top.execute()
    rospy.spin()
    sis.stop()
