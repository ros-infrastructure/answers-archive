    \ argv, \"SEQUITUR_driver\");  \n\nros::NodeHandle nh; \n\nros::Publisher data_topic\
    \ = n.advertise<std_msgs::String>(\"data\", 1000);\n\ntry\n{\n\nser.setPort(\"\

    }\n\nros::Rate loop_rate(5)  \n\nwhile(ros::ok()){\n\nros::spinOnce();\n\nif(ser.available())\n\

    \nresult.data = ser.read(ser.available());\n\nROS_INFO_STREAM(\"Read: \" << result.data);\n\
    \ndata_topic.publish(result);\n\n}\nloop_rate.sleep();\n} \n\n}"
