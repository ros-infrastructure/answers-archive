Have 2 subscriber with different callback groups and while executing getting error as 'terminate called without an active exception'
ros2 version Galactic. Please find the code below

code:

    from  rclpy . callback_groups  import  MutuallyExclusiveCallbackGroup,  ReentrantCallbackGroup 
    import rclpy
    import threading
    from rclpy.node import Node
    from std_msgs.msg import Int32
    
    
    class Test(Node):
    
        def __init__(self ):
            super().__init__('Test')
            cb_group1 = MutuallyExclusiveCallbackGroup()
            cb_group2 = MutuallyExclusiveCallbackGroup()
            self.subscription = self.create_subscription(
                Int32,
                'topic1',
                self.callback1,
                qos_profile = 1,
                callback_group = cb_group1 )
            
            self.obstacle_subscription = self.create_subscription(
                    Int32,
                    'topic2',
                    self.callback2, 
                    qos_profile = 1,
                    callback_group = cb_group2)
            
    
        def callback1(self, msg):
            while True:
                self.get_logger().info('callback1')
    
        def callback2(self, msg):
            while True:
                self.get_logger().info('callback2')
    
    
    def main(args=None):
        rclpy.init(args=args)
        test_node = Test()
        executor = rclpy.executors.MultiThreadedExecutor()
        executor.add_node(test_node)
        spin_thread = threading.Thread(target=executor.spin, daemon=True)
        spin_thread.start()
        test_node.destroy_node()
        rclpy.shutdown()
    
    
    if __name__ == '__main__':
        main()
