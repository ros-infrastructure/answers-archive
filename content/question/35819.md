    \ `<geometry>`. \n\n    <link name=\"left_gripper\">\n      <visual>\n       \
    \ <geometry>\n          <mesh filename=\"package://pr2_description/meshes/gripper_v0/l_finger.dae\"\
    />\n        </geometry>\n      </visual>\n    </link>"
