        <arg name ="dir1" value="directory1"/>

        <node pkg="rosbag" type="record" name="recorder" args="-o $(arg dir1)/($arg foo)"/>

    and you launch this file like `roslaunch my_pkg my_launch.launch foo:=$(rosparam get foo)`.
