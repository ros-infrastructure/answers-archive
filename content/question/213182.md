  body_markdown: "`qt_build` can be replaced by using Qt in CMake. Here's an [example](https://github.com/allenh1/ros-qt-controller).\n\
    \n    cmake_minimum_required(VERSION 2.8.9)\n    project(ros_qt_controller)\n\
    \    find_package(catkin REQUIRED COMPONENTS roscpp)\n    find_package(Qt5Widgets\
    \ REQUIRED)\n    find_package(Qt5Core REQUIRED)\n    \n    catkin_package(\n \
    \     CATKIN_DEPENDS roscpp geometry_msgs\n      DEPENDS system-lib\n    )\n \
    \   \n    qt5_add_resources(QT_RESOURCES_CPP qt_resources/arrows.qrc)\n    qt5_wrap_cpp(QT_MOC_HPP\
    \ src/ControlWindow.h\n    \t\t\t\t\t\tsrc/RobotThread.h)\n    \n    include_directories(src\
    \ ${catkin_INCLUDE_DIRS})\n    include_directories(${Qt5Widgets_INCLUDE_DIRS})\n\
    \    \n    add_executable(qt_ros_ctrl src/ControlWindow.cpp\n    \t\t\t\t\t\t\
    \   src/RobotThread.cpp\n    \t\t\t\t\t\t   src/main.cpp\n    \t\t\t\t\t\t   ${QT_RESOURCES_CPP}\n\
    \                  \t\t\t   ${QT_MOC_HPP})\n    \n    target_link_libraries(qt_ros_ctrl\
    \ ${QT_LIBRARIES}\n                \t\t  ${catkin_LIBRARIES} Qt5::Widgets)\n \
    \   \n    install(TARGETS qt_ros_ctrl\n      RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}\n\
    \    )"

    \    # Find includes in corresponding build directories\n    set(CMAKE_INCLUDE_CURRENT_DIR\
    \ ON)\n    # Instruct CMake to run moc automatically when needed.\n    set(CMAKE_AUTOMOC\
    \ ON)\n\n    # Find the QtWidgets library\n    find_package(Qt5Widgets)\n\n  \
    \  # Tell CMake to create the helloworld executable\n    add_executable(helloworld\
    \ WIN32 main.cpp)\n\n    # Use the Widgets module from Qt 5.\n    target_link_libraries(helloworld\
    \ Qt5::Widgets)\n\nif you replace every 5 by a 4 you will get it build against\
