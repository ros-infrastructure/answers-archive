  body_markdown: "Here is a solution.\n\n1. Install pcl_conversion (sudo apt install\
    \ ros-eloquent-pcl-conversions)\n\n\nCMakeList\n\n    \n        find_package(PCL\
    \ 1.3 REQUIRED)\n        include_directories(${PCL_INCLUDE_DIRS})\n        link_directories(${PCL_LIBRARY_DIRS})\n\
    \        add_definitions(${PCL_DEFINITIONS})\n        target_link_libraries(\"\
    target\" ${PCL_LIBRARIES})\n\nSubscriber Callback\n   \n     void callback_depth_image_d435\
    \ (const sensor_msgs::msg::PointCloud2::SharedPtr point_cloud2_msgs)\n     {\n\
    \  \n       pcl::PointCloud<pcl::PointXYZ> point_cloud;\n       pcl::fromROSMsg(*point_cloud2_msgs,\
    \ point_cloud);     \n\n       BOOST_FOREACH(const pcl::PointXYZ& pt, point_cloud.points)\n\
    \       {          \n           std::cout  << \"x: \" << pt.x <<\"\\n\";\n   \
    \        std::cout  << \"y: \" << pt.y <<\"\\n\";\n           std::cout  << \"\
    z: \" << pt.z <<\"\\n\";\n           std::cout << \"---------\" << \"\\n\";\n\
    \       }\n     }"
