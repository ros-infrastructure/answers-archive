    int i;
    void point_cloud_callback(const sensor_msgs::PointCloud2& input_cloud)
    {
        pcl::PointCloud<pcl::PointXYZRGB>::Ptr point_cloud (new pcl::PointCloud<pcl::PointXYZRGB>);
        pcl::fromROSMsg(input_cloud, *point_cloud);
        pcl::io::savePCDFile("scene" + std::to_string(i) + ".pcd", *point_cloud);
