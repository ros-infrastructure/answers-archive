    \ basic Python image subscriber & saver: https://gist.github.com/rethink-imcmahon/77a1a4d5506258f3dc1f\n\
    \nIt creates a ROS subscriber, saves a jpeg on image callback, and then overwrites\
    \ it on every subsequent callback invocation (~15 Hz for Baxter's cameras).\n\n\
    The important takeaway is the flow from ROS Image -> CvBridge Converter -> OpenCV2\
    \ -> JPEG file:\n\n    #! /usr/bin/python\n    # Copyright (c) 2015, Rethink Robotics,\
    \ Inc.\n     \n    # Using this CvBridge Tutorial for converting\n    # ROS images\
    \ to OpenCV2 images\n    # http://wiki.ros.org/cv_bridge/Tutorials/ConvertingBetweenROSImagesAndOpenCVImagesPython\n\
    \     \n    # Using this OpenCV2 tutorial for saving Images:\n    # http://opencv-python-tutroals.readthedocs.org/en/latest/py_tutorials/py_gui/py_image_display/py_image_display.html\n\
    \    \n    # rospy for the subscriber\n    import rospy\n    # ROS Image message\n\
    \    from sensor_msgs.msg import Image\n    # ROS Image message -> OpenCV2 image\
    \ converter\n    from cv_bridge import CvBridge, CvBridgeError\n    # OpenCV2\
    \ for saving an image\n    import cv2\n \n    # Instantiate CvBridge\n    bridge\
    \ = CvBridge()\n    \n    def image_callback(msg):\n        print(\"Received an\
    \ image!\")\n        try:\n            # Convert your ROS Image message to OpenCV2\n\
    \            cv2_img = bridge.imgmsg_to_cv2(msg, \"bgr8\")\n        except CvBridgeError,\
    \ e:\n            print(e)\n        else:\n            # Save your OpenCV2 image\
    \ as a jpeg \n            cv2.imwrite('camera_image.jpeg', cv2_img)\n     \n \
    \   def main():\n        rospy.init_node('image_listener')\n        # Define your\
    \ image topic\n        image_topic = \"/cameras/left_hand_camera/image\"\n   \
    \     # Set up your subscriber and define its callback\n        rospy.Subscriber(image_topic,\
    \ Image, image_callback)\n        # Spin until ctrl + c\n        rospy.spin()\n\
    \ \n    if __name__ == '__main__':\n        main()"
