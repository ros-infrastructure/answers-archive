 As https://github.com/ament/ament_lint/issues/132 said, it is possible to exclude specific checkers by setting <package>_FOUND to TRUE. But I don't understand how it works. When looking into the source code of `ament_lint_auto_find_test_dependencies()`, I only found serveral lines related to `<package>_FOUND`:

    # try to find_package() all test dependencies
    foreach(_dep ${${PROJECT_NAME}_TEST_DEPENDS})
      find_package(${_dep} QUIET)
      if(${_dep}_FOUND)
        list(APPEND ${PROJECT_NAME}_FOUND_TEST_DEPENDS ${_dep})
      endif()
    endforeach()
