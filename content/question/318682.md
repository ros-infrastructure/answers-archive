    \ a tutorial for how to do this as well:  http://wiki.ros.org/dynamic_reconfigure/Tutorials\n\
    \nThis question and answer will help you a lot if you want to do it in C++. NOTE:\
    \ the push_back use in the answer code creates issue but the rest of it is good.\
    \ I used this as a start for my code below : https://answers.ros.org/question/12276/is-there-a-c-api-for-a-dynamic-reconfigure-client/\n\
    \nYou'll see move base has a service called \"/move_base/set_parameters\". That\
    \ is the service call you'll substitute into code below.\n\nAnd finally how I\
    \ do it in C++ for raspicam node. I am using integers. You'll need to use floats\
    \ or doubles for inflation radius.\n\n    #include <dynamic_reconfigure/DoubleParameter.h>\n\
    \    #include <dynamic_reconfigure/Reconfigure.h>\n    #include <dynamic_reconfigure/Config.h>\n\
    \    \n        dynamic_reconfigure::ReconfigureRequest srv_req;\n        dynamic_reconfigure::ReconfigureResponse\
    \ srv_resp; \n        dynamic_reconfigure::IntParameter integer_param;\n     \
    \   dynamic_reconfigure::Config conf;\n    \n        int cameraControl(int ss,\
    \ int iso)\n        {\n          conf.ints.resize(2);\n          ROS_INFO_STREAM(\"\
    ROBOT-STATE: cameraControl: shutter_speed: \" << ss << \"\\tISO : \" << iso);\n\
    \          integer_param.name = \"shutter_speed\";\n          integer_param.value\
    \ = ss;\n          conf.ints.at(0) = integer_param;\n    \n          integer_param.name\
    \ = \"ISO\";\n          integer_param.value = iso;\n          conf.ints.at(1)\
    \ = integer_param;\n          srv_req.config = conf;\n    \n          ros::service::call(\"\
    /raspicam_node/set_parameters\", srv_req, srv_resp);\n    \n          return 1;\n\
    \        }"
