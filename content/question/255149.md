    int main(int argc, char **argv)

       initscr();    
       printw("Hello world");  
       refresh();    
       getch();    
       endwin();   
       return 0;

    <build_depend>ncurses++</build_depend>

    pkg_check_modules ( ncurses++ REQUIRED ncurses++ )

    target_link_libraries(test_ncurses ${catkin_LIBRARIES}  ${ncurses++_LIBRARIES})
