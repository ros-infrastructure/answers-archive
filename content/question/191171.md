          mkdir ~/catkin-download
          cd ~/catkin-download
          git clone https://github.com/ros/catkin --branch groovy-devel
          cd catkin
          cmake -DCMAKE_INSTALL_PREFIX=/opt/ros/groovy ./
          make
          sudo make install
