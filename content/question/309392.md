    \ I'll put my code below, but in general my approach is to use the `buffer.lookup_transform`\
    \ to get the transform, AND the `broadcaster.sendTransform` to publish a transform.\
    \ \n\nWhat this does is let me read a rosbag, re-publish any `/tf` or `/tf_static`\
    \ messages, and then when it's time to convert frames I can use `rospy.Time.now()`\
    \ and get the \"current\" transform, where \"current\" is whatever the most recently\
    \ published transform is.\n\nYou might have an issue if your first data appears\
    \ *before* your first `/tf` or `/tf_static` message appears; if you find this\
    \ to be the case then you can optionally drop that data. For my personal use case,\
    \ it doesn't matter. I tried removing the try/catch in the hopes that it'd speed\
    \ up processing, but it didn't :(\n\nHere's the code, enjoy!\n\n    import argparse\n\
    \    import rosbag\n    import rospy\n    import os.path\n    import tf2_ros\n\
    \    from tf2_sensor_msgs.tf2_sensor_msgs import do_transform_cloud\n    \n  \
    \  parser = argparse.ArgumentParser(description='Converts all pointclouds to a\
    \ target frame.')\n    parser.add_argument('input', type=str, help='Input Rosbag')\n\
    \    parser.add_argument('output', type=str, help='Output Rosbag')\n    \n   \
    \ args = parser.parse_args()\n    \n    # Force input/output files to be .bag\
    \ files\n    if \".bag\" not in args.input:\n        print(\"Input file needs\
    \ to have a *.bag ending\")\n        exit()\n    if \".bag\" not in args.output:\n\
    \        print(\"Output file needs to have a *.bag ending\")\n        exit()\n\
    \    \n    # Does the input file exist?\n    if not os.path.isfile(args.input):\n\
    \        print(\"Input file cannot be found.\")\n        exit()\n    \n    # Need\
    \ to initialize a ROS node to do the conversion\n    rospy.init_node(\"converter\"\
    )\n    \n    # Setup the tf2 listener:\n    buffer = tf2_ros.Buffer()\n    listener\
    \ = tf2_ros.TransformListener(buffer)\n    \n    # Setup the tf2 broadcaster\n\
    \    broadcaster = tf2_ros.TransformBroadcaster()\n    \n    # What are you trying\
    \ to transform to?\n    target_frame = \"your_target_frame\"  # NOTE: You could\
    \ use this as an input argument instead, for more flexibility!\n    \n    # Transform\
    \ the data!\n    with rosbag.Bag(args.output, 'w') as output:\n        for topic,\
    \ msg, t in rosbag.Bag(args.input).read_messages():\n    \n            # Inputs\
    \ are copied as the basis for outputs\n            new_msg = msg\n    \n     \
    \       # Re-publish any /tf or /tf_static messages, so you can look those transforms\
    \ up later\n            if topic == \"/tf\":\n                for msg_tf in msg.transforms:\n\
    \                    broadcaster.sendTransform(msg_tf)\n            if topic ==\
    \ \"/tf_static\":\n                for msg_tf in msg.transforms:\n           \
    \         broadcaster.sendTransform(msg_tf)\n    \n            # For my use case,\
    \ I'm trying to transform PointCloud2 messages. You could switch on your own topic\
    \ here.\n            if hasattr(new_msg, 'cloud'):\n                try:\n   \
    \                 transform = buffer.lookup_transform(target_frame,\n        \
    \                                                msg.cloud.header.frame_id,\n\
    \                                                        rospy.Time.now(),\n \
    \                                                       rospy.Duration(0))\n \
    \                   new_msg.cloud = do_transform_cloud(msg.cloud, transform)\n\

    \ to \" + target_frame)\n                    # Choosing to continue here because\
