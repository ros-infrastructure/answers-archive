    \n----------\n\nOk, as promised, a simple `HelloWorld` example follows. Begin\
    \ with reading the [avr_bridge Wiki](http://www.ros.org/wiki/avr_bridge), if you\
    \ haven't yet done so. I'm also assuming you all dear readers already understand\
    \ what Arduino is (as I'm using an Arduino Duemilanove here) and have [Eclipse\
    \ configured](http://www.arduino.cc/playground/Code/Eclipse) to work with it.\
    \ \n\nTake `callResponse.yaml` from the `HelloWorld` example provided in the old\
    \ example project [here](http://www.ros.org/wiki/avr_bridge/Tutorials/Hello%20World),\
    \ for completness, these are it's contents:\n\n    port: /dev/ttyUSB0\n    name:\
    \ callResponse\n    publish: #list of published topics\n     response: #topic\
    \ name\n      type: std_msgs/String  #msg type\n    subscribe:\n     call:\n \
    \     type: std_msgs/String\n\nThen run\n\n    rosrun avr_bridge gen_avr.py callResponse.yaml\
    \ .\n\nin the project's folder where you wish to use `avr_bridge` (generally do\
    \ as it says in the [avr_bridge HelloWorld Wiki tutorial](http://www.ros.org/wiki/avr_bridge/Tutorials/Hello%20World).\
    \ I used the following `main.cpp` (a slight modification of the one provided in\
    \ the aforementioned tutorial):\n\n    #include \"WProgram.h\"\n    #include <stdio.h>\n\
    \    #include <String.h>\n    #include \"avr_ros/Ros.h\"\n    #include \"avr_ros/ros_string.h\"\
    \n    \n    Publisher resp;\n    \n    std_msgs::String call_msg;\n    std_msgs::String\
    \ response_msg;\n    \n    extern \"C\" void __cxa_pure_virtual()\n    {\n   \
    \   cli();\n      for (;;);\n    }\n    \n    void toggle()\n    {\n    \tif (digitalRead(13)\
    \ == LOW ) {\n    \t\tdigitalWrite(13, HIGH);   // set the LED on\n    \t} else\
    \ {\n    \t\tdigitalWrite(13, LOW);    // set the LED off\n    \t}\n    }\n  \
    \  \n    void responseCallback(Msg* msg) {\n    \ttoggle();\n    \n    \tresponse_msg.data.setString(call_msg.data.getRawString());\n\
    \    \tros.publish(resp, &response_msg);\n    }\n    \n    unsigned long pubTimer=0;\n\
    \    void setup(){\n    \t//Set up the serial communicaton and wait a bit to make\
    \ sure\n    \t//that the program and grab its id before we go into any other\n\
    \    \t//set up routines\n    \tSerial.begin(57600);\n    \tfor (int i; i<100;\
    \ i++) {ros.spin();delay(10);}\n    \n    \tpinMode(13, OUTPUT);\n    \tresp =\
    \ ros.advertise(\"response\");\n    \tros.subscribe(\"call\", responseCallback,\
    \ &call_msg);\n    \n    \tcall_msg.data.setMaxLength(30);\n    \tresponse_msg.data.setMaxLength(60);\n\
    \    \n    \tpubTimer = millis();\n    }\n    \n    void loop(){\n    \tros.spin();\n\
    \    \n    \t//Do other work\n    \tdelay(10);\n    }\n    \n    int main(void)\n\
    \    {\n    \tinit();\n    \tsetup();\n    \n    \tfor (;;) { loop(); }\n    \n\
    \    \treturn 0;\n    }\n\nand added those to `Ros.cpp`:\n\n    int ros_putchar(char\
    \ c, FILE *stream)\n    {\n    \tSerial.write(c);\n        return 0;\n    }\n\
