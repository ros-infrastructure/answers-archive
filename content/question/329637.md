        node.get_logger().info('Spawning shuttle with namespace ' + node.get_namespace() + " and name " + req.name )
    
        while not cli.wait_for_service(timeout_sec=3.0):

        future = cli.call_async(req)

        if future.result() is not None:

        node.destroy_node()
        rclpy.shutdown()
