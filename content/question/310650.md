    \ to process it in some way you might consider using the Octomap library or subscribing\
    \ to the octomap_point_cloud_centers message.  \nIn any case this is the type\
    \ of roslaunch i use when working with octomap and a kinect:\n\n    <?xml version=\"\
    1.0\" encoding=\"UTF-8\"?>\n    <launch>\n    \n      <node pkg=\"octomap_server\"\
    \ type=\"octomap_server_node\" name=\"octomap_server\">\n       <param name=\"\
    resolution\" value=\"0.05\"/>\n    \n        <!-- fixed map frame (set to 'map'\
    \ if SLAM or localization running!) -->\n        <param name=\"frame_id\" type=\"\
    string\" value=\"map\"/>\n    \n        <!-- maximum range to integrate (speedup!)\
    \ -->\n        <param name=\"sensor_model/max_range\" value=\"4.0\"/>\n      \
    \  <param name=\"base_frame_id \" value=\"base_link\" type=\"str\"/>\n    \n \
    \       <!-- data source to integrate (PointCloud2) -->\n        <remap from=\"\
    cloud_in\" to=\"/your/cloud/topic/here\"/>\n    \n      </node>\n    \n    </launch>\n\
