    \ ###\n    import rclpy\n    from rclpy.node import Node\n\n    def get_topic_list():\n\
    \        node_dummy = Node(\"_ros2cli_dummy_to_show_topic_list\")\n        topic_list\
    \ = node_dummy.get_topic_names_and_types()\n        node_dummy.destroy_node()\n\
    \        return topic_list\n\n\n    rclpy.init()\n    topic_list = get_topic_list()\n\
    \    for info in topic_list:\n        print(info[0])\n    rclpy.shutdown()\n\n\
