    \ class Node\n    {\n        Node (ros::NodeHandle& nh, ros::NodeHandle& nhPrivate)\n\
    \         // initialization list copy the arguments to the class attributes.\n\
    \        {\n          // create subscriber, publisher using nh_.\n        }\n\n\
    \        // always use shared pointer version of callbacks.\n        subscriberCb\
    \ (const sensor_msgs::ImageConstPtr& image) \n        {\n           imageIn_ =\
    \ image;\n        }\n\n        void process ()\n        {\n           // Use imageIn\
    \ as your input image.\n           sensor_msgs::ImageConstPtr imageIn = imageIn_;\
    \ // See note (1)\n\n           // No new images since last time, do nothing.\n\
    \           if (imageOut_.header.seq == imageIn.header.seq)\n             return;\n\
    \n           // use imageOut_.image and images which are *class attributes*\n\
    \           // to realize the image processing, so it will allocate memory only\
    \ once.\n           // You should have *no* instance of big objects as local variables\
    \ here.\n           // Use directly imageOut_.image as the cv::Mat destination\
    \ object  of your last\n           // OpenCV function call.\n\n           // publish\
    \ imageOut_.toMsg ();\n        }\n\n        virtual void spin ()\n        {\n\
    \          ros::Rate rate (30);\n          while (ros::ok ())\n          {\n \
    \           spinOnce ();\n            rate.sleep ();\n           }\n        }\n\
    \n        virtual void spinOnce ()\n        {\n           process ();\n      \
    \     spinOnce ();\n        }\n\n\n    protected:\n      image_transport::ImageTransport\
    \ it_;\n      image_transport::Subscriber sub_;\n      image_transport::Publisher\
    \ pub_;\n      ros::NodeHandle nh_;\n      ros::NodeHandle nhPrivate_;\n\n   \
    \   sensor_msgs::ImageConstPtr imageIn_;\n      cv_bridge::CvImage imageOut_;\n\
    \    };\n\n    class NodeWithGui : public Node\n    {\n        virtual void spinOnce\
    \ ()\n        {\n           Node::spinOnce ();\n           // and refresh GUI\
    \ using imshow.\n        }\n    };\n\n    int main ()\n    {\n      ros::init\
    \ (...);\n      ros::NodeHandle nh;\n      ros::NodeHandle nhPrivate (\"~\");\n\
    \n      Node* node = 0;\n      if (ros::param::get<bool>(\"~use_gui\"))\n    \
    \   node = new NodeWithGui (nh, nhPrivate);\n      else\n        node = new Node\
    \ (nh, nhPrivate);\n      node->spin ();\n    }\n\nSo the reason is not that you\
    \ are doing your computation in the callback, the reason is that you reallocate\
    \ at *each* callback all your objects!\n\nTo have an efficient node, allocation\
    \ should be done **once**. At the beginning (here when the first image is received).\
    \ Then, old structures should be reused.\n\nAdditional introduced features are:\n\
    \n * Pass the node handle and private node handle explicitly. It will help you\
    \ convert your node to a nodelet one day (see nodelet conversion tutorial).\n\
    \ * **ALWAYS** separate algorithms from display (GUI). Especially in robotics,\
    \ robots should not embed X11 server.\n * No global variables.\n\nAlso, you can\
    \ see that this code is much easier to benchmark as different piece of codes are\
    \ separated. You can, for instance, bench how fast process is very easily. Never\
