  body_markdown: "Here's a simple rospy script that reads from `gazebo/model_states`\
    \ and publishes all transforms as static transform\n\n    #!/usr/bin/env python\
    \  \n    import rospy\n    import tf2_ros\n    import gazebo_msgs.msg\n    import\
    \ geometry_msgs.msg\n    import time\n    \n    import pdb\n    \n    if __name__\
    \ == '__main__':\n        rospy.init_node('gazebo_tf_broadcaster')\n    \n   \
    \     broadcaster = tf2_ros.StaticTransformBroadcaster()\n    \n        publish_frequency\
    \ = rospy.get_param(\"publish_frequency\", 1)\n    \n        last_published =\
    \ None\n        def callback(data):\n            global last_published\n     \
    \       if last_published and publish_frequency > 0.0 and time.time() - last_published\
    \ <= 1.0 / publish_frequency:\n                return\n            transforms\
    \ = []\n            for i in range(len(data.name)):\n                transform\
    \ = geometry_msgs.msg.TransformStamped()\n                transform.header.stamp\
    \ = rospy.Time.now()\n                transform.header.frame_id = \"world\"\n\
    \                transform.child_frame_id = data.name[i]\n                transform.transform.translation.x\
    \ = data.pose[i].position.x\n                transform.transform.translation.y\
    \ = data.pose[i].position.y\n                transform.transform.translation.z\
    \ = data.pose[i].position.z\n                transform.transform.rotation.w =\
    \ data.pose[i].orientation.w\n                transform.transform.rotation.x =\
    \ data.pose[i].orientation.x\n                transform.transform.rotation.y =\
    \ data.pose[i].orientation.y\n                transform.transform.rotation.z =\
    \ data.pose[i].orientation.z\n                transforms.append(transform)\n \
    \           broadcaster.sendTransform(transforms)\n            last_published\
    \ = time.time()\n    \n        rospy.Subscriber(\"/gazebo/model_states\", gazebo_msgs.msg.ModelStates,\
    \ callback)\n        \n        rospy.spin()\n\n"
