    \ * cp * math.cos(theta)\n        pt.y = radius * cp * math.sin(theta)\n     \
    \   pt.z = radius * sp\n        return pt\n\n    ...\n    radius = rospy.get_param(\"\
    ~radius\", 2.0)\n    num_lat = 20\n    num_long = 20\n    for i in range(num_lat):\n\
