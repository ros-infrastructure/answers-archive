  body_markdown: "There are several approaches to using a player driver in ROS:\n\n\
    \ 1. Wrapping the player library in a ROS node (like [erratic_player](http://www.ros.org/wiki/erratic_player)\
    \ does).\n\n 2. Converting the driver to ROS manually.\n\nThe first probably gets\
    \ the job done quicker, but is harder to maintain. With a little practice, the\
    \ second is relatively straightforward. I've done quite a few. If you have follow-up\
    \ questions, feel free to ask.\n\nWhen converting a player driver to a ROS node,\
    \ the class inherited from the\nplayer Driver class is no longer needed and just\
    \ adds unnecessary\ncomplexity.  \n\nMake a simple C++ main program, something\
    \ like this:\n\n    int main(int argc, char** argv)\n    {\n      ros::init(argc,\
    \ argv, NODE);\n      ros::NodeHandle node;\n\n      // topics to read and write\n\
    \      brake_cmd = node.subscribe(\"cmd\", 10, ProcessCommand,\n             \
    \                    ros::TransportHints().tcpNoDelay(true));\n      brake_state\
    \ = node.advertise<art_brake::State>(\"state\", 10);\n\n      if (GetParameters()\
    \ != 0)             // from the player constructor\n        return 1;\n  \n  \
    \    ros::Rate cycle(20.0);                // set driver cycle rate\n  \n    \
    \  if (Setup() != 0)                     // from the player Setup()\n        return\
    \ 2;\n  \n      // Main loop; grab messages off our queue and republish them via\
    \ ROS\n      while(ros::ok())\n        {\n          ...                      \
    \         // activities from player Main()\n          cycle.sleep();         \
    \           // sleep until next cycle\n        }\n  \n      Shutdown();\n  \n\
    \      return 0;\n    }\n"

    void PlayerDriver::Main() 
    {
    
    int argc; // Player Main() method does not take argument 
    char **argv; // What to do with argc and argv??
    
    geometry_msgs::Point commandInput;
    ros::init(argc, argv, "Command");
    
    ros::NodeHandle n;
    ros::Publisher command_pub = n.advertise<geometry_msgs::Point>("servocommand", 1000);
    ros::Rate loop_rate(1);
    
    while (ros::ok())
    {
    
    
    ros::spinOnce();   
    
    ProcessMessages();
    
    //Do some stuff
    commandInput.x = globalVel.v;
    commandInput.y = globalVel.w;
    commandInput.z = 0.0;
    command_pub.publish(commandInput);
    
    //Do some stuff
    
    loop_rate.sleep();
    }
    
    }
