  body_markdown: "Example how to generate Odometry message from encoders.\n\n\n  \
    \  #include \"pr2_mechanism_model/joint.h\"\n    #include \"pr2_controller_interface/controller.h\"\
    \n    #include \"nav_msgs/Odometry.h\"\n    #include \"sensor_msgs/JointState.h\"\
    \n    #include \"tf/transform_broadcaster.h\"\n    #include \"ros/ros.h\"\n  \
    \  #include \"math.h\"\n    \n    \n    double delta_x = 0;\n    double delta_y\
    \ = 0;\n    double delta_a1 = 0;\n    double delta_a2 = 0;\n    double fi = 0;\n\
    \    double current_fi = 0;\n    double last_x = 0;\n    double last_y = 0;\n\
    \    double current_a1, current_a2;\n    double last_a1 = 0;\n    double last_a2\
    \ = 0;\n    \n    \n    void OdomCallback(const sensor_msgs::JointState::ConstPtr&\
    \ msg)\n    {\n      // Getting the positions of right and left back wheels from\
    \ /joint_states topic \n    \n      current_a1 = msg->position[4];\n      current_a2\
    \ = msg->position[5];\n    \n    }\n    \n    \n    int main (int argc, char**\
    \ argv)\n    {\n        ros::init(argc, argv, \"odom_publisher\");\n    \n   \
    \     ros::NodeHandle n;\n        ros::Publisher odom_pub;\n        ros::Subscriber\
    \ odom_sub;\n    \n        odom_sub = n.subscribe(\"/joint_states\",1000,OdomCallback);\n\
    \    \n    \n    \n        odom_pub = n.advertise<nav_msgs::Odometry>(\"odom\"\
    ,100);\n        tf::TransformBroadcaster odom_broadcaster;\n    \n          ros::Rate\
    \ r(1.0);\n    \n          ros::Time current_time, last_time;\n          current_time\
    \ = ros::Time::now();\n          last_time = ros::Time::now();\n    \n       \
    \   double  R = 0.07; //wheel radius\n          double d = 0.3;  //distance between\
    \ two wheels\n          double x = 0;     //initial values\n          double y\
    \ = 0;\n          double th = 0;\n          double vx, vy, vth;\n    \n      \
    \    int count = 0;\n          while(n.ok())\n          {\n                ros::spinOnce();\
    \               // check for incoming messages\n                current_time =\
    \ ros::Time::now();\n    \n                delta_a1 = current_a1 - last_a1;\n\
    \                delta_a2 = current_a2 - last_a2;\n    \n                last_a1\
    \ = current_a1;\n                last_a2 = current_a2;\n                //ROS_INFO(\"\
    delta_a %f\", delta_a);\n                double dt = (current_time - last_time).toSec();\n\
    \    \n                current_fi = -R*(delta_a2 - delta_a1)/d;\n            \
    \    fi +=current_fi;\n                current_fi = 0;\n                ROS_INFO(\"\
    yaw %f\", fi);\n    \n                delta_x = R*delta_a1*cos(fi);\n        \
    \        delta_y = R*delta_a1*sin(fi);\n    \n    \n                x += delta_x;\n\
    \                y += delta_y;\n                th = fi;\n                vx =\
    \ delta_x/dt;\n                vy = delta_y/dt;\n                vth = current_fi/dt;\n\
    \    \n    \n                //since all odometry is 6DOF we'll need a quaternion\
    \ created from yaw\n                   geometry_msgs::Quaternion odom_quat = tf::createQuaternionMsgFromYaw(th);\n\
    \    \n                   //first, we'll publish the transform over tf\n     \
    \              geometry_msgs::TransformStamped odom_trans;\n                 \
    \  odom_trans.header.stamp = current_time;\n                   odom_trans.header.frame_id\
    \ = \"odom\";\n                   odom_trans.child_frame_id = \"base_link\";\n\
    \    \n                   odom_trans.transform.translation.x = -x;\n         \
    \          odom_trans.transform.translation.y = -y;\n                   odom_trans.transform.translation.z\
    \ = 0.0;\n                   odom_trans.transform.rotation = odom_quat;\n    \n\
    \                   //send the transform\n                   odom_broadcaster.sendTransform(odom_trans);\n\
    \    \n                   //next, we'll publish the odometry message over ROS\n\
    \                   nav_msgs::Odometry odom;\n                   odom.header.stamp\
    \ = current_time;\n                   odom.header.frame_id = \"odom\";\n    \n\
    \                   //set the position\n                   odom.pose.pose.position.x\
    \ = -x;\n                   odom.pose.pose.position.y = -y;\n                \
    \   odom.pose.pose.position.z = 0.0;\n                   odom.pose.pose.orientation\
    \ = odom_quat;\n    \n                   //set the velocity\n                \
    \   odom.child_frame_id = \"base_link\";\n                   odom.twist.twist.linear.x\
    \ = vx;\n                   odom.twist.twist.linear.y = vy;\n                \
    \   odom.twist.twist.angular.z = vth;\n    \n                   //publish the\
    \ message\n                   odom_pub.publish(odom);\n    \n                \
    \   last_time = current_time;\n                   r.sleep();\n               \
    \    count++;\n          }\n    \n     return 0;\n    }"
