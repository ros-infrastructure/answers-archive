    <link name="robot_body">
      <visual>
        <origin xyz="0 0 0" rpy="0 0 0" />
        <geometry>
          <mesh filename="./polou_4801_body.dae"/>
        </geometry>
        <material name="blue"/>
      </visual>
      <collision>
        <geometry>
          <mesh filename="./polou_4801_body.dae"/>
        </geometry>
       </collision>
       </link>

> [ERROR] [1649121599.466978255]
> [rviz2]: Could not load resource
> [./polou_4801_body.dae]: Unable to
> open file "./polou_4801_body.dae".
