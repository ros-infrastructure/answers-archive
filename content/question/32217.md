//Οι ταχύτητες u και ω, υπολογίζονται στη συνάρτηση control

{
FILE *file;

double B[3][2];

double pi=3.141592654;

Dt=0.0005;

B[0][1] = 0.0;
B[1][0] = sin(Xj[2]*pi/180);
B[1][1] = 0.0;
B[2][0] = 0.0;
B[2][1] = 1.0;

if(Xj[0] >= Xfinal[0])
 Xk[0] = Xj[0] + (B[0][0]*u1 + B[0][1]*w1)*Dt;
else
 Xk[0] = Xj[0];

if(Xj[1] >= Xfinal[1])
 Xk[1] = Xj[1] + (B[1][0]*u1 + B[1][1]*w1)*Dt;
else
 Xk[1] = Xj[1];

if(Xj[2] >= Xfinal[2])
 Xk[2] = Xj[2] + (B[2][0]*u1 + B[2][1]*w1)*Dt;
else
 Xk[2] = Xj[2];
