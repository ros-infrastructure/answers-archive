  body_markdown: "an example of code that get image from camera then publish the image:\n\
    \n    #include \"camera.h\"\n    \n    \n    \n    CameraDriver::CameraDriver()\n\
    \    {\n    \n        ros::NodeHandle n(\"~\");\n    \n        n.param( \"camera_index\"\
    , id_camera, -1);\n        n.param( \"show\", show , false );\n        n.param(\
    \ \"fps\", fps , 30 );\n    \n        cvi.header.frame_id = \"image\";\n     \
    \   cvi.encoding = sensor_msgs::image_encodings::BGR8;\n    \n    \n    \n   \
    \     it = new image_transport::ImageTransport(n);\n    \n        pub=it->advertise(\"\
    /image_raw\",1);\n    \n    \n        if(id_camera == -1)\n        {\n       \
    \       ROS_WARN(\"camera's id has not recived\");\n              ROS_WARN(\"\
    I will open  every camera that I find :P \");\n        }\n    \n        input_video.open(id_camera);\n\
    \        input_video.set(CV_CAP_PROP_FPS,fps);\n    \n        if(!input_video.isOpened())\n\
    \        {\n            ROS_ERROR(\"Couldn't Open The Camera !\");\n         \
    \   ROS_ERROR(\"Babay :(\");\n            ros::shutdown();\n        }\n    \n\
    \    \n        ros::Rate loop_rate(fps);\n    \n        while (ros::ok())\n  \
    \      {\n    \n            input_video.read(frame);\n            cvi.image=frame;\n\
    \            cvi.header.stamp = ros::Time::now();\n            pub.publish(cvi.toImageMsg());\n\
    \    \n            if(show)\n            {\n                imshow(\"imgout.jpg\"\
    , frame);\n                if(waitKey(1) > 0);\n            }\n    \n        \
    \    loop_rate.sleep();\n    \n        }\n    \n    \n    \n    }\n\n\n\nand .h\n\
