The python script to subscribe in a camera/depth/points topic, transform the point cloud to a open3d one, and save it is the following:
    
    from sensor_msgs.msg import PointCloud2
    from ctypes import * # convert float to uint32
    from PyQt5 import QtWidgets
    import open3d
    import numpy as np
    import rospy
    import time
    import ros_numpy
    class CameraListner():
         def __init__(self):
             self.pc = None
             self.n = 0
             self.listener()
     def callback(self, points):
         self.pc = points
         self.n = self.n + 1
     def listener(self):
         rospy.init_node('ui_config_node', anonymous=True)
         rospy.Subscriber('/camera_1/depth/points', PointCloud2, self.callback)
     class ViewerWidget(QtWidgets.QWidget):
         def __init__(self, subscriber, parent=None):
            self.subscriber = subscriber
            rospy.loginfo('initialization')
        self.vis = open3d.visualization.Visualizer()
        self.point_cloud = None
        self.updater()
    def updater(self):
        rospy.loginfo('start')
        self.first = False
        while (self.subscriber.pc is None):
             time.sleep(2)                   
        self.point_cloud = open3d.geometry.PointCloud()
        self.point_cloud.points = open3d.utility.Vector3dVector(ros_numpy.point_cloud2.pointcloud2_to_xyz_array(self.subscriber.pc))
        print('get points')
        self.vis.add_geometry(self.point_cloud)
        open3d.io.write_point_cloud("~/pointclouds/pcd1.pcd", self.point_cloud)        
        print ('add points')
        while not rospy.is_shutdown():
            self.point_cloud.points =   open3d.utility.Vector3dVector(ros_numpy.point_cloud2.pointcloud2_to_xyz_array(self.subscriber.pc))
        if __name__ == '__main__':
            listener = CameraListner()
            updater = ViewerWidget(listener)
            rospy.spin()
