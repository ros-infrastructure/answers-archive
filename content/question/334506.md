  body_markdown: "> error: redefinition of \u2018ros::Subscriber<std_msgs::Empty>\
    \ **sub**\n\nmeans you are violating the [One Definition Rule](https://en.cppreference.com/w/cpp/language/definition)\
    \ (reusing the same name for a different object, with the same declaration as\
    \ well) in these lines:\n\n>     ros::Subscriber<std_msgs::Empty> sub(\"toggle_led1\"\
    , &led1 );\n>     ros::Subscriber<std_msgs::Empty> sub(\"toggle_led2\", &led2\
    \ );\n>     ros::Subscriber<std_msgs::Empty> sub(\"toggle_led3\", &led3 );\n>\
    \     ros::Subscriber<std_msgs::Empty> sub(\"toggle_all\", &all );\n>     ros::Subscriber<std_msgs::Empty>\
    \ sub(\"toggle_light\", &light );\n\nso a simple solution would be to instead\
    \ have\n\n    ros::Subscriber<std_msgs::Empty> sub_led1 (\"toggle_led1\", &led1);\n\
    \    ros::Subscriber<std_msgs::Empty> sub_led2 (\"toggle_led2\", &led2);\n   \
    \ ros::Subscriber<std_msgs::Empty> sub_led3 (\"toggle_led3\", &led3);\n    ros::Subscriber<std_msgs::Empty>\
    \ sub_all (\"toggle_all\", &all);\n    ros::Subscriber<std_msgs::Empty> sub_light\
    \ (\"toggle_light\", &light);\n\nThis is not a ROS issue but a C++ one, you should\
    \ check some C++ books and the [Publisher Subscriber tutorial](http://wiki.ros.org/ROS/Tutorials/WritingPublisherSubscriber%28c%2B%2B%29)\
    \ among the others.\n"

    #include <ros.h>

      digitalWrite(10, HIGH-digitalRead(10));   // blink the led

    }
    void light( const std_msgs::Empty& toggle_msg){
      digitalWrite(9, HIGH-digitalRead(9));

      delay(1000)
      digitalWrite(9, LOW-digitalRead(9));

      delay(2000)
      digitalWrite(9, LOW-digitalRead(9)); 
      digitalWrite(10, LOW-digitalRead(10)); 
      digitalWrite(11, HIGH-digitalRead(11)); 
      delay(3000)
    }
    ros::Subscriber<std_msgs::Empty> sub("toggle_led1", &led1 );
