    \    terminal-1:\n        $ echo \u201Csource /opt/ros/fuerte/setup.bash\u201D\
    \ >> ~/.bashrc\n        $ . ~/.bashrc\n        $ sudo apt-get install python-rosinstall\
    \ python-rosdep ros-fuerte-brown-remotelab ros-fuerte-control ros-fuerte-common-msgs\
    \ ros-fuerte-slam-gmapping ros-fuerte-hector-slam\n        $ rosws init ~/fuerte_workspace\
    \ /opt/ros/fuerte\n        $ echo \u201Csource ~/fuerte_workspace/setup.bash\u201D\
    \ >> ~/.bashrc\n        $ mkdir ~/fuerte_workspace/sandbox\n        $ source ~/fuerte_workspace/setup.bash\n\
    \        $ rosws set ~/fuerte_workspace/sandbox\n        $ source ~/fuerte_workspace/setup.bash\n\
    \        $ roscd\n        $ cd sandbox\n        $ git clone git://git.assembla.com/usarsim.git\n\
    \            (Under the proxy, try \u201Cgit clone http://git.assembla.com/usarsim.git&#8221\"\
    )\n        $ rosmake usarsim_inf\n        \n        At this point if making usarsim_inf\
    \ is successful, then use these steps,\n        \n        terminal-1: $ roscore\n\
    \        terminal-2: $ export ROS_PACKAGE_PATH=$ROS_PACKAGE_PATH:`pwd`\n     \
    \                   $ roslaunch usarsim_inf usarsim.launch\n        terminal-3:\
    \ $ rosrun teleop_twist_keyboard teleop_twist_keyboard.py\n\nnow navigate using\
    \ key. It works."
