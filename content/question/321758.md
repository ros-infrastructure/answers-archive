    CaoProv.DENSO.VRC\"))\n            bcapReq.vntArgs.append(variant(vt=8, value=\"\
    localhost\"))\n            bcapReq.vntArgs.append(variant(vt=8, value=\"\"))\n\
    \    \n            bcapSrv = rospy.ServiceProxy('bcap_service', bcap)\n      \
    \      response = bcapSrv(bcapReq)\n    \n            # print 'HRESULT:', response.HRESULT\n\
