    \ topic \"pose\" which receives geometry_msgs/Pose2D and then sets the robot's\
    \ pose to the desired one. The following are the lines that I had to add to stageros.cpp:\n\
    \n    #include <geometry_msgs/Pose2D.h>\n\n    #define POSE \"pose\" \n\n    std::vector<ros::Subscriber>\
    \ pose_subs_;\n\n    void\n    StageNode::poseReceived(int idx, const boost::shared_ptr<geometry_msgs::Pose2D\
    \ const>& msg)\n    {\n      boost::mutex::scoped_lock lock(msg_lock);\n    \t\
    Stg::Pose pose;\n    \tpose.x = msg->x;\n    \tpose.y = msg->y;\n    \tpose.z\
    \ =0;\n    \tpose.a = msg->theta;\n    \tthis->positionmodels[idx]->SetPose(pose);\n\
    \    }\n    \n    \n    pose_subs_.push_back(n_.subscribe<geometry_msgs::Pose2D>(mapName(POSE,r),\
    \ 10, boost::bind(&StageNode::poseReceived, this, r, _1)));"
