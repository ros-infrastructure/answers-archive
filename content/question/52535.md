    \ I created a urdf class that contains the same methods and structures as the\
    \ Cpp urdf class.\n\n    import java.io.File;\n    import java.io.IOException;\n\
    \    import java.util.ArrayList;\n    import java.util.List;\n    import org.xml.sax.*;\n\
    \    \n    import javax.xml.parsers.SAXParserFactory;\n    import javax.xml.parsers.ParserConfigurationException;\n\
    \    import javax.xml.parsers.SAXParser;\n    import org.w3c.dom.*;\n    \n  \
    \  import android.util.Log;\n    \n    \n    import javax.xml.parsers.*;\n   \
    \ public class ModelInterface \n    {\n    \n    \tprivate List<Joint> Joints_;\n\
    \    \tprivate List<Link> Links_;\n    \t\n    \tpublic List<Joint> getJoints()\n\
    \    \t{\n    \t\treturn Joints_;\n    \t}\n    \t\n    \tpublic List<Link> getLinks()\n\
    \    \t{\n    \t\treturn Links_;\n    \t}\n    \t\n    \tpublic void initXml(String\
    \ filename) throws SAXException, IOException, ParserConfigurationException\n \
    \   \t{\n    \t\t//This is the constructor of the object. Search for and load\
    \ YAML file for robot parameters. This method uses the YAML beans to do so.\n\
    \    \t\t\t\tJoints_= new ArrayList<Joint>();\n    \t\t\t\tLinks_= new ArrayList<Link>();\n\
    \    \t\t\t\tJoints_.clear();\n    \t\t\t\tLinks_.clear();\n    \t\t\t\t\n   \
    \ \t\t\t\t\n    \t\t\t\tboolean validating=true;\n    \t\t\t\t  \n    \t\t\t\t\
    DocumentBuilderFactory factory= DocumentBuilderFactory.newInstance();\n    \t\t\
    \t\t\n    \t\t\t\t//factory.setValidating(validating);\n    \t\t\t\t\n    \t\t\
    \t\tDocumentBuilder builder = factory.newDocumentBuilder();\n    \t\t\t\tFile\
    \ tempfile = new File(filename);\n    \t\t\t\t\n    \t\t\t\tDocument document\
    \ = builder.parse(tempfile);\n    \t\t\t\t\n    \t\t\t\t\n    \t\t\t\tNodeList\
    \ jointTags = document.getElementsByTagName(\"joint\");\n    \t\t\t\tNodeList\
    \ linkTags = document.getElementsByTagName(\"link\");\n    \t\t\t\t\n    \t\t\t\
    \tfor(int i =0; i<jointTags.getLength();i++)\n    \t\t\t\t{\n    \t\t\t\t\tJoint\
    \ tempjoint=new Joint();\n    \t\t\t\t\n    \t\t\t\t\t\n    \t\t\t\t    tempjoint.setName(jointTags.item(i).getAttributes().getNamedItem(\"\
    name\").getNodeValue());\n    \t\t\t\t\ttempjoint.setType(jointTags.item(i).getAttributes().getNamedItem(\"\
    type\").getNodeValue());\n    \t\t\t\t\ttempjoint.setLowerLimit(Double.parseDouble(jointTags.item(i).getChildNodes().item(9).getAttributes().getNamedItem(\"\
    lower\").getNodeValue()));\n    \t\t\t\t\ttempjoint.setUpperLimit(Double.parseDouble(jointTags.item(i).getChildNodes().item(9).getAttributes().getNamedItem(\"\
    upper\").getNodeValue()));\n    \t\t\t\t\t\n    \t\t\t\t\tJoints_.add(tempjoint);\n\
    \    \t\t\t\t}\n    \t\t\t\t\n    \t\t\t\tfor(int i =0; i<linkTags.getLength();i++)\n\
    \    \t\t\t\t{\n    \t\t\t\t\tLink templink=new Link();\n    \t\t\t\t\ttemplink.setName(linkTags.item(i).getAttributes().getNamedItem(\"\
    name\").getNodeValue());\n    \t\t\t\t\t\n    \t\t\t\t\tLinks_.add(templink);\n\
    \    \t\t\t\t}\n    \t}\n    \t\n     }\n\nIn the initXML (filename) the urdf\
    \ file is parsed using a DOM reader. It required a little debugging to ensure\
    \ that the correct elements were read into the correct spots. It also doesn't\
    \ check the formatting of the urdf file (ie if you change the order of how a link's\
    \ elements are written) but works off the order I had used. It only gets the link\
    \ name, and the joint name, types and limits. Conceivably you could parse the\
    \ entire file if you liked. For reference an example of my urdf order for a link\
