  body_markdown: "Here is the solution that I wanted. both of the above answers helped\
    \ me in getting a beautiful output.\n\nI'm posting the solution now.\n\n    #include\
    \ <ros/ros.h>\n    #include <visualization_msgs/Marker.h>\n    #include <cmath>\n\
    \    \n    int main( int argc, char** argv )\n    {\n      ros::init(argc, argv,\
    \ \"spheres\");\n      ros::NodeHandle n;\n      ros::Publisher marker_pub = n.advertise<visualization_msgs::Marker>(\"\
    ball_trajectory\", 10);\n    \n      ros::Rate r(30);\n    \n      float f = 0.0;\n\
    \      while (ros::ok())\n      {\n    \n        visualization_msgs::Marker sphere_list;\n\
    \        sphere_list.header.frame_id= \"/my_frame\";\n        sphere_list.header.stamp=\
    \ ros::Time::now();\n        sphere_list.ns= \"spheres\";\n        sphere_list.action=\
    \ visualization_msgs::Marker::ADD;\n        sphere_list.pose.orientation.w= 1.0;\n\
    \    \n        sphere_list.id = 0;\n    \n        sphere_list.type = visualization_msgs::Marker::SPHERE_LIST;\n\
    \    \n    \n        // POINTS markers use x and y scale for width/height respectively\n\
    \        sphere_list.scale.x = 0.1;\n        sphere_list.scale.y = 0.1;\n    \
    \    sphere_list.scale.z = 0.1;\n    \n        // Points are green\n        sphere_list.color.r\
    \ = 1.0f;\n        sphere_list.color.a = 1.0;\n    \n    \n        // Create the\
    \ vertices for the points and lines\n        for (uint32_t i = 0; i < 25; ++i)\n\
    \        {\n          float y = sin(f + i / 100.0f * 2 * M_PI);\n          float\
    \ z = cos(f + i / 100.0f * 2 * M_PI);\n    \n          geometry_msgs::Point p;\n\
    \          p.x = (int32_t)i*0.2;\n          p.y = y;\n          p.z = z;\n   \
    \ \n          sphere_list.points.push_back(p);\n    \n        }\n        marker_pub.publish(sphere_list);\n\
    \    \n        r.sleep();\n    \n        f += 0.04;\n      }\n    }\n\nif there\
