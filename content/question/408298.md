Hello.
I am trying to move a YuMi IRB14000 dual arm from a program using ros's moveit library.
I was able to move both the right arm and the left arm independently, one arm at a time, but I cannot figure out how to move both arms at the same time.
I would appreciate it if someone could help me understand even the most trivial details.

Here is the node to move the left arm

    #include <ros/ros.h>
    #include <moveit/move_group_interface/move_group_interface.h>
    #include <moveit/planning_scene_interface/planning_scene_interface.h>
    #include <moveit_visual_tools/moveit_visual_tools.h>
    #include <actionlib/client/simple_action_client.h>
    #include <tf2/LinearMath/Quaternion.h>
    #include <actionlib/client/simple_action_client.h>
    #include <control_msgs/GripperCommandAction.h>
    #include <std_msgs/Float64.h>
    #include <sensor_msgs/JointState.h>
    #include <abb_rapid_sm_addin_msgs/SetSGCommand.h>
    #include <abb_robot_msgs/TriggerWithResultCode.h>
    #include <tf2/LinearMath/Quaternion.h>

    void q(double rx,double ry,double rz,double &q1,double &q2,double &q3,double &q4){
        double rx_rad,ry_rad,rz_rad;
        rx_rad = rx * (M_PI/180);
        ry_rad = ry * (M_PI/180);
        rz_rad = rz * (M_PI/180);//degをradに変換
        tf2::Quaternion myQuaternion;
        myQuaternion.setRPY( rx_rad, ry_rad, rz_rad);//オイラー角（rad）からクォータニオンに変換
        q1=myQuaternion.getX();
        q2=myQuaternion.getY();
        q3=myQuaternion.getZ();
        q4=myQuaternion.getW();
    }

    int main(int argc, char **argv) {
    ros::init(argc, argv, "pick_and_placer_l");
    ros::NodeHandle nh;
    ros::ServiceClient client = nh.serviceClient<abb_rapid_sm_addin_msgs::SetSGCommand>    ("/yumi/rws/sm_addin/set_sg_command");
    ros::ServiceClient client_2 = nh.serviceClient<abb_robot_msgs::TriggerWithResultCode>("/yumi/rws/sm_addin/run_sg_routine");
    ros::AsyncSpinner spinner(2);
    spinner.start();
    moveit::planning_interface::MoveGroupInterface arm("left_arm");
    arm.setPoseReferenceFrame("yumi_base_link");
    ros::Duration(0.5).sleep();
    //////////////////////////////////////////////////////////////////////////////////////////////
    abb_rapid_sm_addin_msgs::SetSGCommand sg;
    sg.request.task = "T_ROB_L";
    sg.request.command = 5;
    sg.request.target_position = 20;
    if(client.call(sg))
    {
    ROS_INFO("success");
    }
    else
    {
    ROS_ERROR("Failed to call service add_two_ints");
    return 1;
    }
    abb_robot_msgs::TriggerWithResultCode run_sg;
    if(client_2.call(run_sg))
    {
    ROS_INFO("success");
    }
    else
    {
    ROS_ERROR("Failed to call service add_two_ints");
    return 1;
    }

    ros::Duration(1.0).sleep();
    //////////////////////////////////////////////////////////////////////////////////////////////
    ROS_INFO("Moving to prepare pose");
    geometry_msgs::PoseStamped pose;
    pose.header.frame_id = "yumi_base_link";
    pose.pose.position.x = 0.3;
    pose.pose.position.y = 0.3;
    pose.pose.position.z = 0.3;
    double rx,ry,rz,q1,q2,q3,q4;
    rx=180;
    ry=0;
    rz=90;
    q(rx,ry,rz,q1,q2,q3,q4);
    pose.pose.orientation.x = q1;
    pose.pose.orientation.y = q2;
    pose.pose.orientation.z = q3;
    pose.pose.orientation.w = q4;
    arm.setPoseTarget(pose);
    if (!arm.move()) {
    ROS_ERROR("Could not move");
    return 1;
    }
    ros::Duration(4).sleep();
    //////////////////////////////////////////////////////////////////////////////////////////////
    pose.pose.position.x += 0.1;
    pose.pose.position.y += 0.1;
    pose.pose.position.z += 0.1;
    arm.setPoseTarget(pose);
    if (!arm.move()) {
    ROS_ERROR("Could not move");
    return 1;
    }
    ros::Duration(4).sleep();

    //////////////////////////////////////////////////////////////////////////////////////////////
    ros::shutdown();
    return 0;
    }
