    \n    glMultMatrixd(static_cast<GLdouble*>( (*cloud_matrices)[i].data() ));//works\
    \ as long as qreal and GLdouble are typedefs to double (might depend on hardware)\
    \ \n\nto\n\n    glMultMatrixf(static_cast<GLfloat*>( (*cloud_matrices)[i].data()\
    \ ));//works as long as qreal and GLfloat are typedefs to float (depends on hardware)"
