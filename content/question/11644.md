    \ command such as rosrun or roslaunch:\n\n    // this code would be called from\
    \ the callback method of your button\n    pid_t pid = 0;\n    pid = fork();\n\
    \    if (pid < 0) {\n      ROS_ERROR(\"Process failed to fork\");\n      exit(-1);\n\
    \    }\n    else if (pid == 0) { // Child process\n      execlp( \"roslaunch\"\
    , \"roslaunch\", \"launchfolder\", \"launchfile.launch\",NULL);\n      // launchfolder\
    \ is the directory where your launchfile is\n      exit(1);\n    }\n    else {\
    \ // parent process\n      launchPID = pid;\n    }\n        \n    // Kill the\
    \ started process when exiting the application (in onExit() and/or onClose() callbacks:\n\
    \    if (launchPID) kill(launchPID, 15);  \n    \nUsing system() instead is much\
    \ easier, but you don't have any control over the processes you created..."
