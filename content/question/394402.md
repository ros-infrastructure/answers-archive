I am currently working on visualization of the data.i have error that 'For frame []: Fixed Frame [map] does not exist' in rviz. how can i solve this problem.my code is below.

include <Wire.h>
include <LSM6.h>
include <ros.h>
include <sensor_msgs/Imu.h>
include <std_msgs/Float32.h> 








ros::NodeHandle  nh;
sensor_msgs::Imu imu_msg;
ros::Publisher imu_pub("imu", &imu_msg);


LSM6 imu;
char report[120];


void setup()
{
  Wire.begin();

  if (!imu.init())
  {
    while (1);
  }
  imu.enableDefault();
  nh.initNode();
  nh.advertise(imu_pub);
  

 

}

void loop()
{
 

  imu.read();

  snprintf(report, sizeof(report),"A: &ax &ay &az &aw G: &gx &gy &gz ",
    imu.a.x, imu.a.y, imu.a.z,
    imu.g.x, imu.g.y, imu.g.z);

         

        imu_msg.orientation.w = 0;
        imu_msg.orientation.x = 0;
        imu_msg.orientation.y = 0;
        imu_msg.orientation.z = 0;
       
          
        // angular velocity
        imu_msg.angular_velocity.x = imu.g.x;
        imu_msg.angular_velocity.y = imu.g.y;
        imu_msg.angular_velocity.z = imu.g.z;

        // linear acceleration
        imu_msg.linear_acceleration.x = imu.a.x;
        imu_msg.linear_acceleration.y = imu.a.y;
        imu_msg.linear_acceleration.z = imu.a.z;
        imu_pub.publish(&imu_msg); // publishing the imu data

  imu_pub.publish( & imu_msg );


  delay(100);
  nh.spinOnce();
}
