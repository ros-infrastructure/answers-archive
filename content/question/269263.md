    \ the problem is that CPP bytes are handled using **signed char** type, not **int**.\n\
    So, you have to do like this:\n\n    void arrayCallback(const std_msgs::ByteMultiArray::ConstPtr&\
    \ array)\n    {\n    \n        int i = 0;\n        // print all the remaining\
    \ numbers\n        for(std::vector<signed char>::const_iterator it = array->data.begin();\
    \ it != array->data.end(); ++it)\n        {\n            Arr[i] = *it;\n     \
    \       i++;\n        }\n    \n        return;\n    }\n\nThere's a tutorial showing\
    \ how to deal with ByteMultiArray messages, you can check it here: https://youtu.be/RMTLKf5q3Ro"
