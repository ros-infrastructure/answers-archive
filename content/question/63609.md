    \n    if __name__ == '__main__':\n\n      init_node('input_test')\n\n      while\
    \ not is_shutdown():\n\t\n          print \"gimme something, please?\"\n     \
    \     something = raw_input()\n          print \"thanks for giving me \" + something\n\
