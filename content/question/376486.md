I am running ROS Noetic on Ubuntu 20.4. I use clion to encode  my program and try to make a robot vacuum cleaner. I want to imporve the progarm and rotate it in the direction that is freer from obstacles, not always rotate counterclockwise. How to jugde whether the obstacle is on the right or the left using "LaserScan.h"？
Here is my code.

#include "Stopper.h"
#include "geometry_msgs/Twist.h"

Stopper::Stopper() {

    commandPub = node.advertise<geometry_msgs::Twist>("cmd_vel",10);

    laserSub = node.subscribe("scan", 1, &Stopper::scanCallback, this);
}


void Stopper::moveForward() {

    geometry_msgs::Twist msg;

    msg.linear.x = FORWARD_SPEED;

    commandPub.publish(msg);
}


void Stopper::scanCallback(const sensor_msgs::LaserScan::ConstPtr &scan) {

    isObstacleInFront = false;

    int minIndex = ceil((MIN_SCAN_ANGLE - scan->angle_min) / scan -> angle_increment);
    int maxIndex = floor((MAX_SCAN_ANGLE - scan->angle_min) / scan -> angle_increment);
    for (int currIndex = minIndex + 1; currIndex <= maxIndex; currIndex++) {
        if (scan->ranges[currIndex] < MIN_DIST_FROM_OBSTACLE) {
            if (scan->ranges[currIndex] >0.01 ){
                isObstacleInFront = true;

                ROS_INFO("Found something.Will stop now!");
                geometry_msgs::Twist msg;
                stopMoving();
                break;
            }
        }
    }
}

void Stopper::startMoving() {
    ros::Rate rate(10);
    ROS_INFO("Start moving");

    while (ros::ok() && !isObstacleInFront) {
        moveForward();
        ros::spinOnce(); 
        rate.sleep();
    }
    stopMoving();
}

void Stopper::stopMoving() {

    while (ros::ok() && isObstacleInFront) {
        ros::Rate rate(10);
        geometry_msgs::Twist msg; 
        msg.linear.x = 0;
        msg.angular.z = 1;
        commandPub.publish(msg);
        ros::spinOnce(); 
        rate.sleep();
    }
}
