namespace gazebo {

    //constructor
    BlinkerLightPlugin::BlinkerLightPlugin()

        this->color_red= ignition::math::Color(1.0,0.0,0.0);
        this->color_green= ignition::math::Color(0.0,1.0,0.0);
        this->color_yellow=ignition::math::Color(1.0,0.5,0.0);

    //destructor
    BlinkerLightPlugin::~BlinkerLightPlugin(){
        cout << "==========VISUAL PLUGIN KILLED==========="<< endl;
        this->rosnode_->shutdown();
        this->callback_queue_thread_.join();
        delete this->rosnode_;

    void BlinkerLightPlugin::Load(rendering::VisualPtr _visual, sdf::ElementPtr _sdf){

        //check if we have the visual
        cout << " INIT STARTED BLINKER" <<endl;
        if (!_visual || !_sdf){
            gzerr << "No visual or SDF element specified. Plugin won't load." << std::endl;
            return;
        }

        this->sdf_ = _sdf;
        this->model_blinker = _visual;


        //getting all the needed elements from the custom XML elements
        this->robot_namespace_ = "";
        if (this->sdf_->HasElement("topicName"))
            this->robot_namespace_ = this->sdf_->Get<std::string>("topicName");

        //starting with the init state 
        this->stop_blink();

        //initialisation of the nodes and the subscriber
        this->rosnode_ = new ros::NodeHandle(this->robot_namespace_);
        ros::SubscribeOptions so = ros::SubscribeOptions::create<std_msgs::Bool>("/"+this->robot_namespace_,1,boost::bind(&BlinkerLightPlugin::ServiceCallback, this, _1),ros::VoidPtr(), &this->model_queue_);
        this->srv_ = this->rosnode_->subscribe(so);
        //init of the callback thread
        this->callback_queue_thread_ = std::thread(std::bind(&BlinkerLightPlugin::VisualQueueThread, this));

        cout << " INIT FINISHED" <<endl;
    }

    //funciton to make the model blink
    void BlinkerLightPlugin::blink()
    {   
        //seting the bool floag at true to start the "infinite" loop
        this->blinking = true;
        while(this->blinking)
        {
            //Changing color of the blinker
            this->model_blinker->SetAmbient(this->color_black);
            usleep(750000);
            this->model_blinker->SetAmbient(this->color_yellow);
            usleep(750000);
        }
        return;
    }

    //function tos stop the blinking
    void BlinkerLightPlugin::stop_blink()
    {
        //Stopping the thread using the bool flag
        this->blinking = false;
        //waiting for the thread to die 
        usleep(750000*2);
        //changing hte color to black
        this->model_blinker->SetAmbient(this->color_black);
    }


     // Callback when using service
    void BlinkerLightPlugin::ServiceCallback(const std_msgs::BoolConstPtr &_msg){

        // retrieving data
        const bool& test = _msg->data;
        cout<<test<<endl;
        if(test)
        {
            cout<<"BLINK"<<endl;
            //if the thread doesn't exist
            if(this->nb_times_thread_started ==0)
            {
                //creating a thread
                this->blink_thread = std::thread(&BlinkerLightPlugin::blink,this);
                //increment a counter to be sure to not start 2 times the thread if the user call 
                // this topic with true then true again
                this->nb_times_thread_started++;
            }

        }else
        {
            cout<<"STOP BLINK"<<endl;
            this->stop_blink();
            //if the thread exist
            if(this->nb_times_thread_started>0)
            {
                //we join it
                this->nb_times_thread_started--;
                //we decrement the counter to avoid trying to join a not started thread
                // if the user call false then false again 
                this->blink_thread.join();
            }
        }
    }


    void BlinkerLightPlugin::VisualQueueThread()
    {
        static const double timeout = 0.01;
        while (this->rosnode_->ok()) {
            model_queue_.callAvailable(ros::WallDuration(timeout));
        }
    }
}
