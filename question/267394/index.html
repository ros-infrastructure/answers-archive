Answers

You can try adding 

find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP jsoncpp)

and changing include_directories and target_link_libraries as

include_directories(
  ${catkin_INCLUDE_DIRS}
  ${JSONCPP_INCLUDE_DIRS}
)

and

target_link_libraries(json
  ${catkin_LIBRARIES}
  ${JSONCPP_LIBRARIES}
)

Good luck.
