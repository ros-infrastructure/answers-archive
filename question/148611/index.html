write the following lines in your Cmake

find_package(Qt4 COMPONENTS QtCore QtGui)
INCLUDE(${QT_USE_FILE})
ADD_DEFINITIONS(${QT_DEFINITIONS})
rosbuild_add_executable(qt_test src/qt_test.cpp)
target_link_libraries(qt_test ${QT_LIBRARIES})
