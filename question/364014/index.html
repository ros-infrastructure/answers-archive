!/usr/bin/env python

import rospy from geometry_msgs.msg import Twist from turtlesim.msg import Pose

def pose_callback(ms): print("msg.theta")

def revolve():

rospy.init_node('turtle_revolve', anonymous=True)
velocity_publisher = rospy.Publisher('/turtle1/cmd_vel', Twist, queue_size=10)
vel_loop_rate = rospy.Rate(10)


while not rospy.is_shutdown():

    rospy.Subscriber("/turtle1/pose", Pose, pose_callback)

    vel_msg = Twist()

    vel_msg.linear.x = 10
    vel_msg.linear.y= 0
    vel_msg.linear.z= 0 
    vel_msg.angular.x= 0
    vel_msg.angular.y= 0
    vel_msg.angular.z= 2

    velocity_publisher.publish(vel_msg)
    vel_loop_rate.sleep()
    rospy.spin()

if name == 'revolve': try: revolve() except rospy.ROSInterruptException: pass
