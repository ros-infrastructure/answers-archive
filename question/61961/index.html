Point Cloud Visualization Error.

This question has also been migrated to Robotics StackExchange as https://robotics.stackexchange.com/questions/50815/point-cloud-visualization-error.

I am trying to visualize point cloud, but it gave error like error C2664: 'void pcl::visualization::CloudViewer::showCloud(const boost::sharedptr &,const std::string &)' : cannot convert parameter 1 from 'pcl::PointCloud' to 'const boost::sharedptr &'. Thanks in advance.

Here is my sample code: 

#include <iostream>

#include <pcl/io/pcd_io.h>
#include <pcl/point_types.h>
#include <pcl/visualization/cloud_viewer.h>

int main (int argc, char** argv) 
{ 
  pcl::PointCloud<pcl::PointXYZ> cloud; 

  // Fill in the cloud data 
  cloud.width    = 5; 
  cloud.height   = 1; 
  cloud.is_dense = false; 
  cloud.points.resize (cloud.width * cloud.height); 

  for (size_t i = 0; i < cloud.points.size (); ++i) 
  { 
    cloud.points[i].x = 1024 * rand () / (RAND_MAX + 1.0f); 
    cloud.points[i].y = 1024 * rand () / (RAND_MAX + 1.0f); 
    cloud.points[i].z = 1024 * rand () / (RAND_MAX + 1.0f); 
  } 

  pcl::visualization::CloudViewer viewer ("Simple Cloud Viewer"); 
  viewer.showCloud (cloud); 

  while (!viewer.wasStopped ()) 
  { 
  } 

  return (0); 
}
