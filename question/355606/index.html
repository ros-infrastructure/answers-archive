def get_position_using_encoders(self, dt, left_enc, right_enc, yaw=0, verbose=False):

        dleft_1 = (left_enc - self.enc_left)

    self.dth += (dright - dleft) / self.wheel_track

    y += dxy_ave * cos(self.dth + yaw)
