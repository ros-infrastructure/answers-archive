I think the issue may lie within clock synchronization because when I checked my tree using rqttftree, I was getting all my links from map->odom->baselink->laserframe. I am getting Costmap2DROS transform timeout warning which has led to "Estimating Initial Pose Warning". Another thing that I am getting is Lost sync with device. Restarting... on rosserialpython serial.node but when I run teleoptwist_keyboard my robot seems to respond satisfactorily.

Below is the Arduino code that I'm using given by James Bruton on youtube. I have initialised Serial.begin with baudrate 115200 and LoopTime=10ms. I wonder if I have to change these values to eliminate syncing use which may remove the warnings.

Please any help would be appreciated. I am a novice as you can tell and have worked with ROS for only a month for an undergraduate project.

float demand1;

void velCallback(  const geometry_msgs::Twist& vel)
{
     demandx = vel.linear.x;
     demandz = vel.angular.z;

     demandx = constrain(demandx, -0.33, 0.33);
     demandz = constrain(demandz, -1, 1);
}

ros::Subscriber<geometry_msgs::Twist> cmd_vel("cmd_vel" , velCallback);  //create a subscriber for ROS "cmd_vel" topic
geometry_msgs::Vector3Stamped speed_msg;                             //create a "speed_msg" ROS message
ros::Publisher speed_pub("speed", &speed_msg);                       //create a publisher to ROS topic "speed" using the "speed_msg" type

    void setup() {
Serial.begin(115200);
nh.getHardware()->setBaud(115200);
nh.initNode();            //init ROS
nh.subscribe(cmd_vel);        //subscribe to cmd_vel

pinMode(4, OUTPUT); // motor PWM pins
pinMode(5, OUTPUT);
pinMode(6, OUTPUT);
pinMode(7, OUTPUT);

void loop() {
  // put your main code here, to run repeatedly:
   nh.spinOnce();

    currentMillis = millis();

    if (currentMillis - previousMillis >= loopTime ) { // start timed loop
        previousMillis = currentMillis;

          // work out the two values for differential drive of each wheel

          demand1 = demandx - (demandz*0.1475);
          demand2 = demandx + (demandz*0.1475)

            encoder0Diff = encoder0Pos - encoder0Prev;  // work out difference from last time
            encoder1Diff = encoder1Pos - encoder1Prev;  // this is the current speed in counts per 10 ms

            encoder0Error = (demand1*88) - encoder0Diff;
            encoder1Error = (demand2*88) - encoder1Diff;

            encoder0Prev = encoder0Pos; // bookmark previous values
            encoder1Prev = encoder1Pos;

        Setpoint1 = demand1*88; // drive wheel 1 at 45.7 counts per 10ms cycle
        Input1 = encoder1Diff;
        PID1.Compute();

        Setpoint2 = demand2*88; // drive wheel 2 at 45.7 counts per 10ms cycle
        Input2 = encoder0Diff;
        PID2.Compute();

        //work out actual motor speed from wheel encoders
        speed_act_left = encoder0Diff/88;
        speed_act_right = encoder1Diff/88;


        // drive motor

        if (Output1>0) {
          Output1a = abs(Output1);
          analogWrite(7, Output1a);
          analogWrite(6, 0);
          }
          else if (Output1<0) {
            Output1a = abs(Output1);
            analogWrite(6, Output1a);
            analogWrite(7, 0);
           }
           else{
            analogWrite(6,0);
            analogWrite(7,0);
            }

            // other motor
            if (Output2>0) {
          Output2a = abs(Output2);
          analogWrite(5, Output2a);
          analogWrite(4, 0);
          }
          else if (Output2<0) {
            Output2a = abs(Output2);
            analogWrite(4, Output2a);
            analogWrite(5, 0);
           }
           else{
            analogWrite(4,0);
            analogWrite(5,0);
            }

            publishSpeed(loopTime);

    } 


}

void publishSpeed(double time) 
{
  speed_msg.header.stamp = nh.now();      //timestamp for odometry data
  speed_msg.vector.x = speed_act_left;    //left wheel speed (in m/s)
  speed_msg.vector.y = speed_act_right;   //right wheel speed (in m/s)
  speed_msg.vector.z = time/1000;         //looptime, should be the same as specified in LOOPTIME (in s)
  speed_pub.publish(&speed_msg);
  nh.spinOnce();
  nh.loginfo("Publishing odometry");
}
