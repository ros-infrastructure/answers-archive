Rosserial, printing garbage to terminal after roslaunch of package.

I'm trying to get a ROS node that I created to communicate RPLIDAR data to an Arduino Uno. The ROS node I built named "arduinobase" subscribes to the /scan topic from the rplidar node in the rplidarros package and publishes that data to a new topic in the form of a Float32MultiArray, which is than subscribed to by an Arduino node. After launching my launch file which launches the arduinobase node, the rplidarnode, and the serialnode.py node which handles the serial communication to the Arduino Uno, the terminal prints line after line of unintelligible garbage, and nothing appears to be recieved by the Arduino. I have tried fiddling with the baud rates of the serialnode.py node, and the Arduino node respectively to no effect thinking that was the problem. I suppose the Arduino Uno is a rather low spec board with limited memory, and I should note that after uploading my Arduino script I get the "Low memory available, stability problems may occur" message in the Arduino IDE console, so maybe upgrading to a more capable board would solve the issue? What should I do?

I should also note that I'm using ROS-Melodic, on a Ubuntu 18.04 virtual machine facilitated by VMware Fusion on a MacBook Pro.

Here is the code to the arduino_base node

#include "ros/ros.h"
#include "sensor_msgs/LaserScan.h"
#include "std_msgs/Float32MultiArray.h"
#include "std_msgs/MultiArrayLayout.h"
#include "std_msgs/MultiArrayDimension.h"


#define RAD2DEG(x) ((x)*180./M_PI) 




class PublisherSubscriber{

private:
ros::NodeHandle nh;
ros::Subscriber sub;
ros::Publisher pub;
std_msgs::Float32MultiArray lidar_data;

public:

PublisherSubscriber(std::string sub_topic_name, int que_size, std::string pub_topic_name){

  sub = nh.subscribe<sensor_msgs::LaserScan>(sub_topic_name, que_size, &PublisherSubscriber::scanCallback,this);
  pub = nh.advertise<std_msgs::Float32MultiArray>(pub_topic_name, que_size);

  ros::spin();

}
void scanCallback(const sensor_msgs::LaserScan::ConstPtr& scan){

    int count = scan->scan_time/scan->time_increment; 
    for(int i = 0; i < count; i++){
        float degree = RAD2DEG(scan->angle_min + scan->angle_increment*i); 
        lidar_data.data.push_back(degree);
        float range = scan->ranges[i]; 
        lidar_data.data.push_back(range);
    }

   pub.publish(lidar_data);
   ros::spin();

}

};




int main(int argc, char **argv){

    ros::init(argc, argv, "arduino_base");

    while(ros::ok()){

    PublisherSubscriber pubsub("/scan", 1000, "lidar_data");    

    }

    return(0);

}

This is the script of Arduino node. 

#include <LiquidCrystal_I2C.h>
#include <ros.h>
#include "std_msgs/Float32MultiArray.h"
#include <Wire.h>


ros::NodeHandle nh;

LiquidCrystal_I2C lcd(0x27, 20, 4);


void scanCallback(const std_msgs::Float32MultiArray& rpdata)
{
  lcd.print(rpdata.data[1]);
  delay(5000);
  lcd.clear();
  lcd.home();
}

ros::Subscriber<std_msgs::Float32MultiArray> sub("lidar_data", &scanCallback);

void setup() {

  pinMode(5,OUTPUT);
  nh.getHardware()->setBaud(115200);

  lcd.init();
  lcd.backlight();

  nh.initNode();
  nh.subscribe(sub);


}

void loop() {
  nh.spinOnce();
  delay(1);

}
