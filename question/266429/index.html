The launch file syntax in that tutorial looks wrong to me. Try this:

<?xml version="1.0"?>
<launch>
    <param name="/robot_description" textfile="$(find your_package_name)/path/to/your/file.urdf"/>
    <!-- The rest of your launch configuration -->
</launch>
