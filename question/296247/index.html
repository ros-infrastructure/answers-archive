Problem with rostopic echo /Drone_temp

Hello! I've some problem with my code. I've tried to ROSify a python code that reads the temperature from multiple DS18B20 sensors. When running the code it appears on the rostopic list, but it doesn't publish/echoing anything. I want to publish the temperature from multiple sensors with the ID of the sensor.

I'm new to python and ROS and I have tried to find a solution to the problem for a while now.

# Temp.py
#!/usr/bin/env python

import glob
import time
import rospy
from std_msgs.msg import Float32

def Temppublish():
    pub = rospy.Publisher('Drone_Temp', Float32, queue_size=10)
    rospy.init_node('ixDS18B20', anonymous=True)
    rate = rospy.Rate(10)

   while True: 
      for sensor in glob.glob("/sys/bus/w1/devices/28-00*/w1_slave"):
         id = sensor.split("/")[5]  
         try:
            f = open(sensor, "r")
            data = f.read()
            f.close()
            if "YES" in data:
               (discard, sep, reading) = data.partition(' t=')
               t = float(reading) / 1000.0
               pub.publish("{} {:.1f}".format(id, t))
            else:
               pub.publish("999.9") 
         except:
            pass  
      time.sleep(3.0)  

   if __name__ == '__main__':
    try:
        Temppublish()
    except rospy.ROSInterruptException:
        pass
