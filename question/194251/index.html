if you follow the tutorial step by step, then you may want this

catkin_package(
     INCLUDE_DIRS include
     LIBRARIES beginner_tutorials
     CATKIN_DEPENDS std_msgs
     CATKIN_DEPENDS rospy
     CATKIN_DEPENDS roscpp
     DEPENDS system_lib)
