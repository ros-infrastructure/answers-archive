Answers

FOund the issue.

This is the correct:

  <link name="head">
  <pose >0.090 0 0.16 0 0 0</pose>
  <collision name="collision">
    <pose >0 0 0 0 0 0</pose>
    <geometry>
      <cylinder>
        <radius>0.01</radius>
        <length>0.02</length>
      </cylinder>
    </geometry>
  </collision>
  <visual name="visual">
    <pose>0 0 0 0 0 0</pose>
    <geometry>
      <cylinder>
        <radius>0.01</radius>
        <length>0.02</length>
      </cylinder>
    </geometry>
  </visual>
</link>
