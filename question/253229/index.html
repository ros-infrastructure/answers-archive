set(CMAKE_FIND_ROOT_PATH /usr /usr/lib/aarch64-linux-gnu /opt/ros/kinetic /lib/aarch64-linux-gnu)

set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM BOTH)

set(CMAKE_LIBRARY_PATH /usr/lib/aarch64-linux-gnu)
