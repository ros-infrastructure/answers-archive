import sys
import rospy
from moveit_commander import RobotCommander, MoveGroupCommander

if __name__=='__main__':

    roscpp_initialize(sys.argv)
    rospy.init_node('moveit_py_demo', anonymous=True)

    scene = PlanningSceneInterface()
    robot = RobotCommander()
    manipulator = MoveGroupCommander("manipulator")
    gripper = MoveGroupCommander("gripper")
    rospy.sleep(1)

    # clean the scene
    scene.remove_world_object("table")
    scene.remove_world_object("part")

    manipulator.set_named_target("home")
    manipulator.go()

    rospy.sleep(1)

    # publish a demo scene
    p = PoseStamped()
    p.header.frame_id = robot.get_planning_frame()

    # add a table
    p.pose.position.x = 0.42+0.5+0.5
    p.pose.position.y = -0.2
    p.pose.position.z = 0.3
    scene.add_box("table", p, (0.5, 1.5, 0.6))

    # add an object to be grasped
    p.pose.position.x = 0.205+0.5+0.2+0.5
    p.pose.position.y = -0.12
    p.pose.position.z = 0.7
    scene.add_box("part", p, (0.2, 0.2, 0.2))

    rospy.sleep(1)
    end_effector=manipulator.get_end_effector_link()
    print ("Current Pose: ", manipulator.get_current_pose(end_effector))

    grasp_pose.pose.position.y = -0.108523288971
    grasp_pose.pose.position.z =  1.1
    grasp_pose.pose.orientation.x =  5.25684782102e-05
    grasp_pose.pose.orientation.y =  0.999963351095
    grasp_pose.pose.orientation.z = -7.74589054838e-05
    grasp_pose.pose.orientation.w =   0.00856082378227
    manipulator.set_pose_target(grasp_pose)
    manipulator.go(wait=True)
    print("testing movement")
    rospy.sleep(2)

    # set the grasp pose
    g.grasp_pose = grasp_pose

    # define the pre-grasp approach
    g.pre_grasp_approach.direction.header.frame_id = "tool0"
    g.pre_grasp_approach.direction.vector.x = 0.0
    g.pre_grasp_approach.direction.vector.y = 0.0
    g.pre_grasp_approach.direction.vector.z = 1.0
    g.pre_grasp_approach.min_distance = 0.05
    g.pre_grasp_approach.desired_distance = 0.1

    g.pre_grasp_posture.header.frame_id = "tool0"
    g.pre_grasp_posture.joint_names = ["joint_1","joint_2","joint_3","joint_4","joint_5","joint_6"]

    pos = JointTrajectoryPoint()
    pos.positions.append(0.0)

    g.pre_grasp_posture.points.append(pos)
    g.allowed_touch_objects = ["part"]
    # set the grasp posture
    g.grasp_posture.header.frame_id = "tool0"
    g.grasp_posture.joint_names = ["joint_1","joint_2","joint_3","joint_4","joint_5","joint_6"]

    pos = JointTrajectoryPoint()
    pos.positions.append(0.2)
    pos.effort.append(0.0)

    g.grasp_posture.points.append(pos)

    g.post_grasp_retreat.direction.header.frame_id =  "base_link"

    g.post_grasp_retreat.desired_distance = 0.25
    g.post_grasp_retreat.min_distance = 0.001

    g.allowed_touch_objects = ["part"]

    g.max_contact_force = 0

    # append the grasp to the list of grasps
    grasps.append(g)
    print ("picking in process")

    rospy.sleep(2)

    # pick the object
    robot.manipulator.pick("part", grasps)
