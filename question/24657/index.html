def inertia(m, x, y, z):
    """Solid cuboid of height h, width w, and length l, and mass m."""
    Ixx = m * (y**2 + z**2) / 12
    Iyy = m * (x**2 + z**2) / 12
    Izz = m * (x**2 + y**2) / 12
    tag = '<inertia ixx="{}" ixy="0.0" ixz="0.0" iyy="{}" iyz="0.0" izz="{}"/>'.format(Ixx, Iyy, Izz)
    return tag
