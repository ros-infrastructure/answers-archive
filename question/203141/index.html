<?xml version="1.0"?> 

<!-- Octomap Server -->
<node pkg="octomap_server" type="octomap_server_node" name="octomap">
    <param name="resolution" value="0.05" />

    <param name="frame_id" type="string" value="base_link" />

    <param name="max_sensor_range" value="20.0" />

    <param name="latch" value="false" />
    <remap from="cloud_in" to="occupied_cells_vis" />

</node>

but unfortunately I cant see anything on Rviz MarkerArray which is listening to occupiedcellsvis. What am I doing wrong here ????
