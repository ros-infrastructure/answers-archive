#!/usr/bin/python
import RPi.GPIO as GPIO
from time import sleep
import rospy
from std_msgs.msg import Int32


counter = 0 
# GPIO Ports
Enc_A = 23 
Enc_B = 24  


def init():
    print "Rotary Encoder Test Program"

    GPIO.setwarnings(True)

    # Use the Raspberry Pi BCM pins
    GPIO.setmode(GPIO.BCM)

    # define the Encoder switch inputs
    GPIO.setup(Enc_A, GPIO.IN)
    GPIO.setup(Enc_B, GPIO.IN)

    # setup an event detection thread for the A encoder switch
    GPIO.add_event_detect(Enc_A, GPIO.RISING, callback=rotation_decode, bouncetime=2) 
    return


def rotation_decode(Enc_A):
    pub= rospy.Publisher("lwheel_ticks", Int32, queue_size=10)
    rospy.init_node("left_encoder", anonymous=True)
    rate = rospy.Rate(10)


    global counter

    sleep(0.002) 
    # read both of the switches
    Switch_A = GPIO.input(Enc_A)
    Switch_B = GPIO.input(Enc_B)

    if (Switch_A == 1) and (Switch_B == 0) : # A then B ->
        counter += 1
        print  counter
        # at this point, B may still need to go high, wait for it
        while Switch_B == 0:
            Switch_B = GPIO.input(Enc_B)
        # now wait for B to drop to end the click cycle
        while Switch_B == 1:
            Switch_B = GPIO.input(Enc_B)
        return

    elif (Switch_A == 1) and (Switch_B == 1): # B then A <-
        counter -= 1
        print  counter
        # A is already high, wait for A to drop to end the click cycle
        while Switch_A == 1:
            Switch_A = GPIO.input(Enc_A)
        return

    else: # discard all other combinations
        return

    while not rospy.is_shutdown():
        rotation_decode.publish(counter)
        rate.sleep()


def main():
    '''
    The main routine.

    '''
    try:

        init()
        while True :
            #
            # wait for an encoder click
            sleep(0.1)

    except KeyboardInterrupt: # Ctrl-C to terminate the program
        GPIO.cleanup()


if __name__ == '__main__':

    try:
        rotation_decode()
    except rospy.ROSInterruptException:
        pass
    main()
