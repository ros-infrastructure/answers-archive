This script is implemented to control the Ar done to follow waypoints. I have a problem with the poseCallback function it's not reading the position properly.

    #!/usr/bin/env python
    import rospy
    from geometry_msgs.msg import Twist,Point,Pose
    import math
    import time
    from std_msgs.msg import Empty
    from gazebo_msgs.msg import ModelStates
    from tf.transformations import euler_from_quaternion

    x = 0
    y = 0
    theta = 0



    def poseCallback(msg):
        global x
        global y
        global theta

        x = msg.pose[1].position.x
        y = msg.pose[1].position.y
        rot_q = msg.pose[1].orientation
        (roll, pitch, theta) = euler_from_quaternion([rot_q.x, rot_q.y, rot_q.z, rot_q.w])


    def go_to_goal(x_goal,y_goal):
        global x
        global y,theta

        velocity_message=Twist()
        rate = rospy.Rate(10) # 10hz

        while True:
            k_linear=0.5
            rate.sleep()
            distance = abs(math.sqrt((x_goal-x)**2+(y_goal-y)**2))
            linear_speed = distance * k_linear

            k_angular = 4.0
            desired_angle_goal = math.atan2(y_goal-y, x_goal-x)
            angular_speed = (desired_angle_goal-theta)*k_angular

            velocity_message.linear.x = linear_speed
            velocity_message.angular.z = angular_speed

            velocity_publisher.publish(velocity_message)
            print 'x=' , x , 'y=' , y , 'theta =', theta

            if (distance<0.01):
                time.sleep(5)               #sleep for 5 seconds
                break

    if __name__ == '__main__':
        try:

            rospy.init_node('motion_pose',anonymous = True)


            velocity_publisher = rospy.Publisher('/cmd_vel',Twist,queue_size=1)
            sub = rospy.Subscriber("/gazebo/model_states",ModelStates,poseCallback)

            time.sleep(5)

            go_to_goal(12.0,10.1)

        except rospy.ROSInterruptException:
            rospy.loginfo("node terminated.")
