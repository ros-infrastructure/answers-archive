macro(get_vcpkg)

        CONFIGURE_COMMAND ""

        UPDATE_COMMAND ""
        BUILD_COMMAND "${CMAKE_CURRENT_BINARY_DIR}/vcpkg-prefix/src/vcpkg/bootstrap-vcpkg.sh"

endmacro()
