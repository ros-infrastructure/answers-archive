Something like will do the job.

void callback(msg::type & msg)
{
  static int counter = 0;
  counter++;
  std::cout << counter << std::endl;
}


int main()
{
  ros::init(argc, argv, "node_name");

  ros::Subscriber sub = n.subscribe("your_sub_topic", 1, callback);

  ros::spin();

}
