This is my publisher node:

#include <ros/ros.h>
#include <image_transport/image_transport.h>
#include <opencv2/highgui/highgui.hpp>
//  to display images using OpenCV's simple GUI capabilities.
#include <opencv2/imgproc/imgproc.hpp>
#include <cv_bridge/cv_bridge.h>

#define TOPIC "my_image_topic"
#define PUBLISHER_NAME "my_image_pub"

int main(int argc, char** argv)
{
  std:: cout<< "Starting " << PUBLISHER_NAME << "\nPublishing image to TOPIC "<< TOPIC;
  ros::init(argc, argv, PUBLISHER_NAME);
  // Create publisher
  ros::NodeHandle nh;
  image_transport::ImageTransport it(nh);
  image_transport::Publisher pub = it.advertise(TOPIC, 1);

  //    user-specified (on the command line) color image from disk using OpenCV
  cv::Mat image = cv::imread(argv[1], CV_LOAD_IMAGE_COLOR); // Lấy ảnh từ disk
                                                            // Đưa về kiểu dữ liệu 
  cv::waitKey(30);
     //    then convert it to the ROS type (image message)
  sensor_msgs::ImagePtr msg = cv_bridge::CvImage(std_msgs::Header(), "bgr8", image).toImageMsg();
  ros::Rate loop_rate(5);

  while (nh.ok()) 
  {
    pub.publish(msg);
    ros::spinOnce();
    loop_rate.sleep();
  }
  return 0;
}
