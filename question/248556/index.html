Here is my code, CMakeLists.txt and package.xml

#include <ros/ros.h>
#include <image_transport/image_transport.h>
#include <opencv2/highgui/highgui.hpp>
#include <cv_bridge/cv_bridge.h>

void imageCallback(const sensor_msgs::ImageConstPtr& msg)
{
  try
  {
    cv::imshow("view", cv_bridge::toCvShare(msg, "bgr8")->image);
    cv::waitKey(30);
  }
  catch (cv_bridge::Exception& e)
  {
    ROS_ERROR("Could not convert from '%s' to 'bgr8'.", msg->encoding.c_str());
  }
}

int main(int argc, char **argv)
{
  ros::init(argc, argv, "image_listener");
  ros::NodeHandle nh;
  cv::namedWindow("view");
  cv::startWindowThread();
  image_transport::ImageTransport it(nh);
  image_transport::Subscriber sub = it.subscribe("/camera/rgb/image_raw", 1, imageCallback);
  ros::spin();
  cv::destroyWindow("view");
}

Not an answer, but roscpp rospy std_msgs cv_bridge message_filters pluginlib rosconsole roslib sensor_msgs image_transport tf are not DEPENDS, but CATKIN_DEPENDS. Also: you have LIBRARIES ${PROJECT_NAME}, but you don't actually build any libraries.

Asked by gvdhoorn on 2016-11-21 10:41:57 UTC

Finally: as far as I know, OpenCV is not a Catkin package, so you can't / shouldn't add it to the find_package(.. COMPONENTS .. ) list.
