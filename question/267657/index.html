catkin_package(
  INCLUDE_DIRS include

    geometry_msgs
    roscpp
    std_msgs
    tf2_geometry_msgs

include_directories(
  include
  ${catkin_INCLUDE_DIRS}

  clean_viz
  ${catkin_LIBRARIES}
