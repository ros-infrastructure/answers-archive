int main(int argc, char **argv)

  if (pcl::io::loadPCDFile<pcl::PointXYZ> ("/home/user/Desktop/Lidar-0.pcd",*cloud) == -1)

  std::cout<<"Loaded"

  while (ros::ok())

    {

      pcl::toROSMsg(*cloud, output);
