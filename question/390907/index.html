Currently, I am creating a c ++ client node using the following Python Serviceserver. The purpose is to open and close the gripper.Is it possible to use this python file with c ++ only by creating a C ++ client node as described in the ROSWiki service communication tutorial?I edited CMakeLists.txt and package.xml by referring to the tutorial and other articles. I'm learning ROS. Can you tell me if anyone can do it as easily as possible?I was able to confirm the operation of this service node alone.I only want to use "switchgripper_status".

mycobotcommunication/scripts/mycobotservices.py

    #!/usr/bin/env python2
    import time
    import rospy
    from mycobot_communication.srv import *

    from pymycobot.mycobot import MyCobot

    mc = None


    def create_handle():
        global mc
        rospy.init_node("mycobot_services")
        rospy.loginfo("start ...")
        port = rospy.get_param("~port")
        baud = rospy.get_param("~baud")
        rospy.loginfo("%s,%s" % (port, baud))
        mc = MyCobot(port, baud)


    def create_services():
        rospy.Service("switch_gripper_status", GripperStatus, switch_status)
        rospy.Service("switch_pump_status", PumpStatus, toggle_pump)
        rospy.loginfo("ready")
        rospy.spin()



    def switch_status(req):
        if mc:
            if req.Status:
                mc.set_gripper_state(0, 80)
            else:
                mc.set_gripper_state(1, 80)

        return GripperStatusResponse(True)



    if __name__ == "__main__":
        # print(MyCobot.__dict__)
        create_handle()
        output_robot_message()
        create_services()

mycobot_communication/scripts/client.py

#!/usr/bin/env python2
import time
import rospy
from mycobot_communication.srv import *

from pymycobot.mycobot import MyCobot


if __name__ == "__main__":
    rospy.wait_for_service("switch_gripper_status")
    try:
        service_call = rospy.ServiceProxy("switch_gripper_status",GripperStatus)
        service_call(True)
    except rospy.ServiceException, e:
        print ("Service call failed: %s" % e)
