trajectory_msgs::JointTrajectory l_traj_goal;

for(unsigned int i=0; i < l_gpik_res.solution.joint_state.name.size(); i ++)

l_traj_goal.points.positions[i] = l_gpik_res.solution.joint_state.position[i];
