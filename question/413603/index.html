windows library linking issue

Hello, I am trying to make a ros package running on windows that uses the xinput library to get input from an xbox controller. I got a test program working with gcc and the xinput library works with no issues, but when I tried to use the library with ros2 and added it to my CMakeLists the package will not build.

This is the cmake file for the test code that works:

cmake_minimum_required(VERSION 3.24)
project(untitled4)
set(CMAKE_CXX_STANDARD 17)
set(XINPUT_LIBRARY "C:/Windows/System32/Xinput1_4.dll")
add_executable(untitled4 main.cpp)
target_link_libraries(untitled4 ${XINPUT_LIBRARY})

And this is the code that works:

#include <windows.h>
#include <xinput.h>
#include <iostream>
#include <cmath>

int main() {
    // Initialize XInput
    XINPUT_STATE state;
    ZeroMemory(&state, sizeof(XINPUT_STATE));
    DWORD result = XInputGetState(0, &state);
    if (result != ERROR_SUCCESS) {
        std::cerr << "Failed to initialize XInput no controller detected\n";
        return 1;
    }

    while (true) {
        result = XInputGetState(0, &state);
        if (result == ERROR_SUCCESS) {
            std::cout << "r: " << state.Gamepad.bRightTrigger << "  l: " << state.Gamepad.bLeftTrigger << std::endl;
        } else {
            std::cerr << "Failed to get XInput state\n";
            break;
        }
    }
    return 0;
}
