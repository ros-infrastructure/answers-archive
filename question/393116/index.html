bool YuMiController::init(abb::robot::EGMHardwareInterface* robot_hw, ros::NodeHandle& nh) {    

// get joint names from the parameter server
if (!nh.getParam("joint_names", joint_names_)) {
    ROS_ERROR("yumi_control: Could not parse joint names");
}

num_joints_ = joint_names_.size();

// Check Joint number
if (num_joints_ < 1) {
    ROS_ERROR_STREAM("yumi_control: Zero joint names");
    return false;
}

velocity_joint_interface_ = robot_hw->get<hardware_interface::VelocityJointInterface>();
// position_joint_interface_ = robot_hw->get<hardware_interface::PositionJointInterface>();

for (size_t i = 0; i < num_joints_; ++i)
{   
    try
    {
        velocity_joint_handles_[i] = velocity_joint_interface_->getHandle(joint_names_[i]);
        // position_joint_handles_[i] = position_joint_interface_->getHandle(joint_names_[i]);
    }
    catch (const hardware_interface::HardwareInterfaceException &ex)
    {
        ROS_ERROR_STREAM(
            "yumi_control: Exception getting joint handles: " << ex.what());
        return false;
    }

    return true;

    Eigen::VectorXd q_init, qd_init;

    q_init.resize(num_joints_);
    qd_init.resize(num_joints_);

    //Get robot current state
    for (int i = 0; i < num_joints_; i++)
    {
        q_init(i)  = velocity_joint_handles_[i].getPosition();
        qd_init(i) = velocity_joint_handles_[i].getVelocity();
    }

    qp.init(nh, q_init, qd_init);

}
return true;}

There is a chance this is caused by EGM not being active yet, in which case abb_robot_driver will refuse to start (new) controllers.

If that's the case, I would expect you see an error similar to the following in the console where you started abb_robot_driver:

EGM is inactive! Refusing to start controller(s): [..]

or

EGM is active, but not in running mode (i.e. listening for command references)! Refusing to start controller(s): [..]

can you check?
