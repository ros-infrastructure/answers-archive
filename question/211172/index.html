Sensor msg to an opencv image

This question has also been migrated to Robotics StackExchange as https://robotics.stackexchange.com/questions/68044/sensor-msg-to-an-opencv-image.

I am attempting to convert a sensormsgs/Image msg into something displayable in opencv. I am using cvbridge to accomplish this conversion. When I catkin build, I get:

Here is my callback function and includes:

#include "opencv2/opencv.hpp"
#include <iostream>
#include <ros/ros.h>
#include "sensor_msgs/Image.h"
#include <cv_bridge/cv_bridge.h>
#include <sensor_msgs/image_encodings.h>

void call_back(const sensor_msgs::ImageConstPtr& msg) {

          cv_bridge::CvImagePtr cv_ptr;
          try
          {
            cv_ptr = cv_bridge::toCvCopy(msg, sensor_msgs::image_encodings::BGR8);
          }
          catch (cv_bridge::Exception& e)
          {
            ROS_ERROR("cv_bridge exception: %s", e.what());
            return;
          }

          namedWindow("please", CV_WINDOW_AUTOSIZE);
          imshow("please", cv_ptr->image);
          waitKey(5);
      }
