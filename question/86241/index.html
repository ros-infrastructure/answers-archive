Hello, I'm trying to read data from a camera stream from vrep. The publisher is running fine, rotopic echo returns nonzero data being streamed, but if I subscribe and try to print out the data i get only zeros. I think I'm doing something wrong, here's the code:

#include <ros/ros.h>
#include <sensor_msgs/Image.h>
#include "std_msgs/String.h"

void cameraROSCallback(const sensor_msgs::Image::ConstPtr& msg){
        for(int a=0; a<256; a++)
            {
            for(int b=0; b<256; b++)
                {ROS_INFO("[%3i],[%3i]: [%u]", a,b,msg->data[a,b]);}
            }
    }

int main(int argc, char **argv)
{
  ros::init(argc, argv, "image_listener");
  ros::NodeHandle n;

  ros::Subscriber sub = n.subscribe("vrep/visionSensorData", 1, cameraROSCallback);
  //ROS_INFO("I heard: [%s]", sub.getTopic().c_str());
  ros::spin();

Try this:

ROS_INFO("[%3i],[%3i]: [%u]", a,b,msg->data[a * b]);

msg->data is a one dimensional vector.
