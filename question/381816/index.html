#include <ros/ros.h>
#include <sensor_msgs/PointCloud2.h>
#include <pcl/io/pcd_io.h>
#include <pcl/visualization/pcl_visualizer.h>
#include <pcl_conversions/pcl_conversions.h>

int i;
void point_cloud_callback(const sensor_msgs::PointCloud2& input_cloud)
{
    pcl::PointCloud<pcl::PointXYZRGB>::Ptr point_cloud (new pcl::PointCloud<pcl::PointXYZRGB>);
    pcl::fromROSMsg(input_cloud, *point_cloud);
    pcl::io::savePCDFile("scene" + std::to_string(i) + ".pcd", *point_cloud);
    ++i;
}


int main(int argc, char* argv[])
{
    ros::init(argc, argv, "mkv2pcd");
    ros::NodeHandle nh;
    ros::Subscriber point_cloud_sub = nh.subscribe("/points2", 1000, point_cloud_callback);

    pcl::PointCloud<pcl::PointXYZRGB>::Ptr point_cloud(new pcl::PointCloud<pcl::PointXYZRGB>);
    pcl::io::loadPCDFile<pcl::PointXYZRGB>("scene153.pcd", *point_cloud);

    pcl::visualization::PCLVisualizer viewer("viewer");
    viewer.addPointCloud(point_cloud);

    viewer.spin ();

    ros::spin();
    return 0;
}
