This is what I did after compiling:

cd /usr/local/

sudo mkdir -p fotonic

sudo cp -r ~/Downloads/fz-linux-api_x64/include/. /usr/local/fotonic

sudo cp -r ~/Downloads/fz-linux-api_x64/lib/. /usr/local/fotonic
