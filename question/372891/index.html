bool executeTrajectory(const trajectorymsgs::JointTrajectory& trajectory)
{
  // Create a Follow Joint Trajectory Action Client
  actionlib::SimpleActionClient<controlmsgs::FollowJointTrajectoryAction> ac (
    "niryoonefollowjointtrajectorycontroller/followjoint_trajectory", true);



if (!ac.waitForServer(ros::Duration(2.0)))
  {
    ROS_ERROR("Could not connect to action server");
    return false;
  }



controlmsgs::FollowJointTrajectoryGoal goal;
  goal.trajectory = trajectory;
  goal.goaltime_tolerance = ros::Duration(1.0);



return ac.sendGoalAndWait(goal) == actionlib::SimpleClientGoalState::SUCCEEDED;
}



bool waitForSubscribers(ros::Publisher &pub, ros::Duration timeout)
{
  if (pub.getNumSubscribers() > 0)
    return true;
  ros::Time start = ros::Time::now();
  ros::Rate waitTime(0.5);
  while (ros::Time::now() - start < timeout)
  {
    waitTime.sleep();
    if (pub.getNumSubscribers() > 0)
      break;
  }
  return pub.getNumSubscribers() > 0;
}
