int main(int argc, char* argv[]) {
ros::init(argc, argv, "pub_pcl");

PointCloudT::Ptr demo (new PointCloudT);
demo->header.frame_id = "some_tf_frame";
demo->height = demo->width = 1;
demo->points.push_back (pcl::PointXYZ(1.0, 2.0, 3.0));
