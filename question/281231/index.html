    192.168.122.22 NotePC

roslaunch  myrobo_navigation amcl.launch

[4] rqt_graph

export ROS_HOSTNAME=Notepc 
export ROS_MASTER_URI=http://192.168.122.22:11311 
export ROS_IP=192.168.122.22 
rqt_graph
