Declare the lib:

#include <std_msgs/Float64MultiArray.h> //include the lib

use it:

std_msgs::Float64MultiArray Atest; //declare Atest
Atest.data.resize(5); //resize the array to assign to existent values
Atest.data[4] = 5; //put the desired value
