Hi,

The problem can by solved by adding the following lines

for (int count = 0; count < x.size(); count ++)
{
  geometry_msgs::Point tmp_p;

  tmp_p.x=x[count];
  tmp_p.y=y[count];
  line_strip.points.push_back(tmp_p);
}
