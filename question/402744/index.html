descartes_core::TrajectoryPtPtr makeCartesianPoint(const Eigen::Isometry3d& pose, double dt)
{
  using namespace descartes_core;
  using namespace descartes_trajectory;

  return TrajectoryPtPtr( new CartTrajectoryPt( TolerancedFrame(pose), TimingConstraint(dt)) );
}

descartes_core::TrajectoryPtPtr makeTolerancedCartesianPoint(const Eigen::Isometry3d& pose, double dt)
{
  using namespace descartes_core;
  using namespace descartes_trajectory;
  return TrajectoryPtPtr( new AxialSymmetricPt(pose, M_PI / 12.0, AxialSymmetricPt::Z_AXIS, TimingConstraint(dt)) );
}

std::vector<descartes_core::TrajectoryPtPtr> makePath(const std::vector<std::vector<float>>& trajectoryPath)
{

  // Inicialize output variable
  std::vector<descartes_core::TrajectoryPtPtr> result; 
  // Time between points
  const static double time_between_points = 3;

  // Getting current pos and tranform to joints
  std::vector<double> start_joints = getCurrentJointState("joint_states");
  descartes_core::TrajectoryPtPtr pt (new descartes_trajectory::JointTrajectoryPt(start_joints));

  // First point will be current pos
  result.push_back(pt);

  // Create a pose facing down
  Eigen::Isometry3d pose = Eigen::Isometry3d::Identity();
  pose *= Eigen::AngleAxisd(M_PI, Eigen::Vector3d::UnitX());

  // Moving to the input trajectory
  for (const auto& point6D : trajectoryPath)
  {
    Eigen::Isometry3d pattern_pt = Eigen::Isometry3d::Identity();
    pattern_pt.translation() = Eigen::Vector3d(point6D[0], point6D[1], point6D[2]);
    descartes_core::TrajectoryPtPtr pt = makeTolerancedCartesianPoint(pattern_pt * pose, time_between_points);
    result.push_back(pt);
  }

  // Coming back to initial pos
  result.push_back(pt);   

  return result;
}

bool executeTrajectory(const trajectory_msgs::JointTrajectory& trajectory)
{
  // Create a Follow Joint Trajectory Action Client
  actionlib::SimpleActionClient<control_msgs::FollowJointTrajectoryAction> ac ("joint_trajectory_action", true);
  if (!ac.waitForServer(ros::Duration(2.0)))
  {
    ROS_ERROR("Could not connect to action server");
    return false;
  }

  control_msgs::FollowJointTrajectoryGoal goal;
  goal.trajectory = trajectory;
  goal.goal_time_tolerance = ros::Duration(0.1);

  return ac.sendGoalAndWait(goal) == actionlib::SimpleClientGoalState::SUCCEEDED;
}
