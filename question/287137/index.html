<!-- 
    ROS navigation stack with velocity smoother and safety (reactive) controller
-->
<launch>


  <arg name="initial_pose_x" default="0.0"/>
  <arg name="initial_pose_y" default="0.0"/>
  <arg name="initial_pose_a" default="0.0"/>
  <arg name="namespace" default="tf" />
  <arg name="custom_amcl_launch_file" default="/home/sk4ll/leia_robot/asus_xtion_pro_amcl.launch.xml"/>

  <include file="$(arg custom_amcl_launch_file)">
        <arg name="tf_prefix" value="$(arg namespace)"/>
    <arg name="scan_topic" value="scan" />
    <arg name="use_map_topic" value="true" />
    <arg name="initial_pose_x" value="$(arg initial_pose_x)" />
    <arg name="initial_pose_y" value="$(arg initial_pose_y)" />
    <arg name="initial_pose_a" value="$(arg initial_pose_a)" />
    <arg name="global_frame_id" value="/map"/>

  </include>

  <include file="$(find turtlebot_navigation)/launch/includes/velocity_smoother.launch.xml"/>
  <include file="$(find turtlebot_navigation)/launch/includes/safety_controller.launch.xml"/>

  <arg name="odom_frame_id"   default="odom"/>
  <arg name="base_frame_id"   default="base_footprint"/>
  <arg name="global_frame_id" default="/map"/>
  <arg name="odom_topic" default="odom" />
  <arg name="laser_topic" default="scan" />
  <arg name="custom_param_file" default="/home/sk4ll/leia_robot/param/dummy.yaml"/>
  <arg name="3d_sensor" default="$(env TURTLEBOT_3D_SENSOR)"/>  <!-- r200, kinect, asus_xtion_pro -->



  <node pkg="move_base" type="move_base" respawn="false" name="move_base" output="screen">
    <remap from="map" to="/map" />
    <rosparam file="/home/sk4ll/leia_robot/param/costmap_common_params.yaml" command="load" ns="global_costmap" />
    <rosparam file="/home/sk4ll/leia_robot/param/costmap_common_params.yaml" command="load" ns="local_costmap" />   
    <rosparam file="/home/sk4ll/leia_robot/param/local_costmap_params.yaml" command="load" />   
    <rosparam file="/home/sk4ll/leia_robot/param/global_costmap_params.yaml" command="load" />
    <rosparam file="/home/sk4ll/leia_robot/param/dwa_local_planner_params.yaml" command="load" />
    <rosparam file="/home/sk4ll/leia_robot//param/move_base_params.yaml" command="load" />
    <rosparam file="/home/sk4ll/leia_robot/param/global_planner_params.yaml" command="load" />
    <rosparam file="/home/sk4ll/leia_robot/param/navfn_global_planner_params.yaml" command="load" />

     <!-- external params file that could be loaded into the move_base namespace -->
     <rosparam file="$(arg custom_param_file)" command="load" /> 

   <param name="global_costmap/global_frame" value="$(arg global_frame_id)"/>
    <param name="global_costmap/robot_base_frame" value="$(arg base_frame_id)"/>
    <param name="local_costmap/global_frame" value="$(arg odom_frame_id)"/>
    <param name="local_costmap/robot_base_frame" value="$(arg base_frame_id)"/>
    <param name="DWAPlannerROS/global_frame_id" value="$(arg odom_frame_id)"/>

    <remap from="cmd_vel" to="navigation_velocity_smoother/raw_cmd_vel"/>
