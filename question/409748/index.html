<link name="${link}" />
<gazebo>
  <plugin name="imu_plugin" filename="libhector_gazebo_ros_imu.so">
    <updateRate>${update_rate}</updateRate>
    <bodyName>${link}</bodyName>
    <frameId>${link}</frameId>
    <topicName>${imu_topic}</topicName>
