For completeness, this is my CMakeLists.txt file

cmake_minimum_required(VERSION 3.5)
project(fancy_pkg)

# Default to C++14
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 14)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Fancy.msg"
  DEPENDENCIES builtin_interfaces
)

target_include_directories(fancy_lib PUBLIC include)
ament_target_dependencies(fancy_lib rclcpp)
rosidl_target_interfaces(fancy_lib ${PROJECT_NAME} "rosidl_typesupport_cpp")

install(TARGETS fancy_lib
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME}
)

install(DIRECTORY include/
  DESTINATION include/
)

if (BUILD_TESTING)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(test_fancy_stuff test/fancy_stuff.cpp)
    target_link_libraries(test_fancy_stuff fancy_lib)
endif()

ament_export_include_directories(include)
ament_export_libraries(fancy_lib)
ament_export_dependencies(rclcpp)
ament_export_dependencies(rosidl_default_runtime)

ament_package()
