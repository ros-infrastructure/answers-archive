rosidl_generate_interfaces SONAME

How do I override the default SONAME given to the shared objects generated by rosidl_generate_interfaces? I would like to put versioning information in the shared objects I create.

Answers

This CMakeLists.txt snippet seems to do what I want:

file(
  GLOB_RECURSE INTERFACE_FILES
  RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
  ./action/*.action
  ./msg/*.msg
  ./srv/*.srv
)

rosidl_generate_interfaces(
  ${PROJECT_NAME} ${INTERFACE_FILES}
  DEPENDENCIES action_msgs builtin_interfaces sensor_msgs
)

set_target_properties(
  ${PROJECT_NAME}__python
  ${PROJECT_NAME}__rosidl_generator_c
  ${PROJECT_NAME}__rosidl_typesupport_c
  ${PROJECT_NAME}__rosidl_typesupport_c__pyext
  ${PROJECT_NAME}__rosidl_typesupport_cpp
  ${PROJECT_NAME}__rosidl_typesupport_introspection_c
  ${PROJECT_NAME}__rosidl_typesupport_introspection_c__pyext
  ${PROJECT_NAME}__rosidl_typesupport_introspection_cpp
  PROPERTIES
  VERSION ${PROJECT_VERSION}
  SOVERSION ${PROJECT_VERSION_MAJOR}
)
