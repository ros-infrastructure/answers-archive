callback does not update values

here is my code:

#include "ros/ros.h"
#include "turtlesim/Pose.h"

float x;

void callback(const turtlesim::Pose::ConstPtr& msg)
{
    x=msg->x;
}

int main(int argc, char **argv)
{
    ros::init(argc,argv, "turtletest");
    ros::NodeHandle n;
    ros::Subscriber sub=n.subscribe("/turtle1/pose",100,callback);
    ros::Rate loop(10);
    std::cout<<x<<std::endl;    //DOESNT PRINT UPDATED VALUE
    ros::spin();
    return 0;
}

when I print out x, it should show the default turtlesim value of 5.44444, but it shows 0
