If want to know the duration of a taks just perform a difference:

ros::Time begin = ros::Time::now();
while(...)
{
  //Do some work
}
ros::Time end = ros::Time::now();
ros::Duration time_diff = end - begin;
std::cout << "Seconds: " <<  time_diff.toSec() << std::endl;
