ros::WallTime can be used as follows

ros::WallTime start_, end_;

start_ = ros::WallTime::now();

// **** do something here... ****

end_ = ros::WallTime::now();

// print results
double execution_time = (end_ - start_).toNSec() * 1e-6;
ROS_INFO_STREAM("Exectution time (ms): " << execution_time);
