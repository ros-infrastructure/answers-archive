rosbridge_websocket Unable to import

Hi, im having the following issue when calling an action server from rosbridge_suite: SO Ubuntu 20.04 ROS2 Galactic

Unable to import actionserverinterfaces.msg from package actionserverinterfaces. Caused by: No module named 'actionserverinterfaces.msg'

Im trying to figure it out why the rosbridgewebsocket is trying to import "actionserver_interfaces.msg"? The interface is not define as a message.

My javascript code in case it helps:

var actionCustomClient = new ROSLIB.ActionClient({
    ros: ros,
    serverName: "/move2_preset",
    actionName: "action_server_interfaces/action/Move2Preset"
});

var goal = new ROSLIB.Goal({
    actionClient: actionCustomClient,
    goalMessage: {
        goal_pose: "init"
    }
});

goal.on('feedback', function (feedback) {
    document.getElementById("actionResult").innerHTML = JSON.stringify(feedback);
});

goal.on('result', function (result) {
    document.getElementById("actionFeedback").innerHTML = JSON.stringify(result);
});

goal.send();
